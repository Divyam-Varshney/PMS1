   454→// — Revenue trend badge — up/down/neutral % vs previous period —
   455→function RevenueTrendBadge({
   456→  series,
   457→}: {
   458→  series: Array<{ date: string; revenue: number; orders: number }>;
   459→}) {
   460→  if (!series || series.length < 2) {
   461→    return (
   462→      <Badge variant="outline" className="gap-1 text-xs">
   463→        <Minus className="size-3 text-muted-foreground" />
   464→        No trend
   465→      </Badge>
   466→    );
   467→  }
   468→
   469→  // Compare last 3 days vs previous 3 days (more stable than 1 vs 1)
   470→  const recent = series.slice(-3).reduce((s, d) => s + d.revenue, 0);
   471→  const previous = series.slice(-6, -3).reduce((s, d) => s + d.revenue, 0);
   472→
   473→  if (previous === 0) {
   474→    return (
   475→      <Badge variant="outline" className="gap-1 text-xs bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900/50">
   476→        <Activity className="size-3" />
   477→        Active
   478→      </Badge>
   479→    );
   480→  }
   481→
   482→  const pct = ((recent - previous) / previous) * 100;
   483→  const isUp = pct >= 0;
   484→  const isFlat = Math.abs(pct) < 1;
   485→
   486→  return (
   487→    <Badge
   488→      variant="outline"
   489→      className={cn(
   490→        "gap-1 text-xs font-medium",
   491→        isFlat
   492→          ? "bg-muted text-muted-foreground border-border"
   493→          : isUp
   494→            ? "bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900/50"
   495→            : "bg-rose-50 text-rose-700 border-rose-200 dark:bg-rose-950/40 dark:text-rose-300 dark:border-rose-900/50"
   496→      )}
   497→    >
   498→      {isFlat ? (
   499→        <Minus className="size-3" />
   500→      ) : isUp ? (
   501→        <TrendingUp className="size-3" />
   502→      ) : (
   503→        <TrendingDown className="size-3" />
   504→      )}
   505→      {isFlat ? "Flat" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   506→    </Badge>
   507→  );
   508→}

   510→// — Revenue sparkline — pure SVG, no chart library —
   511→function RevenueSparkline({
   512→  series,
   513→}: {
   514→  series: Array<{ date: string; revenue: number; orders: number }>;
   515→}) {
   516→  if (!series || series.length === 0) {
   517→    return (
   518→      <div className="flex h-[80px] items-center justify-center text-xs text-muted-foreground">
   519→        No revenue data yet
   520→      </div>
   521→    );
   522→  }
   523→
   524→  const W = 600;
   525→  const H = 80;
   526→  const PAD = 4;
   527→  const max = Math.max(...series.map((d) => d.revenue), 1);
   528→  const min = Math.min(...series.map((d) => d.revenue), 0);
   529→  const range = max - min || 1;
   530→
   531→  const points = series.map((d, i) => {
   532→    const x = PAD + (i / (series.length - 1)) * (W - PAD * 2);
   533→    const y = H - PAD - ((d.revenue - min) / range) * (H - PAD * 2);
   534→    return [x, y] as const;
   535→  });
   536→
   537→  // Smooth path using simple line segments (clean + minimal)
   538→  const linePath = points
   539→    .map(([x, y], i) => (i === 0 ? `M ${x} ${y}` : `L ${x} ${y}`))
   540→    .join(" ");
   541→  const areaPath = `${linePath} L ${points[points.length - 1][0]} ${H - PAD} L ${points[0][0]} ${H - PAD} Z`;
   542→
   543→  return (
   544→    <div className="w-full">
   545→      <svg
   546→        viewBox={`0 0 ${W} ${H}`}
   547→        preserveAspectRatio="none"
   548→        className="w-full h-[80px]"
   549→      >
   550→        <defs>
   551→          <linearGradient id="revSparkGrad" x1="0" y1="0" x2="0" y2="1">
   552→            <stop offset="0%" stopColor="#10b981" stopOpacity={0.25} />
   553→            <stop offset="100%" stopColor="#10b981" stopOpacity={0} />
   554→          </linearGradient>
   555→        </defs>
   556→        <path d={areaPath} fill="url(#revSparkGrad)" />
   557→        <path
   558→          d={linePath}
   559→          fill="none"
   560→          stroke="#10b981"
   561→          strokeWidth={2}
   562→          strokeLinejoin="round"
   563→          strokeLinecap="round"
   564→          vectorEffect="non-scaling-stroke"
   565→        />
   566→        {points.map(([x, y], i) => (
   567→          <circle
   568→            key={i}
   569→            cx={x}
   570→            cy={y}
   571→            r={2.5}
   572→            fill="#10b981"
   573→            vectorEffect="non-scaling-stroke"
   574→          />
   575→        ))}
   576→      </svg>
