    95→    95→// ---------------------------------------------------------------------------
    96→    96→// Key auto-generation: "Cash on Delivery" → "cod", "UPI" → "upi",
    97→    97→// "Razorpay" → "razorpay", "QR Code" → "qc". Multi-word → initials joined;
    98→    98→// single-word → lowercased whole word. Special-cased common terms so
    99→    99→// "QR Code" → "qr" (more conventional than "qc").
   100→   100→// ---------------------------------------------------------------------------
   101→   101→function generateKey(label: string): string {
   102→   102→  const trimmed = label.trim();
   103→   103→  if (!trimmed) return "";
   104→   104→  const words = trimmed.split(/\s+/).filter(Boolean);
   105→   105→  if (words.length === 1) return words[0].toLowerCase();
   106→   106→  // Common multi-word payment terms — use conventional short forms.
   107→   107→  const lower = trimmed.toLowerCase();
   108→   108→  if (lower.includes("qr")) return "qr";
   109→   109→  if (lower.includes("upi")) return "upi";
   110→   110→  if (lower.includes("cash") && lower.includes("delivery")) return "cod";
   111→   111→  if (lower.includes("net banking")) return "netbanking";
   112→   112→  return words.map((w) => w[0]).join("").toLowerCase();
   113→   113→}

   115→   115→/** Safe JSON parse of PaymentMethod.config — never throws. */
   116→   116→function parseConfig<T = Record<string, unknown>>(raw?: string | null): T {
   117→   117→  if (!raw) return {} as T;
   118→   118→  try {
   119→   119→    return JSON.parse(raw) as T;
   120→   120→  } catch {
   121→   121→    return {} as T;
   122→   122→  }
   123→   123→}

   221→   221→  async function save() {
   222→   222→    if (!form.label.trim()) {
   223→   223→      toast.error("Label is required");
   224→   224→      return;
   225→   225→    }
   226→   226→    if (!form.key.trim()) {
   227→   227→      toast.error("Key is required");
   228→   228→      return;
   229→   229→    }
   230→   230→    // Razorpay-specific validation — require both fields IF the admin is
   231→   231→    // enabling the method OR entering any value. Allows saving an empty
   232→   232→    // disabled Razorpay method (the seeded default).
   233→   233→    if (form.key === "razorpay" && form.isActive) {
   234→   234→      if (!razorpayCreds.keyId?.trim() || !razorpayCreds.keySecret?.trim()) {
   235→   235→        toast.error("Razorpay Key ID and Key Secret are required to enable this method");
   236→   236→        return;
   237→   237→      }
   238→   238→    }
   239→   239→    setSaving(true);
   240→   240→    const payload = {
   241→   241→      label: form.label.trim(),
   242→   242→      description: form.description.trim(),
   243→   243→      icon: form.icon.trim(),
   244→   244→      gateway: form.gateway.trim(),
   245→   245→      config: buildConfig(),
   246→   246→      displayOrder: parseInt(form.displayOrder, 10) || 0,
   247→   247→      isActive: form.isActive,
   248→   248→    };
   249→   249→    const r = editing
   250→   250→      ? await run(() => api.patch(`/api/admin/payment-methods/${editing.id}`, payload), {
   251→   251→          success: "Payment method updated",
   252→   252→          error: "Update failed",
   253→   253→        })
   254→   254→      : await run(
   255→   255→          () => api.post("/api/admin/payment-methods", { key: form.key.trim(), ...payload }),
   256→   256→          { success: "Payment method created", error: "Create failed" }
   257→   257→        );
   258→   258→    setSaving(false);
   259→   259→    if (r) {
   260→   260→      setOpen(false);
   261→   261→      qc.invalidateQueries({ queryKey: ["admin-payment-methods"] });
   262→   262→    }
   263→   263→  }

   287→   287→  // --- QR image upload ------------------------------------------------------
   288→   288→  async function onQrImagePicked(e: React.ChangeEvent<HTMLInputElement>) {
   289→   289→    const file = e.target.files?.[0];
   290→   290→    if (!file || !editing) return;
   291→   291→    // Basic client-side validation — server re-validates.
   292→   292→    if (!file.type.startsWith("image/")) {
   293→   293→      toast.error("Please upload an image file");
   294→   294→      return;
   295→   295→    }
   296→   296→    if (file.size > 4 * 1024 * 1024) {
   297→   297→      toast.error("Image must be under 4MB");
   298→   298→      return;
   299→   299→    }
   300→   300→    const fd = new FormData();
   301→   301→    fd.append("file", file);
   302→   302→    const r = await run(
   303→   303→      () => api.upload<{ qrImage: string }>(`/api/admin/payment-methods/${editing.id}/qr-image`, fd),
   304→   304→      { success: "QR image uploaded", error: "Upload failed" }
   305→   305→    );
   306→   306→    if (r) {
   307→   307→      setQrImage(r.qrImage);
   308→   308→      qc.invalidateQueries({ queryKey: ["admin-payment-methods"] });
   309→   309→    }
   310→   310→    // Reset file input so the same file can be re-uploaded if needed.
   311→   311→    if (qrFileInputRef.current) qrFileInputRef.current.value = "";
   312→   312→  }
