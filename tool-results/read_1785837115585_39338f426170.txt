   527→/// Notification template defaults. Seeded into NotificationTemplate table.
   528→///
   529→/// All template bodies are generated through `darkEmailTemplate()` so they
   530→/// share a consistent premium DARK theme:
   531→///   • Slate-900 page background (#0f172a) + slate-800 card (#1e293b)
   532→///   • Emerald→teal gradient header (#059669 → #0d9488) with store name
   533→///   • Inline CSS only (Gmail/Outlook strip <style> blocks)
   534→///   • Table-based 600px layout — the most reliable pattern for cross-client
   535→///     rendering across Gmail / Outlook / Apple Mail / Yahoo
   536→///   • System font stack so no external font loads are required
   537→///   • Dark-mode meta tags (color-scheme + supported-color-schemes + theme-color)
   538→///     so email clients that respect OS dark-mode preferences render correctly
   539→///   • Footer with store contact info on a darker slate band
   540→/// All {{variables}} are replaced at send time by renderTemplate().
   541→function darkEmailTemplate(opts: { eyebrow: string; content: string }): string {
   542→  return `<!DOCTYPE html>
   543→<html lang="en" xmlns="http://www.w3.org/1999/xhtml">
   544→<head>
   545→  <meta charset="UTF-8" />
   546→  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
   547→  <meta name="color-scheme" content="dark" />
   548→  <meta name="supported-color-schemes" content="dark" />
   549→  <meta name="theme-color" content="#0f172a" />
   550→  <title>${opts.eyebrow}</title>
