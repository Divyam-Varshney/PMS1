    17→//     delivery") require the full phrase to appear in the query.

   348→/**
   349→ * Tokenize a query into lowercase words (split on whitespace + punctuation).
   350→ * "How to track my order?" → ["how", "to", "track", "my", "order"]
   351→ */
   352→function tokenize(query: string): string[] {
   353→  return query
   354→    .toLowerCase()
   355→    .split(/[^a-z0-9]+/g)
   356→    .filter((t) => t.length > 1);
   357→}
   358→
   359→/**
   360→ * Find the best-matching FAQ entry for a user's query.
   361→ *
   362→ * Algorithm:
   363→ *   - For each FAQ entry, score it by counting how many of its keywords
   364→ *     appear in the query (either as a substring or as whole tokens).
   365→ *   - Multi-word keywords require the full phrase to be a substring of the
   366→ *     query (e.g. "cash on delivery" must appear verbatim).
   367→ *   - Return the entry with the highest score, or null if no entry scored > 0.
   368→ *   - Ties are broken by array order (first entry wins).
   369→ *
   370→ * Returns `{ faq, score }` or `null`.
   371→ */
   372→export function matchFaq(query: string): { faq: PharmacyFaq; score: number } | null {
   373→  if (!query || typeof query !== "string") return null;
   374→  const q = query.toLowerCase();
   375→  const tokens = new Set(tokenize(query));
   376→
   377→  let best: { faq: PharmacyFaq; score: number } | null = null;
   378→
   379→  for (const faq of PHARMACY_FAQS) {
   380→    let score = 0;
   381→    for (const kw of faq.keywords) {
   382→      const k = kw.toLowerCase();
   383→      if (k.includes(" ")) {
   384→        // Multi-word keyword — require verbatim substring match.
   385→        if (q.includes(k)) score += 3;
   386→      } else {
   387→        // Single-word keyword — match if it appears as a whole token OR
   388→        // as a substring of any token (handles plurals like "delivery"/"deliveries").
   389→        if (tokens.has(k)) score += 2;
   390→        else if (q.includes(k)) score += 1;
   391→      }
   392→    }
   393→    if (score > 0 && (!best || score > best.score)) {
   394→      best = { faq, score };
   395→    }
   396→  }
   397→
   398→  // Require a minimum score of 2 to avoid false positives on tiny queries.
   399→  if (!best || best.score < 2) return null;
   400→  return best;
   401→}
