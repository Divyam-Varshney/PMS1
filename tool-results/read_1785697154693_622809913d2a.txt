   122→   122→export function OrdersView() {
   123→   123→  const navigate = useAdminStore((s) => s.navigate);
   124→   124→  const qc = useQueryClient();
   125→   125→  const [search, setSearch] = useState("");
   126→   126→  const [debouncedSearch, setDebouncedSearch] = useState("");
   127→   127→  const [statuses, setStatuses] = useState<Set<string>>(
   128→   128→    () => new Set(DEFAULT_STATUS_PRESET)
   129→   129→  );
   130→   130→  const [paymentStatuses, setPaymentStatuses] = useState<Set<string>>(new Set());
   131→   131→  const [paymentMethod, setPaymentMethod] = useState("all");
   132→   132→  const [from, setFrom] = useState("");
   133→   133→  const [to, setTo] = useState("");
   134→   134→  const [rxOnly, setRxOnly] = useState(false);
   135→   135→  const [notesOnly, setNotesOnly] = useState(false);
   136→   136→  const [compact, setCompact] = useState(false);
   137→   137→  const [page, setPage] = useState(1);
   138→   138→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   139→   139→  const [bulkBusy, setBulkBusy] = useState(false);
   140→   140→  const pageSize = 20;
   141→   141→
   142→   142→  // --------------------- Debounced search (300ms) ---------------------
   143→   143→  useEffect(() => {
   144→   144→    const t = setTimeout(() => setDebouncedSearch(search), 300);
   145→   145→    return () => clearTimeout(t);
   146→   146→  }, [search]);
   147→   147→
   148→   148→  // Reset page to 1 whenever any filter changes (so we don't end up on a
   149→   149→  // page that doesn't exist for the new filter set).
   150→   150→  useEffect(() => {
   151→   151→    setPage(1);
   152→   152→  }, [debouncedSearch, statuses, paymentStatuses, paymentMethod, from, to, rxOnly, notesOnly]);
   153→   153→
   154→   154→  const queryParams = useMemo(() => {
   155→   155→    const p = new URLSearchParams({
   156→   156→      page: String(page),
   157→   157→      pageSize: String(pageSize),
   158→   158→    });
   159→   159→    if (debouncedSearch.trim()) p.set("search", debouncedSearch.trim());
   160→   160→    if (statuses.size > 0) p.set("statuses", Array.from(statuses).join(","));
   161→   161→    if (paymentStatuses.size > 0) p.set("paymentStatuses", Array.from(paymentStatuses).join(","));
   162→   162→    if (paymentMethod !== "all") p.set("paymentMethod", paymentMethod);
   163→   163→    if (from) p.set("from", from);
   164→   164→    if (to) p.set("to", to);
   165→   165→    if (rxOnly) p.set("prescriptionRequired", "true");
   166→   166→    if (notesOnly) p.set("hasNotes", "true");
   167→   167→    return p.toString();
   168→   168→  }, [debouncedSearch, statuses, paymentStatuses, paymentMethod, from, to, rxOnly, notesOnly, page]);
   169→   169→
   170→   170→  const { data, isLoading } = useQuery({
   171→   171→    queryKey: ["admin-orders", queryParams],
   172→   172→    queryFn: () =>
   173→   173→      api.get<{ items: OrderListItem[]; total: number; totalPages: number; page: number }>(
   174→   174→        `/api/admin/orders?${queryParams}`
   175→   175→      ),
   176→   176→  });
   177→   177→
   178→   178→  const { data: stats } = useQuery<Stats>({
   179→   179→    queryKey: ["admin-orders-stats"],
   180→   180→    queryFn: () => api.get<Stats>("/api/admin/orders/stats"),
   181→   181→    refetchInterval: 60_000, // refresh stats every minute
   182→   182→  });
   183→   183→
   184→   184→  const items = data?.items ?? [];
   185→   185→
   186→   186→  // --------------------- Selection helpers ---------------------
   187→   187→  const allSelected = items.length > 0 && items.every((o) => selectedIds.has(o.id));
   188→   188→  const someSelected = items.some((o) => selectedIds.has(o.id));
   189→   189→
   190→   190→  function toggleAll() {
   191→   191→    const next = new Set(selectedIds);
   192→   192→    if (allSelected) items.forEach((o) => next.delete(o.id));
   193→   193→    else items.forEach((o) => next.add(o.id));
   194→   194→    setSelectedIds(next);
   195→   195→  }
   196→   196→  function toggleOne(id: string) {
   197→   197→    const next = new Set(selectedIds);
   198→   198→    if (next.has(id)) next.delete(id);
   199→   199→    else next.add(id);
   200→   200→    setSelectedIds(next);
   201→   201→  }
   202→   202→  function clearSelection() {
   203→   203→    setSelectedIds(new Set());
   204→   204→  }
   205→   205→
   206→   206→  const headerChecked: boolean | "indeterminate" = allSelected
   207→   207→    ? true
   208→   208→    : someSelected
   209→   209→      ? "indeterminate"
   210→   210→      : false;
   211→   211→
   212→   212→  // --------------------- Filter helpers ---------------------
   213→   213→  function toggleStatus(s: string) {
   214→   214→    const next = new Set(statuses);
   215→   215→    if (next.has(s)) next.delete(s);
   216→   216→    else next.add(s);
   217→   217→    setStatuses(next);
   218→   218→  }
   219→   219→  function togglePaymentStatus(s: string) {
   220→   220→    const next = new Set(paymentStatuses);
   221→   221→    if (next.has(s)) next.delete(s);
   222→   222→    else next.add(s);
   223→   223→    setPaymentStatuses(next);
   224→   224→  }
   225→   225→  function clearAllFilters() {
   226→   226→    setStatuses(new Set());
   227→   227→    setPaymentStatuses(new Set());
   228→   228→    setPaymentMethod("all");
   229→   229→    setFrom("");
   230→   230→    setTo("");
   231→   231→    setRxOnly(false);
   232→   232→    setNotesOnly(false);
   233→   233→    setSearch("");
   234→   234→    setDebouncedSearch("");
   235→   235→  }
   236→   236→
   237→   237→  const activeFilterCount =
   238→   238→    statuses.size + paymentStatuses.size + (paymentMethod !== "all" ? 1 : 0) +
   239→   239→    (from ? 1 : 0) + (to ? 1 : 0) + (rxOnly ? 1 : 0) + (notesOnly ? 1 : 0) +
   240→   240→    (debouncedSearch.trim() ? 1 : 0);
   241→   241→
   242→   242→  // --------------------- Export CSV ---------------------
   243→   243→  function exportCsv(selectedOnly = false) {
   244→   244→    const shouldExportSelected = selectedOnly && selectedIds.size > 0;
   245→   245→    const url = shouldExportSelected
   246→   246→      ? `/api/admin/orders/export?ids=${Array.from(selectedIds).join(",")}`
   247→   247→      : `/api/admin/orders/export?${queryParams}`;
   248→   248→    const a = document.createElement("a");
   249→   249→    a.href = url;
   250→   250→    a.download = `orders-${new Date().toISOString().slice(0, 10)}.csv`;
   251→   251→    document.body.appendChild(a);
   252→   252→    a.click();
   253→   253→    a.remove();
   254→   254→    toast.success(
   255→   255→      shouldExportSelected
   256→   256→        ? `Exporting ${selectedIds.size} selected order(s)`
   257→   257→        : "Exporting all matching orders"
   258→   258→    );
   259→   259→  }
   260→   260→
   261→   261→  // --------------------- Print invoices for selected ---------------------
   262→   262→  // Sequentially fetches each invoice PDF and triggers a print. Browsers
   263→   263→  // usually collapse multiple print() calls into one job per page — we
   264→   264→  // open each invoice in a hidden iframe and print it.
   265→   265→  async function printInvoices() {
   266→   266→    if (selectedIds.size === 0) return;
   267→   267→    const ids = Array.from(selectedIds);
   268→   268→    toast.info(`Preparing ${ids.length} invoice(s) for printing...`);
   269→   269→    for (const id of ids) {
   270→   270→      try {
   271→   271→        const buf = await api.raw(`/api/admin/orders/${id}/invoice`);
   272→   272→        const blob = new Blob([buf], { type: "application/pdf" });
   273→   273→        const url = URL.createObjectURL(blob);
   274→   274→        const iframe = document.createElement("iframe");
   275→   275→        iframe.style.display = "none";
   276→   276→        iframe.src = url;
   277→   277→        document.body.appendChild(iframe);
   278→   278→        iframe.onload = () => {
   279→   279→          try {
   280→   280→            iframe.contentWindow?.focus();
   281→   281→            iframe.contentWindow?.print();
   282→   282→          } catch (e) {
   283→   283→            console.error("[invoices] print failed", e);
   284→   284→          }
   285→   285→          // Clean up the iframe after a short delay (let the print dialog
   286→   286→          // finish using the blob URL first).
   287→   287→          setTimeout(() => {
   288→   288→            URL.revokeObjectURL(url);
   289→   289→            iframe.remove();
   290→   290→          }, 5000);
   291→   291→        };
   292→   292→      } catch (e: any) {
   293→   293→        toast.error(`Failed to fetch invoice for order ${id.slice(-6)}`);
   294→   294→      }
   295→   295→      // Small delay between prints so the browser doesn't choke.
   296→   296→      await new Promise((r) => setTimeout(r, 800));
   297→   297→    }
   298→   298→  }
   299→   299→
   300→   300→  // --------------------- Bulk status update ---------------------
   301→   301→  async function bulkUpdateStatus(nextStatus: string) {
   302→   302→    if (selectedIds.size === 0) return;
   303→   303→    const label = ORDER_STATUS_LABEL[nextStatus] || nextStatus;
   304→   304→    if (!confirm(`Update ${selectedIds.size} selected order(s) to "${label}"?`)) return;
   305→   305→    setBulkBusy(true);
   306→   306→    const r = await run(
   307→   307→      () =>
   308→   308→        api.post<{ updated: number }>("/api/admin/orders/bulk", {
   309→   309→          ids: Array.from(selectedIds),
   310→   310→          action: "status",
   311→   311→          status: nextStatus,
   312→   312→        }),
   313→   313→      { success: "Bulk update complete", error: "Bulk update failed", silent: true }
   314→   314→    );
   315→   315→    setBulkBusy(false);
   316→   316→    if (r) {
   317→   317→      toast.success(`${r.updated} order(s) marked as ${label}`);
   318→   318→      clearSelection();
   319→   319→      qc.invalidateQueries({ queryKey: ["admin-orders"] });
   320→   320→      qc.invalidateQueries({ queryKey: ["admin-orders-stats"] });
   321→   321→    }
   322→   322→  }
   323→   323→
   324→   324→  // Inline single-order status change from the list
   325→   325→  async function inlineStatusChange(orderId: string, nextStatus: string) {
   326→   326→    const label = ORDER_STATUS_LABEL[nextStatus] || nextStatus;
   327→   327→    const r = await run(
   328→   328→      () => api.patch(`/api/admin/orders/${orderId}/status`, { status: nextStatus }),
   329→   329→      { success: `Order marked as ${label}`, error: "Status update failed", silent: true }
   330→   330→    );
   331→   331→    if (r) {
   332→   332→      toast.success(`Order marked as ${label}`);
   333→   333→      qc.invalidateQueries({ queryKey: ["admin-orders"] });
   334→   334→      qc.invalidateQueries({ queryKey: ["admin-orders-stats"] });
   335→   335→    }
   336→   336→  }
   337→   337→
   338→   338→  return (
   339→   339→    <div>
   340→   340→      <PageHeader
   341→   341→        title="Orders"
   342→   342→        description="View, filter, and manage customer orders."
   343→   343→        actions={
   344→   344→          <Button variant="outline" size="sm" className="gap-1.5" onClick={() => exportCsv(false)}>
   345→   345→            <Download className="size-4" /> Export All
   346→   346→          </Button>
   347→   347→        }
   348→   348→      />
   349→   349→
   350→   350→      {/* ----------------------- Quick Stats Bar ----------------------- */}
   351→   351→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3 mb-4">
   352→   352→        <StatCard
   353→   353→          icon={<ShoppingCart className="size-4" />}
   354→   354→          label="Total Orders"
   355→   355→          value={stats?.total?.toString() ?? "—"}
   356→   356→          tint="bg-emerald-50 text-emerald-700 border-emerald-200"
   357→   357→        />
   358→   358→        <StatCard
   359→   359→          icon={<Clock className="size-4" />}
   360→   360→          label="Pending"
   361→   361→          value={stats?.pending?.toString() ?? "—"}
   362→   362→          tint="bg-amber-50 text-amber-700 border-amber-200"
   363→   363→          onClick={() => toggleStatus("pending")}
   364→   364→          active={statuses.has("pending")}
   365→   365→        />
   366→   366→        <StatCard
   367→   367→          icon={<Truck className="size-4" />}
   368→   368→          label="Shipped"
   369→   369→          value={stats?.shipped?.toString() ?? "—"}
   370→   370→          tint="bg-orange-50 text-orange-700 border-orange-200 dark:bg-orange-950/30 dark:text-orange-300 dark:border-orange-900/50"
   371→   371→          onClick={() => {
   372→   372→            // Toggle both packed + out_for_delivery together

   374→   374→            const bothActive = next.has("packed") && next.has("out_for_delivery");
   375→   375→            if (bothActive) {
   376→   376→              next.delete("packed");
   377→   377→              next.delete("out_for_delivery");
   378→   378→            } else {
   379→   379→              next.add("packed");
   380→   380→              next.add("out_for_delivery");
   381→   381→            }
   382→   382→            setStatuses(next);
   383→   383→          }}
   384→   384→          active={statuses.has("packed") && statuses.has("out_for_delivery")}
   385→   385→        />
   386→   386→        <StatCard
   387→   387→          icon={<CheckCircle2 className="size-4" />}
   388→   388→          label="Delivered"
   389→   389→          value={stats?.delivered?.toString() ?? "—"}
   390→   390→          tint="bg-emerald-50 text-emerald-700 border-emerald-200"
   391→   391→          onClick={() => toggleStatus("delivered")}
   392→   392→          active={statuses.has("delivered")}
   393→   393→        />
   394→   394→        <StatCard
   395→   395→          icon={<XCircle className="size-4" />}
   396→   396→          label="Cancelled"
   397→   397→          value={stats?.cancelled?.toString() ?? "—"}
   398→   398→          tint="bg-rose-50 text-rose-700 border-rose-200"
   399→   399→          onClick={() => toggleStatus("cancelled")}
   400→   400→          active={statuses.has("cancelled")}
   401→   401→        />
   402→   402→        <StatCard
   403→   403→          icon={<TrendingUp className="size-4" />}
   404→   404→          label="Today's Revenue"
   405→   405→          value={stats ? formatCurrency(stats.todayRevenue) : "—"}
   406→   406→          tint="bg-emerald-50 text-emerald-700 border-emerald-200"
   407→   407→          subtitle={
   408→   408→            stats
   409→   409→              ? `Yesterday: ${formatCurrency(stats.yesterdayRevenue)}`
   410→   410→              : undefined
   411→   411→          }
   412→   412→        />
   413→   413→      </div>
   414→   414→
   415→   415→      {/* ----------------------- Filter Bar ----------------------- */}
   416→   416→      <Card className="mb-4">
   417→   417→        <CardContent className="pt-4 pb-4 space-y-3">
   418→   418→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-3">
   419→   419→            <div className="relative lg:col-span-2">
   420→   420→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   421→   421→              <Input
   422→   422→                value={search}
   423→   423→                onChange={(e) => setSearch(e.target.value)}
   424→   424→                placeholder="Order #, customer name, email, phone"
   425→   425→                className="pl-9 h-10"
   426→   426→              />
   427→   427→            </div>
   428→   428→            <Select value={paymentMethod} onValueChange={setPaymentMethod}>
   429→   429→              <SelectTrigger className="h-10"><SelectValue placeholder="Payment Method" /></SelectTrigger>
   430→   430→              <SelectContent>
   431→   431→                <SelectItem value="all">All Methods</SelectItem>
   432→   432→                {PAYMENT_METHOD_OPTIONS.map((m) => (
   433→   433→                  <SelectItem key={m} value={m}>{PAYMENT_METHOD_LABEL[m]}</SelectItem>
   434→   434→                ))}
   435→   435→              </SelectContent>
   436→   436→            </Select>
   437→   437→            <Input type="date" value={from} onChange={(e) => setFrom(e.target.value)} className="h-10" />
   438→   438→            <Input type="date" value={to} onChange={(e) => setTo(e.target.value)} className="h-10" />
   439→   439→          </div>
   440→   440→
   441→   441→          {/* Status multi-select chips */}
   442→   442→          <div className="flex flex-wrap items-center gap-2">
   443→   443→            <span className="text-xs font-medium text-muted-foreground uppercase tracking-wide mr-1">
   444→   444→              Status:
   445→   445→            </span>
   446→   446→            {STATUS_OPTIONS.map((s) => (
   447→   447→              <ChipToggle
   448→   448→                key={s}
   449→   449→                active={statuses.has(s)}
   450→   450→                onClick={() => toggleStatus(s)}
   451→   451→              >
   452→   452→                {ORDER_STATUS_LABEL[s]}
   453→   453→              </ChipToggle>
   454→   454→            ))}
   455→   455→          </div>
   456→   456→
   457→   457→          {/* Payment status multi-select chips */}
   458→   458→          <div className="flex flex-wrap items-center gap-2">
   459→   459→            <span className="text-xs font-medium text-muted-foreground uppercase tracking-wide mr-1">
   460→   460→              Payment:
