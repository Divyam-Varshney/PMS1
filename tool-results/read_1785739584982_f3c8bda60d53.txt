   342→function OrderStatusBadge({
   343→  status,
   344→  size = "md",
   345→}: {
   346→  status: string;
   347→  size?: "sm" | "md" | "lg";
   348→}) {
   349→  const v = STATUS_VISUAL[status] || STATUS_VISUAL.pending;
   350→  const Icon = v.icon;
   351→  const sizeCls =
   352→    size === "lg"
   353→      ? "text-sm px-3 py-1.5 gap-1.5"
   354→      : size === "sm"
   355→        ? "text-[11px] px-2 py-0.5 gap-1"
   356→        : "text-xs px-2.5 py-1 gap-1";
   357→  const iconCls = size === "lg" ? "size-4" : "size-3";
   358→  const v2 = PAYMENT_VISUAL[status];
   359→  const badgeCls = v2 ? v2.badge : "bg-stone-100 text-stone-700 border-stone-200 dark:bg-stone-800/60 dark:text-stone-300 dark:border-stone-700";
   360→  // Use order status color (from STATUS_VISUAL dot) for text contrast mapping.
   361→  // We map common order statuses to their corresponding payment-like colors:
   362→  const orderStatusColor: Record<string, string> = {
   363→    pending: "bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-950/50 dark:text-amber-300 dark:border-amber-900/60",
   364→    confirmed: "bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-950/50 dark:text-emerald-300 dark:border-emerald-900/60",
   365→    packed: "bg-teal-100 text-teal-800 border-teal-200 dark:bg-teal-950/50 dark:text-teal-300 dark:border-teal-900/60",
   366→    out_for_delivery: "bg-cyan-100 text-cyan-800 border-cyan-200 dark:bg-cyan-950/50 dark:text-cyan-300 dark:border-cyan-900/60",
   367→    delivered: "bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-950/50 dark:text-emerald-300 dark:border-emerald-900/60",
   368→    cancelled: "bg-rose-100 text-rose-800 border-rose-200 dark:bg-rose-950/50 dark:text-rose-300 dark:border-rose-900/60",
   369→    returned: "bg-stone-100 text-stone-700 border-stone-200 dark:bg-stone-800/60 dark:text-stone-300 dark:border-stone-700",
   370→  };
   371→  return (
   372→    <span
   373→      className={cn(
   374→        "inline-flex items-center rounded-full border font-medium",
   375→        orderStatusColor[status] || badgeCls,
   376→        sizeCls
   377→      )}
   378→    >
   379→      <span className={cn("size-1.5 rounded-full", v.dot)} />
   380→      <Icon className={iconCls} />
   381→      {v.label}
   382→    </span>
   383→  );
   384→}

   455→export function OrderDetailView({ id }: { id: string }) {
   456→  const back = useAdminStore((s) => s.back);
   457→  const navigate = useAdminStore((s) => s.navigate);
   458→  const qc = useQueryClient();
   459→
   460→  const { data: order, isLoading } = useQuery({
   461→    queryKey: ["admin-order", id],
   462→    queryFn: () => api.get<OrderDetail>(`/api/admin/orders/${id}`),
   463→  });
   464→
   465→  // -------- Local UI state --------
   466→  const [busy, setBusy] = useState(false);
   467→  const [cancelOpen, setCancelOpen] = useState(false);
   468→  const [cancelReason, setCancelReason] = useState("");
   469→
   470→  // Payment management
   471→  const [paymentStatus, setPaymentStatus] = useState("pending");
   472→  const [paymentIdDraft, setPaymentIdDraft] = useState("");
   473→  const [paymentNote, setPaymentNote] = useState("");
   474→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false);
   475→  const [savingPayment, setSavingPayment] = useState(false);
   476→
   477→  // Notes
   478→  const [notes, setNotes] = useState<OrderNote[]>([]);
   479→  const [newNote, setNewNote] = useState("");
   480→  const [savingNote, setSavingNote] = useState(false);
   481→
   482→  // Mobile action sheet
   483→  const [mobileActionsOpen, setMobileActionsOpen] = useState(false);
   484→
   485→  // Sync payment + notes state with order data
   486→  useEffect(() => {
   487→    if (order) {
   488→      setPaymentStatus(order.paymentStatus);
   489→      setPaymentIdDraft(order.paymentId || "");
   490→      setNotes(order.orderNotes ?? []);
   491→    }
   492→  }, [order?.paymentStatus, order?.paymentId, order?.orderNotes]);
   493→
   494→  // -------- Handlers --------
   495→  async function changeStatus(status: string, reason?: string, note?: string) {
   496→    setBusy(true);
   497→    const r = await run(
   498→      () => api.patch(`/api/admin/orders/${id}/status`, { status, reason, note }),
   499→      { success: `Order marked as ${STATUS_VISUAL[status]?.label || status}`, error: "Status update failed" }
   500→    );
   501→    setBusy(false);
   502→    if (r) {
   503→      qc.invalidateQueries({ queryKey: ["admin-order", id] });
   504→      qc.invalidateQueries({ queryKey: ["admin-orders"] });
   505→      qc.invalidateQueries({ queryKey: ["admin-orders-stats"] });
   506→      setMobileActionsOpen(false);
   507→    }
   508→  }
   509→
   510→  async function downloadInvoice() {
   511→    try {
   512→      const buf = await api.raw(`/api/admin/orders/${id}/invoice`);
   513→      const blob = new Blob([buf], { type: "application/pdf" });
   514→      const url = URL.createObjectURL(blob);
   515→      const a = document.createElement("a");
   516→      a.href = url;
   517→      a.download = `${order?.orderNumber || "invoice"}.pdf`;
   518→      a.click();
   519→      URL.revokeObjectURL(url);
   520→      toast.success("Invoice downloaded");
   521→    } catch (e: any) {
   522→      toast.error(e?.message || "Failed to download invoice");
   523→    }
   524→  }
   525→
   526→  function printInvoice() {
   527→    window.open(`/api/admin/orders/${id}/invoice`, "_blank", "noopener");
   528→  }
   529→
   530→  function contactCustomer() {
   531→    if (order?.customer?.phone) {
   532→      window.location.href = `tel:${order.customer.phone}`;
   533→    } else if (order?.shipPhone) {
   534→      window.location.href = `tel:${order.shipPhone}`;
   535→    } else {
   536→      toast.error("No phone number on this order");
   537→    }
   538→  }
   539→
   540→  function copyOrderId() {
   541→    if (!order) return;
   542→    navigator.clipboard.writeText(order.id).then(
   543→      () => toast.success("Order ID copied"),
   544→      () => toast.error("Failed to copy")
   545→    );
   546→  }
   547→
   548→  // -------- Payment update --------
   549→  async function confirmPaymentUpdate() {
   550→    if (!order) return;
   551→    if (paymentStatus === order.paymentStatus && paymentIdDraft === (order.paymentId || "")) {
   552→      toast.info("No changes to save");
   553→      setPaymentDialogOpen(false);
   554→      return;
   555→    }
   556→    setSavingPayment(true);
   557→    const r = await run(
   558→      () =>
   559→        api.patch(`/api/admin/orders/${id}/payment`, {
   560→          paymentStatus,
   561→          paymentId: paymentIdDraft.trim() || undefined,
   562→          note: paymentNote.trim() || undefined,
   563→        }),
   564→      {
   565→        success: "Payment status updated",
   566→        error: "Failed to update payment",
   567→        silent: true,
   568→      }
   569→    );
   570→    setSavingPayment(false);
   571→    if (r) {
   572→      setPaymentDialogOpen(false);
   573→      setPaymentNote("");
   574→      qc.invalidateQueries({ queryKey: ["admin-order", id] });
   575→      toast.success("Payment status updated");
   576→    }
   577→  }
   578→
   579→  // -------- Notes CRUD --------
   580→  async function addNote() {
   581→    if (!newNote.trim()) return;
   582→    setSavingNote(true);
