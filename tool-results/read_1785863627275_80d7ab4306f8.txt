   441→export function PrescriptionDetailView({ id }: { id: string }) {
   442→  const back = useAdminStore((s) => s.back);
   443→  const navigate = useAdminStore((s) => s.navigate);
   444→  const qc = useQueryClient();
   445→  const [adminNotes, setAdminNotes] = useState("");
   446→  const [lightbox, setLightbox] = useState<string | null>(null);
   447→  const [busy, setBusy] = useState(false);
   448→
   449→  // Convert-to-order dialog state
   450→  const [convertOpen, setConvertOpen] = useState(false);
   451→  const [lineItems, setLineItems] = useState<Array<{ productId: string; qty: number; name: string; price: number }>>([]);
   452→  const [productSearch, setProductSearch] = useState("");
   453→  const [searchResults, setSearchResults] = useState<any[]>([]);
   454→  const [searching, setSearching] = useState(false);
   455→  const [converting, setConverting] = useState(false);
   456→  const [selectedPaymentMethod, setSelectedPaymentMethod] = useState("qr");
   457→
   458→  // Fetch active payment methods for the convert dialog dropdown.
   459→  // Dynamic — automatically reflects newly added methods from Admin Settings.
   460→  const { data: paymentMethods } = useQuery({
   461→    queryKey: ["admin", "payment-methods", "active"],
   462→    queryFn: () => api.get<{ items: Array<{ key: string; label: string }> }>("/api/admin/payment-methods"),
   463→    staleTime: 60_000,
   464→  });
   465→  const activePaymentMethods = (paymentMethods?.items || []).filter((m: any) => m.isActive !== false);
   466→
   467→  const { data: p, isLoading } = useQuery({
   468→    queryKey: ["admin-prescription", id],
   469→    queryFn: () => api.get<PrescriptionDetail>(`/api/admin/prescriptions/${id}`),
   470→  });
   471→
   472→  useEffect(() => {
   473→    if (p && adminNotes === "" && p.adminNotes) setAdminNotes(p.adminNotes);
   474→  }, [p, adminNotes]);
   475→
   476→  async function update(status: string) {
   477→    setBusy(true);
   478→    const r = await run(() =>
   479→      api.patch(`/api/admin/prescriptions/${id}`, { status, adminNotes }), {
   480→      success: `Prescription ${STATUS_LABEL[status] || status}`,
   481→      error: "Update failed",
   482→    });
   483→    setBusy(false);
   484→    if (r) qc.invalidateQueries({ queryKey: ["admin-prescription", id] });
   485→  }
   486→
   487→  async function searchProducts(q: string) {
   488→    setProductSearch(q);
   489→    if (q.trim().length < 2) { setSearchResults([]); return; }
   490→    setSearching(true);
   491→    try {
   492→      const res = await api.get<{ items: any[] }>(
   493→        `/api/admin/products?search=${encodeURIComponent(q)}&pageSize=8`
   494→      );
   495→      setSearchResults(res.items || []);
   496→    } catch { setSearchResults([]); }
   497→    setSearching(false);
   498→  }
   499→
   500→  function addProduct(prod: any) {
   501→    if (lineItems.some((li) => li.productId === prod.id)) return;
   502→    setLineItems([...lineItems, { productId: prod.id, qty: 1, name: prod.name, price: prod.sellingPrice }]);
   503→    setProductSearch("");
   504→    setSearchResults([]);
   505→  }
   506→
   507→  function removeLineItem(productId: string) {
   508→    setLineItems(lineItems.filter((li) => li.productId !== productId));
   509→  }
   510→
   511→  function setLineQty(productId: string, qty: number) {
   512→    setLineItems(lineItems.map((li) => li.productId === productId ? { ...li, qty: Math.max(1, qty) } : li));
   513→  }
   514→
   515→  async function convertToOrder() {
   516→    if (lineItems.length === 0) return;
   517→    setConverting(true);
   518→    const r = await run(() =>
   519→      api.post<{ id: string }>(`/api/admin/prescriptions/${id}/convert`, {
   520→        items: lineItems.map((li) => ({ productId: li.productId, qty: li.qty })),
   521→        paymentMethod: selectedPaymentMethod,
   522→      }), {
   523→      success: "Order created from prescription",
   524→      error: "Conversion failed",
   525→    });
   526→    setConverting(false);
   527→    if (r) {
   528→      setConvertOpen(false);
   529→      setLineItems([]);
   530→      qc.invalidateQueries({ queryKey: ["admin-prescription", id] });
   531→      qc.invalidateQueries({ queryKey: ["admin-prescriptions"] });
   532→      navigate({ name: "order-detail", id: r.id });
   533→    }
   534→  }
   535→
   536→  if (isLoading) {
   537→    return (
   538→      <div className="space-y-4">
   539→        <Button variant="ghost" size="sm" onClick={back} className="gap-1.5">
   540→          <ArrowLeft className="size-4" /> Back
   541→        </Button>
   542→        <Card><CardContent className="pt-6 h-48 bg-muted/30 animate-pulse rounded-xl" /></Card>
   543→      </div>
   544→    );
   545→  }
   546→  if (!p) {
   547→    return (
   548→      <div className="space-y-4">
   549→        <Button variant="ghost" size="sm" onClick={back} className="gap-1.5">
   550→          <ArrowLeft className="size-4" /> Back
   551→        </Button>
   552→        <EmptyState title="Prescription not found" />
   553→      </div>
   554→    );
   555→  }
   556→
   557→  return (
   558→    <div className="space-y-5">
   559→      {/* Header */}
   560→      <div className="flex items-center gap-2 flex-wrap">
   561→        <Button variant="ghost" size="sm" onClick={back} className="gap-1.5">
   562→          <ArrowLeft className="size-4" /> Back
   563→        </Button>
   564→        <h1 className="text-xl sm:text-2xl font-semibold tracking-tight">
   565→          Prescription
   566→        </h1>
   567→        <StatusBadge status={p.status} />
   568→        <span className="hidden sm:inline-flex items-center gap-1 text-xs text-muted-foreground ml-auto">
   569→          <Calendar className="size-3.5" />
   570→          {formatDateTime(p.createdAt)}
   571→        </span>
   572→      </div>
   573→
   574→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   575→        {/* Main column */}
   576→        <div className="lg:col-span-2 space-y-4">
   577→          {/* Images */}
   578→          <Card>
   579→            <CardHeader className="pb-3">
   580→              <CardTitle className="text-base flex items-center gap-2">
   581→                <FileImage className="size-4 text-emerald-600" /> Prescription Images ({p.images?.length || 0})
   582→              </CardTitle>
   583→            </CardHeader>
   584→            <CardContent>
   585→              {p.images?.length ? (
   586→                <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   587→                  {p.images.map((img: string, i: number) => (
   588→                    <button
   589→                      key={i}
   590→                      onClick={() => setLightbox(img)}
   591→                      className="relative group aspect-square rounded-md overflow-hidden border bg-muted/30 hover:border-emerald-400 transition-colors"
   592→                    >
   593→                      <img src={img} alt={`Prescription ${i + 1}`} className="size-full object-cover" />
   594→                      <div className="absolute inset-0 bg-black/0 group-hover:bg-black/30 transition flex items-center justify-center">
   595→                        <Maximize2 className="size-5 text-white opacity-0 group-hover:opacity-100" />
   596→                      </div>
   597→                    </button>
   598→                  ))}
   599→                </div>
   600→              ) : (
   601→                <EmptyState title="No images attached" />
   602→              )}
   603→            </CardContent>
   604→          </Card>
   605→
   606→          {/* Customer Notes */}
   607→          <Card>
   608→            <CardHeader className="pb-3">
   609→              <CardTitle className="text-base">Customer Notes</CardTitle>
   610→            </CardHeader>
   611→            <CardContent>
   612→              <p className="text-sm text-foreground whitespace-pre-wrap min-h-[60px]">{p.notes || "—"}</p>
   613→            </CardContent>
   614→          </Card>
   615→        </div>
   616→
   617→        {/* Side column */}
   618→        <div className="space-y-4">
   619→          {/* Customer */}
   620→          <Card>
   621→            <CardHeader className="pb-3">
   622→              <CardTitle className="text-base">Customer</CardTitle>
   623→            </CardHeader>
   624→            <CardContent className="text-sm">
   625→              <CustomerDetailBlock
   626→                customer={p.customer}
   627→                extra={<div className="text-xs text-muted-foreground mt-2">Submitted: {formatDateTime(p.createdAt)}</div>}
   628→              />
   629→            </CardContent>
   630→          </Card>
   631→
   632→          {/* Admin Actions */}
   633→          <Card>
   634→            <CardHeader className="pb-3">
   635→              <CardTitle className="text-base">Admin Actions</CardTitle>
   636→            </CardHeader>
   637→            <CardContent className="space-y-3">
   638→              <div className="space-y-1.5">
   639→                <Label>Admin Notes</Label>
   640→                <Textarea rows={4} value={adminNotes} onChange={(e) => setAdminNotes(e.target.value)} placeholder="Internal notes..." />
   641→              </div>
   642→              <div className="grid grid-cols-1 gap-2 sm:grid-cols-3">
   643→                <Button variant="outline" className="text-amber-600 hover:bg-amber-50 dark:hover:bg-amber-950/30" disabled={busy || p.status === "under_review"} onClick={() => update("under_review")}>
   644→                  <CheckCircle2 className="size-4 mr-1" /> Mark Reviewing
   645→                </Button>
   646→                <Button variant="outline" className="text-emerald-600 hover:bg-emerald-50 dark:hover:bg-emerald-950/30" disabled={busy} onClick={() => update("verified")}>
   647→                  <CheckCircle2 className="size-4 mr-1" /> Approve
   648→                </Button>
   649→                <Button variant="outline" className="text-destructive hover:bg-rose-50 dark:hover:bg-rose-950/30" disabled={busy} onClick={() => update("rejected")}>
   650→                  <XCircle className="size-4 mr-1" /> Reject
   651→                </Button>
   652→              </div>
   653→              {p.status === "converted" ? (
   654→                <div className="rounded-md border border-emerald-200 bg-emerald-50 dark:bg-emerald-950/30 dark:border-emerald-900/50 p-3 text-center text-sm text-emerald-700 dark:text-emerald-300">
   655→                  <FileCheck2 className="mx-auto mb-1 size-5" />
   656→                  Converted to an order
   657→                  {p.convertedOrderId && (
   658→                    <Button
   659→                      variant="link"
   660→                      size="sm"
   661→                      className="mt-1 block h-auto p-0 text-emerald-700 dark:text-emerald-300"
   662→                      onClick={() => navigate({ name: "order-detail", id: p.convertedOrderId! })}
   663→                    >
   664→                      View order →
   665→                    </Button>
   666→                  )}
   667→                </div>
   668→              ) : (
   669→                <Button
   670→                  className="w-full gap-2 bg-emerald-600 hover:bg-emerald-700 text-white"
   671→                  disabled={busy}
   672→                  onClick={() => setConvertOpen(true)}
   673→                >
   674→                  <FileCheck2 className="size-4" /> Create Order from Prescription
   675→                </Button>
   676→              )}
   677→              <p className="text-[11px] text-muted-foreground">
   678→                Add the products from this prescription, then create an order with
   679→                automatic pricing-engine pricing.
   680→              </p>
   681→            </CardContent>
   682→          </Card>
   683→        </div>
   684→      </div>
   685→
   686→      {/* Convert-to-order dialog */}
   687→      <Dialog open={convertOpen} onOpenChange={setConvertOpen}>
   688→        <DialogContent className="max-w-2xl">
   689→          <DialogHeader>
   690→            <DialogTitle>Create Order from Prescription</DialogTitle>
   691→            <DialogDescription>
   692→              Search and add the products mentioned in the prescription. Pricing
   693→              is computed automatically by the discount engine.
   694→            </DialogDescription>
   695→          </DialogHeader>
   696→          <div className="space-y-3">
   697→            <div className="relative">
   698→              <Input
   699→                placeholder="Search products by name, composition, SKU..."
   700→                value={productSearch}
   701→                onChange={(e) => searchProducts(e.target.value)}
   702→              />
   703→              {searchResults.length > 0 && (
   704→                <div className="absolute z-10 mt-1 max-h-60 w-full overflow-y-auto rounded-md border bg-background shadow-lg">
   705→                  {searchResults.map((prod) => (
   706→                    <button
   707→                      key={prod.id}
   708→                      onClick={() => addProduct(prod)}
   709→                      className="flex w-full items-center justify-between border-b px-3 py-2 text-left text-sm last:border-b-0 hover:bg-accent"
   710→                    >
   711→                      <div className="min-w-0">
   712→                        <p className="truncate font-medium">{prod.name}</p>
   713→                        <p className="truncate text-xs text-muted-foreground">
   714→                          {prod.brand?.name} · Rs. {prod.sellingPrice}
   715→                        </p>
   716→                      </div>
   717→                      <span className="ml-2 shrink-0 text-xs text-primary">+ Add</span>
   718→                    </button>
   719→                  ))}
   720→                </div>
   721→              )}
   722→              {searching && <p className="mt-1 text-xs text-muted-foreground">Searching...</p>}
   723→            </div>
   724→
   725→            {lineItems.length === 0 ? (
   726→              <div className="rounded-md border border-dashed p-6 text-center text-sm text-muted-foreground">
   727→                No products added yet. Search above to add medicines from the prescription.
   728→              </div>
   729→            ) : (
   730→              <div className="space-y-2">
   731→                {lineItems.map((li) => (
   732→                  <div key={li.productId} className="flex items-center gap-2 rounded-md border p-2">
   733→                    <div className="min-w-0 flex-1">
   734→                      <p className="truncate text-sm font-medium">{li.name}</p>
   735→                      <p className="text-xs text-muted-foreground">Rs. {li.price}</p>
   736→                    </div>
   737→                    <Input
   738→                      type="number"
   739→                      min={1}
   740→                      value={li.qty}
   741→                      onChange={(e) => setLineQty(li.productId, parseInt(e.target.value) || 1)}
   742→                      className="h-8 w-16 text-center"
   743→                    />
   744→                    <Button variant="ghost" size="icon" className="size-8 text-destructive" onClick={() => removeLineItem(li.productId)}>
   745→                      <XCircle className="size-4" />
   746→                    </Button>
   747→                  </div>
   748→                ))}
   749→                <div className="flex justify-between border-t pt-2 text-sm font-semibold">
   750→                  <span>{lineItems.length} item(s)</span>
   751→                  <span>Rs. {lineItems.reduce((s, li) => s + li.price * li.qty, 0).toFixed(2)}</span>
   752→                </div>
   753→              </div>
   754→            )}
   755→
   756→            {/* Payment Method Selector — defaults to QR Code Payment */}
   757→            <div className="space-y-2">
   758→              <label className="text-sm font-medium">Payment Method</label>
   759→              <select
   760→                value={selectedPaymentMethod}
   761→                onChange={(e) => setSelectedPaymentMethod(e.target.value)}
   762→                className="w-full rounded-md border border-input bg-background px-3 py-2 text-sm ring-offset-background focus:outline-none focus:ring-2 focus:ring-ring focus:ring-offset-2"
   763→              >
   764→                {activePaymentMethods.length > 0 ? (
   765→                  activePaymentMethods.map((m: any) => (
   766→                    <option key={m.key} value={m.key}>{m.label}</option>
   767→                  ))
   768→                ) : (
   769→                  <>
   770→                    <option value="qr">QR Code Payment</option>
   771→                    <option value="cod">Cash on Delivery</option>
   772→                    <option value="upi">UPI Payment</option>
   773→                    <option value="razorpay">Razorpay</option>
   774→                  </>
   775→                )}
   776→              </select>
   777→              <p className="text-xs text-muted-foreground">
   778→                The selected payment method will be saved on the order and displayed everywhere (invoice, customer portal, email).
   779→              </p>
   780→            </div>
   781→          </div>
   782→          <DialogFooter>
   783→            <Button variant="outline" onClick={() => setConvertOpen(false)}>Cancel</Button>
   784→            <Button
   785→              disabled={lineItems.length === 0 || converting}
   786→              onClick={convertToOrder}
   787→              className="gap-2 bg-emerald-600 hover:bg-emerald-700 text-white"
   788→            >
   789→              {converting ? <Loader2 className="size-4 animate-spin" /> : <FileCheck2 className="size-4" />}
   790→              Create Order
   791→            </Button>
   792→          </DialogFooter>
   793→        </DialogContent>
   794→      </Dialog>
   795→
   796→      {/* Lightbox */}
   797→      {lightbox && (
   798→        <Dialog open onOpenChange={(o) => !o && setLightbox(null)}>
   799→          <DialogContent className="max-w-3xl">
   800→            <DialogHeader>
   801→              <DialogTitle>Prescription Image</DialogTitle>
   802→              <DialogDescription>Zoom in for verification.</DialogDescription>
   803→            </DialogHeader>
   804→            <img src={lightbox} alt="Prescription" className="w-full max-h-[70vh] object-contain rounded-md" />
   805→            <DialogFooter>
   806→              <a href={lightbox} download>
   807→                <Button variant="outline" size="sm" className="gap-1.5">
   808→                  <Download className="size-4" /> Download
   809→                </Button>
   810→              </a>
   811→              <Button variant="outline" size="sm" onClick={() => setLightbox(null)} className="gap-1.5">
   812→                <RotateCw className="size-4" /> Close
   813→              </Button>
   814→            </DialogFooter>
   815→          </DialogContent>
   816→        </Dialog>
   817→      )}
   818→    </div>
   819→  );
   820→}
