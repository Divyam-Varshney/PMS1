   236→function NavList({ admin, searchQuery }: { admin: AdminInfo; searchQuery: string }) {
   237→  const view = useAdminStore((s) => s.view);
   238→  const navigate = useAdminStore((s) => s.navigate);
   239→  const [collapsedGroups, setCollapsedGroups] = useState<Set<string>>(new Set());
   240→
   241→  // Fetch lightweight badge counts
   242→  const { data: counts } = useQuery({
   243→    queryKey: ["admin-counts"],
   244→    queryFn: () => api.get<{ pendingOrders: number; lowStock: number; pendingPrescriptions: number; pendingManualRequests: number; unreadNotifications: number }>("/api/admin/counts"),
   245→    refetchInterval: 30 * 1000,
   246→  });
   247→
   248→  const badgeFor = (viewName: string): number | null => {
   249→    if (!counts) return null;
   250→    switch (viewName) {
   251→      case "orders": return counts.pendingOrders || null;
   252→      case "products": return counts.lowStock || null;
   253→      case "prescriptions": return counts.pendingPrescriptions || null;
   254→      case "manual-requests": return counts.pendingManualRequests || null;
   255→      default: return null;
   256→    }
   257→  };
   258→
   259→  const toggleGroup = (label: string) => {
   260→    setCollapsedGroups((prev) => {
   261→      const next = new Set(prev);
   262→      if (next.has(label)) next.delete(label);
   263→      else next.add(label);
   264→      return next;
   265→    });
   266→  };
   267→
   268→  const isVisible = (item: { permission: AdminPermissionKey }) => hasPermission(admin, item.permission);
   269→
   270→  // When searching, show all items flat (ignore group collapse)
   271→  const isSearching = searchQuery.trim().length > 0;
   272→  const query = searchQuery.toLowerCase().trim();
   273→
   274→  const visibleGroups = NAV_GROUPS.map((g) => ({
   275→    ...g,
   276→    items: g.items.filter((item) => {
   277→      if (!isVisible(item)) return false;
   278→      if (!isSearching) return true;
   279→      return item.label.toLowerCase().includes(query);
   280→    }),
   281→  })).filter((g) => g.items.length > 0);
   282→
   283→  // Dashboard is always visible (if permitted)
   284→  const showDashboard = hasPermission(admin, "dashboard") && (!isSearching || "dashboard".includes(query));
   285→
   286→  return (
   287→    <SidebarContent>
   288→      {/* Dashboard — always at top, outside groups */}
   289→      {showDashboard && (
   290→        <SidebarGroup>
   291→          <SidebarMenu>
   292→            <SidebarMenuItem>
   293→              <SidebarMenuButton
   294→                isActive={view.name === "dashboard"}
   295→                onClick={() => navigate({ name: "dashboard" })}
   296→                tooltip="Dashboard"
   297→                className="transition-premium"
   298→              >
   299→                <LayoutDashboard />
   300→                <span>Dashboard</span>
   301→              </SidebarMenuButton>
   302→            </SidebarMenuItem>
   303→          </SidebarMenu>
   304→        </SidebarGroup>
   305→      )}
   306→
   307→      {/* Collapsible nav groups */}
   308→      {visibleGroups.map((g) => {
   309→        const isCollapsed = collapsedGroups.has(g.label) && !isSearching;
   310→        return (
   311→          <Collapsible
   312→            key={g.label}
   313→            open={!isCollapsed}
   314→            // When searching, all groups are forced open so matches are visible.
   315→            // Don't call toggleGroup during search — it would silently mutate
   316→            // the collapsed state and cause unexpected expand/collapse when the
   317→            // search is cleared.
   318→            onOpenChange={() => !isSearching && toggleGroup(g.label)}
   319→          >
   320→            <SidebarGroup>
   321→              <CollapsibleTrigger asChild>
   322→                <SidebarGroupLabel className="cursor-pointer select-none hover:bg-sidebar-accent/60 transition-colors admin-section-label">
   323→                  <span>{g.label}</span>
   324→                  <ChevronRight className={`ml-auto size-3.5 transition-transform ${isCollapsed ? "" : "rotate-90"}`} />
   325→                </SidebarGroupLabel>
   326→              </CollapsibleTrigger>
   327→              <CollapsibleContent>
   328→                <SidebarMenu>
   329→                  {g.items.map((item) => {
   330→                    const isActive =
   331→                      view.name === item.view.name ||
   332→                      (item.view.name === "products" && view.name === "product-edit") ||
   333→                      (item.view.name === "orders" && view.name === "order-detail") ||
   334→                      (item.view.name === "prescriptions" && view.name === "prescription-detail") ||
   335→                      (item.view.name === "manual-requests" && view.name === "manual-request-detail") ||
   336→                      (item.view.name === "customers" && view.name === "customer-detail");
   337→                    return (
   338→                      <SidebarMenuItem key={item.label}>
   339→                        <SidebarMenuButton
   340→                          isActive={isActive}
   341→                          onClick={() => navigate(item.view)}
   342→                          tooltip={item.label}
   343→                          className="transition-premium"
   344→                        >
   345→                          <item.icon />
   346→                          <span>{item.label}</span>
   347→                          {badgeFor(item.view.name) !== null && (
   348→                            <Badge className="ml-auto h-5 min-w-5 justify-center px-1.5 text-[10px] font-bold bg-amber-500 text-white hover:bg-amber-500 animate-badge-pulse">
   349→                              {badgeFor(item.view.name)}
   350→                            </Badge>
   351→                          )}
   352→                        </SidebarMenuButton>
   353→                      </SidebarMenuItem>
   354→                    );
   355→                  })}
   356→                </SidebarMenu>
   357→              </CollapsibleContent>
   358→            </SidebarGroup>
   359→          </Collapsible>
   360→        );
   361→      })}
   362→
   363→      {/* No results when searching */}
   364→      {isSearching && visibleGroups.length === 0 && !showDashboard && (
   365→        <div className="px-4 py-8 text-center text-xs text-muted-foreground">
   366→          No navigation items match “{searchQuery}”
   367→        </div>
   368→      )}
   369→    </SidebarContent>
   370→  );
   371→}

   374→// A3: BREADCRUMB NAVIGATION
   375→// ---------------------------------------------------------------------------
   376→
   377→function Breadcrumbs({ viewName, navigate }: { viewName: string; navigate: (v: AdminView) => void }) {
   378→  const crumbs: Array<{ label: string; view?: AdminView }> = [];
   379→
   380→  // Always start with Dashboard
   381→  crumbs.push({ label: "Dashboard", view: { name: "dashboard" } });
   382→
   383→  // Add parent if this is a detail page
   384→  const parentName = BREADCRUMB_PARENT[viewName];
   385→  if (parentName) {
   386→    crumbs.push({ label: TITLE_MAP[parentName] || parentName, view: { name: parentName } as AdminView });
   387→  }
   388→
   389→  // Add current page (not clickable)
   390→  const currentLabel = TITLE_MAP[viewName] || viewName;
   391→  if (viewName !== "dashboard") {
   392→    crumbs.push({ label: currentLabel });
   393→  }
   394→
   395→  return (
   396→    <nav className="flex items-center gap-1 text-sm text-muted-foreground min-w-0">
   397→      {crumbs.map((crumb, i) => {
   398→        const isLast = i === crumbs.length - 1;
   399→        return (
   400→          <span key={i} className="flex items-center gap-1 min-w-0">
   401→            {i > 0 && <ChevronRight className="size-3.5 shrink-0 text-muted-foreground/50" />}
   402→            {crumb.view && !isLast ? (
   403→              <button
   404→                onClick={() => navigate(crumb.view!)}
   405→                className="hover:text-foreground transition-colors truncate"
   406→              >
   407→                {crumb.label}
   408→              </button>
   409→            ) : (
   410→              <span className={`truncate ${isLast ? "font-semibold text-foreground" : ""}`}>
   411→                {crumb.label}
   412→              </span>
   413→            )}
   414→          </span>
   415→        );
   416→      })}
   417→    </nav>
   418→  );
   419→}

   425→function CommandPalette({ open, onOpenChange, navigate }: { open: boolean; onOpenChange: (v: boolean) => void; navigate: (v: AdminView) => void }) {
   426→  const [query, setQuery] = useState("");
   427→
   428→  const filtered = useMemo(() => {
   429→    if (!query.trim()) return ALL_NAV_ITEMS;
   430→    const q = query.toLowerCase();
   431→    return ALL_NAV_ITEMS.filter((item) => item.label.toLowerCase().includes(q));
   432→  }, [query]);
   433→
   434→  const handleSelect = (view: AdminView) => {
   435→    navigate(view);
   436→    onOpenChange(false);
   437→    setQuery("");
   438→  };
   439→
   440→  return (
   441→    <Dialog open={open} onOpenChange={onOpenChange}>
   442→      <DialogContent className="p-0 gap-0 overflow-hidden max-w-lg shadow-premium-xl">
   443→        {/* Accessibility: DialogTitle is required by Radix Dialog for screen
   444→            reader support. Rendered visually-hidden (sr-only). */}
   445→        <DialogTitle className="sr-only">Command palette</DialogTitle>
   446→        <div className="flex items-center border-b border-border/70 px-4 bg-premium-gradient">
   447→          <Search className="size-4 mr-3 text-muted-foreground shrink-0" />
   448→          <Input
   449→            autoFocus
   450→            placeholder="Type a command or search..."
   451→            value={query}
   452→            onChange={(e) => setQuery(e.target.value)}
   453→            aria-label="Search navigation items"
   454→            className="h-12 border-0 px-0 focus-visible:ring-0 text-sm bg-transparent"
   455→            onKeyDown={(e) => {
   456→              if (e.key === "Enter" && filtered.length > 0) {
   457→                handleSelect(filtered[0].view);
   458→              }
   459→            }}
   460→          />
   461→          <kbd className="ml-2 shrink-0 rounded border bg-muted px-1.5 py-0.5 text-[10px] text-muted-foreground font-mono">ESC</kbd>
   462→        </div>
   463→        <div className="max-h-80 overflow-y-auto p-2 scrollbar-premium">
   464→          {filtered.length === 0 ? (
   465→            <div className="px-3 py-8 text-center text-sm text-muted-foreground">
   466→              No results found
   467→            </div>
   468→          ) : (
   469→            filtered.map((item) => (
   470→              <button
   471→                key={item.label}
   472→                onClick={() => handleSelect(item.view)}
   473→                className="flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm hover:bg-accent transition-premium text-left"
   474→              >
   475→                <span className="flex size-7 items-center justify-center rounded-md bg-primary/10 text-primary shrink-0">
   476→                  <item.icon className="size-3.5" />
   477→                </span>
   478→                <span className="flex-1 font-medium">{item.label}</span>
   479→                <span className="text-[10px] text-muted-foreground uppercase tracking-wide">{item.group}</span>
   480→              </button>
   481→            ))
   482→          )}
   483→        </div>
   484→      </DialogContent>
   485→    </Dialog>
   486→  );
   487→}

   492→
   493→function StoreOpenToggle() {
   494→  const qc = useQueryClient();
   495→  const { data, isLoading } = useQuery({
   496→    queryKey: ["admin-store-status"],
   497→    queryFn: () => api.get<{ open: boolean; openStatus: boolean }>("/api/admin/settings/store-status"),
   498→  });
   499→
   500→  async function toggle(next: boolean) {
   501→    const r = await run(
   502→      () => api.patch("/api/admin/settings/store-status", { openStatus: next }),
   503→      { success: next ? "Store opened" : "Store closed", error: "Failed to toggle" }
   504→    );
   505→    if (r) {
   506→      qc.invalidateQueries({ queryKey: ["admin-store-status"] });
   507→    }
   508→  }
   509→
   510→  const isOpen = !!data?.openStatus;
   511→  return (
   512→    <div className={`flex items-center gap-2 px-3 py-2 rounded-lg border transition-colors ${isOpen ? "border-emerald-200/60 bg-emerald-50/50 dark:bg-emerald-950/20 dark:border-emerald-800/40" : "border-border bg-muted/30"}`}>
   513→      <Store className={`size-4 ${isOpen ? "text-emerald-600 dark:text-emerald-400" : "text-muted-foreground"}`} />
   514→      <div className="flex-1 min-w-0">
   515→        <div className="text-xs font-medium">Store Status</div>
   516→        <div className={`text-[11px] ${isOpen ? "text-emerald-700 dark:text-emerald-300" : "text-muted-foreground"}`}>
   517→          {isLoading ? "Loading…" : data?.open ? "Open for orders" : "Closed"}
   518→        </div>
   519→      </div>
   520→      <Switch
   521→        checked={isOpen}
   522→        disabled={isLoading}
   523→        onCheckedChange={toggle}
   524→        aria-label="Toggle store open/close"
   525→      />
   526→    </div>
   527→  );
   528→}

   596→  const [searchQuery, setSearchQuery] = useState("");

   599→  const [cmdOpen, setCmdOpen] = useState(false);
   600→
   601→  // A4: Keyboard shortcut listener (Cmd+K / Ctrl+K)
   602→  useEffect(() => {
   603→    const handler = (e: KeyboardEvent) => {
   604→      if ((e.metaKey || e.ctrlKey) && e.key === "k") {
   605→        e.preventDefault();
   606→        setCmdOpen((v) => !v);
   607→      }
   608→    };
   609→    window.addEventListener("keydown", handler);
   610→    return () => window.removeEventListener("keydown", handler);
   611→  }, []);
   612→
   613→  // E7: Restore saved dark mode preference on mount
   614→  useEffect(() => {
   615→    const savedTheme = localStorage.getItem("admin-theme");
   616→    if (savedTheme === "dark") {
   617→      document.documentElement.classList.add("dark");
   618→    }
   619→  }, []);
   620→
   621→  // Current date for the topbar widget (B6: date widget)
   622→  const today = new Date();
   623→  const dateStr = today.toLocaleDateString("en-IN", { weekday: "short", day: "2-digit", month: "short" });

   637→          </div>

   642→
   643→        <NavList admin={admin} searchQuery={searchQuery} />
   644→        <SidebarFooterArea admin={admin} onLogout={onLogout} />
   645→      </Sidebar>
   646→
   647→      <SidebarInset>
   648→        {/* Topbar with breadcrumbs + command palette trigger + date widget */}
   649→        <header className="sticky top-0 z-30 flex h-14 items-center gap-2 border-b border-border/70 bg-background/95 backdrop-blur-md px-4 shadow-premium-sm">

   679→            variant="ghost"

   684→              if (isDark) {
   685→                document.documentElement.classList.remove("dark");
   686→                localStorage.setItem("admin-theme", "light");
   687→              } else {
   688→                document.documentElement.classList.add("dark");
   689→                localStorage.setItem("admin-theme", "dark");
   690→              }
   691→            }}
   692→            title="Toggle dark mode"
   693→          >
   694→            <Sun className="size-4 rotate-0 scale-100 transition-all dark:-rotate-90 dark:scale-0" />

   715→            {roleLabel(admin.role)}
   716→          </Badge>
   717→
   718→          <Button variant="outline" size="sm" onClick={onLogout} className="hidden md:flex transition-premium">
   719→            <LogOut className="size-4" />

   721→          <Button variant="ghost" size="icon" className="md:hidden" onClick={onLogout}>
   722→            <LogOut className="size-4" />
   723→          </Button>
   724→        </header>
   725→
   726→        {/* A6: Page transition animations — the key includes the view name AND
   727→            any id field so navigating between two records of the same type
   728→            (e.g. order-detail → different order-detail) re-triggers the

   754→/** Wrapper to fetch admin from /me on mount; renders children only when logged in. */
   755→export function AdminGuard({
   756→  onLoggedOut,
   757→  children,
   758→}: {
   759→  onLoggedOut: () => void;
   760→  children: (admin: AdminInfo) => ReactNode;
   761→}) {
   762→  const [admin, setAdmin] = useState<AdminInfo | null>(null);
   763→  const [loading, setLoading] = useState(true);
   764→
   765→  useEffect(() => {
   766→    let mounted = true;
   767→    api
   768→      .get<AdminInfo>("/api/admin-auth/me")
   769→      .then((a) => mounted && setAdmin(a))
   770→      .catch(() => mounted && onLoggedOut())
   771→      .finally(() => mounted && setLoading(false));
   772→    return () => {
   773→      mounted = false;
   774→    };
   775→  }, [onLoggedOut]);
   776→
   777→  if (loading) {
   778→    return (
   779→      <div className="min-h-screen flex items-center justify-center bg-background">
   780→        <Loader2 className="size-8 animate-spin text-primary" />
   781→      </div>
   782→    );
   783→  }
   784→  if (!admin) return null;
   785→  return <>{children(admin)}</>;
   786→}

   789→  await run(() => api.post("/api/admin-auth/logout"), { success: "Signed out" });
   790→}
