    51→// HTML helpers — simple regex-based formatters. These are intentionally
    52→// lightweight (no DOM parser); they handle the common case of pasted HTML
    53→// being either minified-on-one-line or overly indented.

    56→/** Pretty-print HTML: put each block-level tag on its own line + indent. */
    57→function formatHtml(html: string): string {
    58→  if (!html.includes("<")) return html; // not HTML
    59→  const out = html
    60→    .replace(/^\s+/gm, "")
    61→    .replace(/>\s+</g, ">\n<")
    62→    .replace(/[ \t]{2,}/g, " ");
    63→  const lines = out.split("\n").map((l) => l.trim()).filter(Boolean);
    64→  let depth = 0;
    65→  const indentStr = "  ";
    66→  const result: string[] = [];
    67→  for (const line of lines) {
    68→    const isClosing = /^<\//.test(line);
    69→    const isSelfClosed = /\/>$/.test(line) || /^<[^>]+\/>$/.test(line);
    70→    const isOpening = /^<[^/!?][^>]*[^/]>$/.test(line) && !isSelfClosed;
    71→    const voidMatch = /^<(br|img|hr|input|meta|link|area|base|col|embed|source|track|wbr)\b/i.test(line);
    72→    if (isClosing) depth = Math.max(0, depth - 1);
    73→    result.push(indentStr.repeat(depth) + line);
    74→    if (isOpening && !isClosing && !voidMatch) depth += 1;
    75→  }
    76→  return result.join("\n");
    77→}

    79→/** Minify HTML: collapse whitespace between tags + strip leading/trailing
    80→ *  whitespace on each line. Preserves a single space inside text runs. */
    81→function minifyHtml(html: string): string {
    82→  if (!html.includes("<")) return html.trim();
    83→  return html
    84→    .replace(/\s+/g, " ")
    85→    .replace(/>\s+/g, ">")
    86→    .replace(/\s+</g, "<")
    87→    .trim();
    88→}

    90→/** Count the number of lines in a string (always >= 1). */
    91→function countLines(s: string): number {
    92→  if (!s) return 0;
    93→  return s.split("\n").length;
    94→}

   108→/** Keys whose name suggests a marketing/promotional template. */
   109→const MARKETING_KEY_PATTERNS = [
   110→  /^newsletter/i,
   111→  /^promo/i,
   112→  /^campaign/i,
   113→  /^marketing/i,
   114→  /^broadcast/i,

   120→function categorize(t: any): TemplateCategory {
   121→  if (t.channel === "whatsapp") return "whatsapp";
   122→  const rawKey: string = t.key || "";
   123→  if (MARKETING_KEY_PATTERNS.some((re) => re.test(rawKey))) return "marketing";
   124→  const baseKey = rawKey.replace(/_(email|whatsapp)$/, "");
   125→  if (baseKey.startsWith("admin")) return "admin";
   126→  if (t.channel === "email") return "customer";
   127→  return "other";
   128→}

   513→// ----------------------------------------------------------------------------
   514→// TemplateCard — the main editing card for a single template. Includes
   515→// subject + body editors, variables editor, HTML format/minify helpers,
   516→// live preview, and a sticky footer with active toggle + save/delete.
   517→// ----------------------------------------------------------------------------
   518→function TemplateCard({ template, onChanged }: { template: any; onChanged: () => void }) {
   519→  const [subject, setSubject] = useState(template.subject || "");
   520→  const [body, setBody] = useState(template.body || "");
   521→  const [isActive, setIsActive] = useState(template.isActive);
   522→  const [saving, setSaving] = useState(false);
   523→  const [showPreview, setShowPreview] = useState(false);
   524→
   525→  let initialVars: string[] = [];
   526→  try {
   527→    initialVars = template.variables ? JSON.parse(template.variables) : [];
   528→  } catch (e) { console.error("[templates] error:", e); }
   529→
   530→  const [vars, setVars] = useState<string[]>(initialVars);
   531→  const [varsDraft, setVarsDraft] = useState("");
   532→
   533→  function commitVarsDraft() {
   534→    if (!varsDraft.trim()) return;
   535→    const parts = varsDraft
   536→      .split(/[\s,]+/)
   537→      .map((p) => p.trim().replace(/^{{|}}$/g, ""))
   538→      .filter(Boolean);
   539→    if (!parts.length) {
   540→      setVarsDraft("");
   541→      return;
   542→    }
   543→    const merged = Array.from(new Set([...vars, ...parts]));
   544→    setVars(merged);
   545→    setVarsDraft("");
   546→  }
   547→
   548→  function removeVar(v: string) {
   549→    setVars(vars.filter((x) => x !== v));
   550→  }
   551→
   552→  async function save() {
   553→    setSaving(true);
   554→    const r = await run(
   555→      () => api.put(`/api/admin/notifications/templates/${template.id}`, {
   556→        subject,
   557→        body,
   558→        isActive,
   559→        variables: vars,
   560→      }),
   561→      { success: "Template saved", error: "Save failed" }
   562→    );
   563→    setSaving(false);
   564→    if (r) onChanged();
   565→  }
   566→
   567→  async function del() {
   568→    if (!confirm("Delete this template?")) return;
   569→    const r = await run(() => api.del(`/api/admin/notifications/templates/${template.id}`), {
   570→      success: "Template deleted", error: "Delete failed",
   571→    });
   572→    if (r) onChanged();
   573→  }
   574→
   575→  const isHtml = body.includes("<") && body.includes(">");
   576→  const category = categorize(template);
   577→  const meta = CHANNEL_META[category];
   578→  const ChannelIcon = meta.icon;
   579→
   580→  function onFormat() {
   581→    if (!isHtml) {
   582→      toast.info("Body doesn't look like HTML — nothing to format.");
   583→      return;
   584→    }
   585→    setBody(formatHtml(body));
   586→    toast.success("HTML formatted");
   587→  }
   588→
   589→  function onMinify() {
   590→    if (!isHtml) {
   591→      toast.info("Body doesn't look like HTML — nothing to minify.");
   592→      return;
   593→    }
   594→    setBody(minifyHtml(body));
   595→    toast.success("HTML minified");
   596→  }
   597→
   598→  return (
   599→    <Card className="overflow-hidden rounded-xl border-border/50 p-0 shadow-premium-sm transition-premium hover:shadow-premium">
   600→      {/* Header — icon + name + key on the left, channel + HTML badge on the right */}
   601→      <CardHeader className="flex-row items-center justify-between space-y-0 border-b border-border/50 bg-gradient-to-br from-muted/40 to-muted/10 px-4 py-3">
   602→        <div className="flex min-w-0 items-center gap-3">
   603→          <div className={`flex size-9 shrink-0 items-center justify-center rounded-lg ${meta.tint} shadow-premium-sm`}>
   604→            <ChannelIcon className="size-4" />
   605→          </div>
   606→          <div className="min-w-0">
   607→            <CardTitle className="truncate text-sm font-semibold">{template.name}</CardTitle>
   608→            <div className="truncate font-mono text-[11px] text-muted-foreground">{template.key}</div>
   609→          </div>
   610→        </div>
   611→        <div className="flex shrink-0 items-center gap-1.5">
   612→          {isHtml && (
   613→            <Badge variant="outline" className="gap-1 border-amber-200 bg-amber-50 text-amber-700 admin-badge-amber">
   614→              <Code className="size-3" /> HTML
   615→            </Badge>
   616→          )}
   617→          <Badge variant="outline" className={`capitalize ${meta.tint}`}>
   618→            {meta.shortLabel}
   619→          </Badge>
   620→        </div>
   621→      </CardHeader>
   622→      <CardContent className="space-y-3 p-4">
   623→        <div className="space-y-1.5">
   624→          <Label className="text-xs font-medium text-muted-foreground">Subject</Label>
   625→          <Input value={subject} onChange={(e) => setSubject(e.target.value)} className="h-9" />
   626→        </div>
   627→        <div className="space-y-1.5">
   628→          <div className="flex items-center justify-between gap-2">
   629→            <Label className="text-xs font-medium text-muted-foreground">
   630→              Body {isHtml && <span className="text-amber-600 dark:text-amber-400">(HTML detected)</span>}
   631→            </Label>
   632→            <div className="flex items-center gap-0.5">
   633→              {isHtml && (
   634→                <>
   635→                  <Button
   636→                    size="sm"
   637→                    variant="ghost"
   638→                    className="h-7 gap-1 text-xs"
   639→                    onClick={onFormat}
   640→                    title="Pretty-print HTML (add newlines + indentation)"
   641→                  >
   642→                    <Wand2 className="size-3" /> Format
   643→                  </Button>
   644→                  <Button
   645→                    size="sm"
   646→                    variant="ghost"
   647→                    className="h-7 gap-1 text-xs"
   648→                    onClick={onMinify}
   649→                    title="Remove extra whitespace from HTML"
   650→                  >
   651→                    <Wand2 className="size-3" /> Minify
   652→                  </Button>
   653→                </>
   654→              )}
   655→              <Button size="sm" variant="ghost" className="h-7 gap-1 text-xs" onClick={() => setShowPreview(!showPreview)}>
   656→                <Eye className="size-3" /> {showPreview ? "Edit" : "Preview"}
   657→              </Button>
   658→            </div>
   659→          </div>
   660→          {showPreview && isHtml ? (
   661→            <div className="overflow-hidden rounded-lg border border-border/50 shadow-premium-sm">
   662→              {/* Browser-like chrome for the preview — gives the email template
   663→                  a real "rendered email" framing rather than a plain div. */}
   664→              <div className="flex items-center gap-1.5 border-b border-border/50 bg-muted/40 px-3 py-1.5">
   665→                <span className="size-2 rounded-full bg-rose-400" />
   666→                <span className="size-2 rounded-full bg-amber-400" />
   667→                <span className="size-2 rounded-full bg-emerald-400" />
   668→                <span className="ml-2 text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   669→                  Email Preview
   670→                </span>
   671→              </div>
   672→              <div
   673→                className="min-h-[120px] max-h-[400px] overflow-y-auto bg-white p-4 text-sm scrollbar-premium dark:bg-white"
   674→                dangerouslySetInnerHTML={{ __html: sanitizeHtml(body) }}
   675→              />
   676→            </div>
   677→          ) : (
   678→            <Textarea
   679→              rows={8}
   680→              value={body}
   681→              onChange={(e) => setBody(e.target.value)}
   682→              className={EDITOR_TEXTAREA_CLASS}
   683→            />
   684→          )}
   685→          <div className="flex items-center justify-between text-[11px] text-muted-foreground">
   686→            <span>{body.length.toLocaleString()} chars · {countLines(body)} lines</span>
   687→            {isHtml && <span className="text-amber-600 dark:text-amber-400">HTML</span>}
   688→          </div>
   689→        </div>
   690→        {/* Variables editor — admin can add/remove placeholder tokens. */}
   691→        <div className="space-y-1.5">
   692→          <Label className="text-xs font-medium text-muted-foreground">Variables</Label>
   693→          <div className="flex flex-wrap items-center gap-1.5 rounded-md border border-border/50 bg-muted/30 p-2 min-h-[2.5rem]">
   694→            {vars.length === 0 && (
   695→              <span className="text-xs italic text-muted-foreground">No variables yet — add one below.</span>
   696→            )}
   697→            {vars.map((v) => (
   698→              <span
   699→                key={v}
   700→                className="inline-flex items-center gap-1 rounded border border-border/50 bg-background px-1.5 py-0.5 font-mono text-xs shadow-premium-sm"
   701→              >
   702→                <code>{`{{${v}}}`}</code>
   703→                <button
   704→                  type="button"
   705→                  onClick={() => removeVar(v)}
   706→                  className="text-[10px] leading-none text-muted-foreground transition-colors hover:text-destructive"
   707→                  title={`Remove {{${v}}}`}
   708→                >
   709→                  ✕
   710→                </button>
   711→              </span>
   712→            ))}
   713→          </div>
   714→          <div className="flex items-center gap-2">
   715→            <Input
   716→              value={varsDraft}
   717→              onChange={(e) => setVarsDraft(e.target.value)}
   718→              onKeyDown={(e) => {
   719→                if (e.key === "Enter" || e.key === ",") {
   720→                  e.preventDefault();
   721→                  commitVarsDraft();
   722→                }
   723→              }}
   724→              placeholder="Add variable, e.g. orderNumber (Enter or comma to add)"
   725→              className="h-8 font-mono text-xs"
   726→            />
   727→            <Button
   728→              type="button"
   729→              size="sm"
   730→              variant="outline"
   731→              className="h-8 gap-1"
   732→              onClick={commitVarsDraft}
   733→              disabled={!varsDraft.trim()}
   734→            >
   735→              <Plus className="size-3.5" /> Add
   736→            </Button>
   737→          </div>
   738→          <p className="text-[11px] text-muted-foreground">
   739→            Use these tokens in the subject/body as <code>{`{{token}}`}</code>. Press Enter or comma to add.
   740→          </p>
   741→        </div>
   742→        {/* Footer — active toggle + actions, sticky to the card bottom. */}
   743→        <div className="flex items-center justify-between rounded-lg border border-border/50 bg-muted/20 px-3 py-2.5">
   744→          <label className="flex cursor-pointer items-center gap-2">
   745→            <Switch checked={isActive} onCheckedChange={setIsActive} />
   746→            <span className="flex items-center gap-1 text-xs font-medium">
   747→              {isActive
   748→                ? <ToggleRight className="size-3.5 text-emerald-600" />
   749→                : <ToggleLeft className="size-3.5 text-muted-foreground" />}
   750→              {isActive ? "Active" : "Inactive"}
   751→            </span>
   752→          </label>
   753→          <div className="flex items-center gap-2">
   754→            <Button size="sm" variant="ghost" className="text-destructive hover:bg-destructive/5" onClick={del} title="Delete template">
   755→              <Trash2 className="size-3.5" />
   756→            </Button>
   757→            <Button size="sm" onClick={save} disabled={saving} className="btn-premium gap-1.5">
   758→              {saving ? <Loader2 className="size-3.5 animate-spin" /> : <Save className="size-3.5" />}
   759→              Save
   760→            </Button>
   761→          </div>
   762→        </div>
   763→      </CardContent>
   764→    </Card>
   765→  );
   766→}
