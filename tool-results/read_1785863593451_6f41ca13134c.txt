     1→// ============================================================================
     2→// File: src/components/shared/product-card.tsx
     3→// Purpose: Reusable product card used in HomeView, ShopView, related products,
     4→//          search results. Shows image, name, brand, MRP strike + selling price,
     5→//          discount %, prescription badge, add-to-cart button, wishlist heart,
     6→//          compare toggle, and a quick-add button.
     7→// Role: Single visual treatment for product tiles across the customer site.
     8→// ============================================================================
     9→
    10→"use client";
    11→
    12→import { Card } from "@/components/ui/card";
    13→import { Button } from "@/components/ui/button";
    14→import { Badge } from "@/components/ui/badge";
    15→import {
    16→  ShoppingCart,
    17→  FileText,
    18→  Star,
    19→  Loader2,
    20→  Plus,
    21→  Zap,
    22→  Bell,
    23→  GitCompare,
    24→  Sparkles,
    25→  X,
    26→} from "lucide-react";
    27→import { useState, memo, useRef } from "react";
    28→import { useMutation, useQueryClient } from "@tanstack/react-query";
    29→import {
    30→  api,
    31→  qk,
    32→  Product,
    33→  ProductRecommendationsResponse,
    34→} from "@/components/customer/api";
    35→import { useUI } from "@/lib/store";
    36→import { useCompare } from "@/components/customer/use-compare";
    37→import { ProductImage } from "./product-image";
    38→import { WishlistButton } from "./wishlist-button";
    39→import { ShareButton } from "./share-button";
    40→import { toast } from "sonner";
    41→import { formatCurrency } from "@/lib/format";
    42→import { motion, AnimatePresence } from "framer-motion";
    43→import { cn } from "@/lib/utils";
    44→
    45→export const ProductCard = memo(function ProductCard({ product }: { product: Product }) {
    46→  const navigate = useUI((s) => s.navigate);
    47→  const qc = useQueryClient();
    48→  const { has, toggle, isFull, count } = useCompare();
    49→  const [adding, setAdding] = useState(false);
    50→  const [hovered, setHovered] = useState(false);
    51→  const inStock = (Number(product.stock) || 0) > 0;
    52→  const inCompare = has(product.id);
    53→
    54→  // Null-safe numeric access — wishlist/cache may return partial product objects.
    55→  // baseDiscountPct is the admin-confirmed discount % the customer should see
    56→  // (it auto-follows (MRP - Selling) / MRP * 100 by default; admin can override).
    57→  const mrp = Number(product.mrp) || 0;
    58→  const sellingPrice = Number(product.sellingPrice) || 0;
    59→  const discountPct =
    60→    mrp > sellingPrice
    61→      ? Math.round(((mrp - sellingPrice) / mrp) * 100)
    62→      : Math.round(Number(product.baseDiscountPct) || 0);
    63→
    64→  const addToCartMutation = useMutation({
    65→    mutationFn: () => api.post("/api/cart/add", { productId: product.id, qty: 1 }),
    66→    onMutate: () => setAdding(true),
    67→    onSuccess: () => {
    68→      qc.invalidateQueries({ queryKey: qk.cart });
    69→      toast.success("Added to cart", { description: product.name });
    70→      // Fire-and-forget: fetch complementary products and show a
    71→      // non-intrusive "Frequently bought together" suggestion toast.
    72→      showRecommendationToast();
    73→    },
    74→    onError: (e: Error) => toast.error(e.message),
    75→    onSettled: () => setAdding(false),
    76→  });
    77→
    78→  // Recommendation toast — uses the existing /api/catalog/recommendations/[productId]
    79→  // endpoint (powered by src/lib/recommendation-engine.ts). Picks the top 2
    80→  // complementary (frequentlyBought) items that are in stock and NOT the same
    81→  // as the product just added. Shows a compact JSX toast with thumbnails +
    82→  // "Add" buttons. Auto-dismisses after 6 seconds.
    83→  //
    84→  // Throttling: a ref tracks the timestamp of the last toast shown so we don't
    85→  // spam the customer when they rapid-fire add multiple products. The toast is
    86→  // suppressed for 8 seconds after each show.
    87→  const lastToastAtRef = useRef(0);
    88→  const addRecommendedMutation = useMutation({
    89→    mutationFn: (productId: string) =>
    90→      api.post("/api/cart/add", { productId, qty: 1 }),
    91→    onSuccess: () => {
    92→      qc.invalidateQueries({ queryKey: qk.cart });
    93→      toast.success("Added to cart");
    94→    },
    95→    onError: (e: Error) => toast.error(e.message),
    96→  });
    97→
    98→  const showRecommendationToast = async () => {
    99→    // Throttle: at most one recommendation toast every 8 seconds.
   100→    const now = Date.now();
   101→    if (now - lastToastAtRef.current < 8000) return;
   102→    lastToastAtRef.current = now;
   103→
   104→    try {
   105→      const recs = await api<ProductRecommendationsResponse>(
   106→        `/api/catalog/recommendations/${encodeURIComponent(product.id)}`
   107→      );
   108→      const picks = (recs.frequentlyBought ?? [])
   109→        .filter((p) => Number(p.stock) > 0 && p.id !== product.id)
   110→        .slice(0, 2);
   111→      if (picks.length === 0) {
   112→        // Fallback to related if no complementary items exist.
   113→        const relatedPicks = (recs.related ?? [])
   114→          .filter((p) => Number(p.stock) > 0 && p.id !== product.id)
   115→          .slice(0, 2);
   116→        if (relatedPicks.length === 0) return;
   117→        renderRecommendationToast(relatedPicks, "You might also like");
   118→        return;
   119→      }
   120→      renderRecommendationToast(picks, "Frequently bought together");
   121→    } catch {
   122→      // Silently fail — recommendations are a nice-to-have, not critical.
   123→    }
   124→  };
   125→
   126→  const renderRecommendationToast = (
   127→    picks: Product[],
   128→    heading: string
   129→  ) => {
   130→    const toastId = `rec-${product.id}-${Date.now()}`;
   131→    toast.custom(
   132→      (t) => (
   133→        <div
   134→          className="pointer-events-auto w-full max-w-sm rounded-xl border border-emerald-100 bg-white p-3 shadow-lg dark:border-emerald-900/40 dark:bg-card"
   135→          data-testid="recommendation-toast"
   136→        >
   137→          <div className="mb-2 flex items-center gap-2">
   138→            <div className="flex size-7 items-center justify-center rounded-md bg-emerald-100 dark:bg-emerald-900/40">
   139→              <Sparkles className="size-3.5 text-emerald-600 dark:text-emerald-400" />
   140→            </div>
   141→            <p className="text-xs font-bold uppercase tracking-wide text-foreground">
   142→              {heading}
   143→            </p>
   144→            <button
   145→              onClick={() => toast.dismiss(t)}
   146→              className="ml-auto rounded-md p-1 text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   147→              aria-label="Dismiss"
   148→            >
   149→              <X className="size-3.5" />
   150→            </button>
   151→          </div>
   152→          <div className="space-y-2">
   153→            {picks.map((p) => {
   154→              const sp = Number(p.sellingPrice) || 0;
   155→              const mrp = Number(p.mrp) || 0;
   156→              const addingThis = addRecommendedMutation.isPending
   157→                && addRecommendedMutation.variables === p.id;
   158→              return (
   159→                <div
   160→                  key={p.id}
   161→                  className="flex items-center gap-2.5 rounded-lg border border-border/60 bg-muted/30 p-2"
   162→                >
   163→                  <button
   164→                    onClick={() => {
   165→                      navigate({ name: "product", productId: p.id, slug: p.slug });
   166→                      toast.dismiss(t);
   167→                    }}
   168→                    className="size-12 shrink-0 overflow-hidden rounded-md bg-background"
   169→                    aria-label={`View ${p.name}`}
   170→                  >
   171→                    <ProductImage
   172→                      name={p.name}
   173→                      brandName={p.brand?.name}
   174→                      primaryImage={p.primaryImage}
   175→                      images={p.images}
   176→                      size="sm"
   177→                      className="!h-full !w-full !text-base"
   178→                    />
   179→                  </button>
   180→                  <div className="min-w-0 flex-1">
   181→                    <p className="line-clamp-2 break-words text-xs font-semibold leading-tight text-foreground">
   182→                      {p.name}
   183→                    </p>
   184→                    <div className="mt-0.5 flex items-baseline gap-1">
   185→                      <span className="text-xs font-bold text-foreground">
   186→                        {formatCurrency(sp)}
   187→                      </span>
   188→                      {mrp > sp && (
   189→                        <span className="text-[10px] text-muted-foreground line-through">
   190→                          {formatCurrency(mrp)}
   191→                        </span>
   192→                      )}
   193→                    </div>
   194→                  </div>
   195→                  <button
   196→                    onClick={() => {
   197→                      addRecommendedMutation.mutate(p.id);
   198→                    }}
   199→                    disabled={addingThis}
   200→                    className="flex shrink-0 items-center gap-1 rounded-md bg-emerald-600 px-2.5 py-1.5 text-[11px] font-semibold text-white shadow-sm transition-colors hover:bg-emerald-700 disabled:opacity-60"
   201→                    aria-label={`Add ${p.name} to cart`}
   202→                  >
   203→                    {addingThis ? (
   204→                      <Loader2 className="size-3 animate-spin" />
   205→                    ) : (
   206→                      <Plus className="size-3" />
   207→                    )}
   208→                    Add
   209→                  </button>
   210→                </div>
   211→              );
   212→            })}
   213→          </div>
   214→        </div>
   215→      ),
   216→      {
   217→        id: toastId,
   218→        duration: 6000,
   219→        // Position bottom-right on desktop, top on mobile so it doesn't cover
   220→        // the sticky mobile CTA bar at the bottom.
   221→        position: typeof window !== "undefined" && window.innerWidth < 640
   222→          ? "top-center"
   223→          : "bottom-right",
   224→      }
   225→    );
   226→  };
   227→
   228→  const go = () => navigate({ name: "product", productId: product.id, slug: product.slug });
   229→
   230→  const onAdd = async (e: React.MouseEvent) => {
   231→    e.stopPropagation();
   232→    // Optimistically check auth via the me query cache; the API will 401 if not authed
   233→    const me = qc.getQueryData<{ id: string } | null>(qk.me);
   234→    if (!me) {
   235→      toast.info("Please login to add items to your cart");
   236→      navigate({ name: "auth", mode: "login" });
   237→      return;
   238→    }
   239→    addToCartMutation.mutate();
   240→  };
   241→
   242→  const onCompareToggle = (e: React.MouseEvent) => {
   243→    e.stopPropagation();
   244→    if (inCompare) {
   245→      toggle({ id: product.id, name: product.name, slug: product.slug, sellingPrice: Number(product.sellingPrice), mrp: Number(product.mrp), primaryImage: product.primaryImage, brandName: product.brand?.name, prescriptionRequired: product.prescriptionRequired, stock: Number(product.stock) || 0, composition: product.composition });
   246→      toast.success("Removed from compare");
   247→      return;
   248→    }
   249→    if (isFull) {
   250→      toast.info("Compare tray is full", { description: "Remove a product to add another (max 4)." });
   251→      return;
   252→    }
   253→    toggle({ id: product.id, name: product.name, slug: product.slug, sellingPrice: Number(product.sellingPrice), mrp: Number(product.mrp), primaryImage: product.primaryImage, brandName: product.brand?.name, prescriptionRequired: product.prescriptionRequired, stock: Number(product.stock) || 0, composition: product.composition });
   254→    toast.success("Added to compare", { description: `${count + 1}/4 products selected` });
   255→  };
   256→
   257→  return (
   258→    <div className="h-full">
   259→      <Card
   260→        onClick={go}
   261→        onMouseEnter={() => setHovered(true)}
   262→        onMouseLeave={() => setHovered(false)}
   263→        className={cn(
   264→          "group relative h-full cursor-pointer gap-0 overflow-hidden p-0 py-0",
   265→          "border-border/40 rounded-2xl",
   266→          "transition-all duration-300 ease-out",
   267→          "hover:-translate-y-1 hover:border-primary/25",
   268→          "hover:shadow-lg hover:shadow-emerald-100/50 dark:hover:shadow-emerald-950/30",
   269→          "shadow-sm dark:shadow-none",
   270→          "active:scale-95"
   271→        )}
   272→      >
   273→        {/* Image — clean white background, subtle zoom on hover */}
   274→        <div className="relative aspect-square overflow-hidden bg-gradient-to-br from-accent/20 to-accent/5">
   275→          <ProductImage
   276→            name={product.name}
   277→            brandName={product.brand?.name}
   278→            primaryImage={product.primaryImage}
   279→            images={product.images}
   280→            size="xl"
   281→            className="!h-full !w-full !text-6xl transition-transform duration-500 ease-out group-hover:scale-105"
   282→          />
   283→
   284→          {/* Subtle gradient overlay on hover */}
   285→          <div
   286→            className={cn(
   287→              "pointer-events-none absolute inset-0 bg-gradient-to-t from-black/20 via-transparent to-transparent transition-opacity duration-300",
   288→              hovered ? "opacity-100" : "opacity-0"
   289→            )}
   290→          />
   291→
   292→          {/* Discount badge — clean pill */}
   293→          {discountPct > 0 && (
   294→            <div className="absolute left-2.5 top-2.5">
   295→              <span className="inline-flex items-center rounded-full bg-emerald-600 px-2 py-0.5 text-[10px] font-bold text-white shadow-md">
   296→                {discountPct}% OFF
   297→              </span>
   298→            </div>
   299→          )}
   300→
   301→          {/* Product feature badges */}
   302→          {discountPct === 0 && product.isBestSeller && (
   303→            <div className="absolute left-2.5 top-2.5">
   304→              <span className="inline-flex items-center rounded-full bg-amber-500 px-2 py-0.5 text-[10px] font-bold text-white shadow-md">
   305→                ★ Best Seller
   306→              </span>
   307→            </div>
   308→          )}
   309→          {discountPct === 0 && !product.isBestSeller && product.isTrending && (
   310→            <div className="absolute left-2.5 top-2.5">
   311→              <span className="inline-flex items-center rounded-full bg-rose-500 px-2 py-0.5 text-[10px] font-bold text-white shadow-md">
   312→                🔥 Trending
   313→              </span>
   314→            </div>
   315→          )}
   316→
   317→          {/* Wishlist heart (top-right) — pulses subtly on card hover */}
   318→          <div
   319→            className={cn(
   320→              "absolute right-1.5 top-1.5 transition-transform duration-300",
   321→              hovered && "animate-pulse"
   322→            )}
   323→          >
   324→            <WishlistButton productId={product.id} />
   325→          </div>
   326→
   327→          {/* Share button (top-right, below wishlist heart) — slides in on
   328→              hover. Small icon-only button that opens the premium ShareButton
   329→              dialog. Uses a glassmorphism chip that matches the WishlistButton
   330→              visual language so the two stacked actions feel cohesive.
   331→              stopPropagation prevents the card click from firing. */}
   332→          <AnimatePresence>
   333→            {hovered && (
   334→              <motion.div
   335→                key="share-toggle"
   336→                initial={{ opacity: 0, y: -8, scale: 0.85 }}
   337→                animate={{ opacity: 1, y: 0, scale: 1 }}
   338→                exit={{ opacity: 0, y: -8, scale: 0.85 }}
   339→                transition={{ duration: 0.2, ease: "easeOut" }}
   340→                className="absolute right-1.5 top-11"
   341→                onClick={(e) => e.stopPropagation()}
   342→              >
   343→                <ShareButton
   344→                  productName={product.name}
   345→                  slug={product.slug}
   346→                  productId={product.id}
   347→                  tagline={product.shortDescription || undefined}
   348→                  variant="icon"
   349→                  className="size-8 rounded-full bg-white/90 p-0 shadow-sm ring-1 ring-black/5 backdrop-blur-md hover:bg-white hover:shadow-md hover:ring-black/10 dark:bg-black/60 dark:ring-white/10 dark:hover:bg-black/80 dark:hover:ring-white/20"
   350→                />
   351→              </motion.div>
   352→            )}
   353→          </AnimatePresence>
   354→
   355→          {/* Compare toggle (top-left, below discount badge) — appears on
   356→              hover. Filled state shows the compare icon in emerald. */}
   357→          <AnimatePresence>
   358→            {hovered && (
   359→              <motion.button
   360→                key="compare-toggle"
   361→                initial={{ opacity: 0, x: -12 }}
   362→                animate={{ opacity: 1, x: 0 }}
   363→                exit={{ opacity: 0, x: -12 }}
   364→                transition={{ duration: 0.2, ease: "easeOut" }}
   365→                onClick={onCompareToggle}
   366→                aria-label={inCompare ? "Remove from compare" : "Add to compare"}
   367→                className={cn(
   368→                  "absolute left-1.5 flex size-8 items-center justify-center rounded-full shadow-md ring-1 backdrop-blur transition-colors",
   369→                  discountPct > 0 ? "top-9" : "top-1.5",
   370→                  inCompare
   371→                    ? "bg-emerald-600 text-white ring-emerald-700/50"
   372→                    : "bg-white/90 text-foreground ring-black/10 hover:bg-white"
   373→                )}
   374→              >
   375→                <GitCompare className="size-4" />
   376→              </motion.button>
   377→            )}
   378→          </AnimatePresence>
   379→
   380→          {/* Prescription badge (bottom-left, so it doesn't clash with the heart) */}
   381→          {product.prescriptionRequired && (
   382→            <Badge
   383→              variant="secondary"
   384→              className="absolute bottom-2 left-2 gap-1 bg-amber-100 text-amber-800 hover:bg-amber-100"
   385→            >
   386→              <FileText className="size-3" />
   387→              Rx
   388→            </Badge>
   389→          )}
   390→
   391→          {/* Quick Add floating button — slides up from the bottom-right of the
   392→              image area on hover. Calls the same add-to-cart logic as the bottom CTA.
   393→              Only rendered for in-stock products. */}
   394→          <AnimatePresence>
   395→            {hovered && inStock && (
   396→              <motion.button
   397→                key="quick-add"
   398→                initial={{ opacity: 0, y: 16 }}
   399→                animate={{ opacity: 1, y: 0 }}
   400→                exit={{ opacity: 0, y: 16 }}
   401→                transition={{ duration: 0.18, ease: "easeOut" }}
   402→                whileTap={{ scale: 0.95 }}
   403→                onClick={onAdd}
   404→                disabled={adding}
   405→                aria-label="Quick add to cart"
   406→                className="absolute bottom-2 right-2 flex items-center gap-1.5 rounded-full bg-gradient-to-r from-emerald-600 to-teal-600 px-3 py-1.5 text-xs font-semibold text-white shadow-lg shadow-emerald-600/30 ring-1 ring-white/30 transition-shadow hover:shadow-emerald-600/50 disabled:opacity-70"
   407→              >
   408→                {adding ? (
   409→                  <Loader2 className="size-3.5 animate-spin" />
   410→                ) : (
   411→                  <Plus className="size-3.5" />
   412→                )}
   413→                Add
   414→              </motion.button>
   415→            )}
   416→          </AnimatePresence>
   417→
   418→          {/* Out of stock overlay — prominent red badge + subtle gray veil.
   419→              Disabled card click would be too aggressive (customers still
   420→              want to view details), so we keep the card clickable but
   421→              visually mark it as unavailable. */}
   422→          {(Number(product.stock) || 0) <= 0 && (
   423→            <>
   424→              {/* Subtle gray veil over the image */}
   425→              <div className="absolute inset-0 bg-foreground/40 backdrop-grayscale backdrop-blur-[1px]" />
   426→              {/* Prominent red "Out of Stock" badge */}
   427→              <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2">
   428→                <span className="flex items-center gap-1 rounded-full bg-rose-600 px-3 py-1 text-xs font-bold uppercase tracking-wide text-white shadow-lg ring-2 ring-white/60">
   429→                  Out of Stock
   430→                </span>
   431→              </div>
   432→            </>
   433→          )}
   434→
   435→          {/* Low-stock urgency badge — shows when stock is between 1 and 10.
   436→              Creates a sense of urgency to encourage purchase. */}
   437→          {inStock && (Number(product.stock) || 0) > 0 && (Number(product.stock) || 0) <= 10 && (
   438→            <div className="absolute bottom-2 right-2">
   439→              <span className="flex items-center gap-1 rounded-full bg-amber-500/95 px-2 py-0.5 text-[10px] font-bold text-white shadow-sm backdrop-blur-sm">
   440→                <Zap className="size-2.5" />
   441→                Only {Math.max(1, Math.floor(Number(product.stock)))} left
   442→              </span>
   443→            </div>
   444→          )}
   445→        </div>
   446→
   447→        {/* Body — `min-w-0` is critical: without it, the flex child's intrinsic
   448→            min-width:auto would let long product names blow past the card width
   449→            and break the grid layout on mobile. Combined with `break-words` on
   450→            the title, this guarantees 2-line truncation regardless of length.
   451→            Premium consistent spacing: `gap-2` between elements + uniform `p-4`
   452→            padding across all breakpoints for visual consistency.
   453→            `flex-1 flex-col` + `mt-auto` on the CTA ensures all cards in a row
   454→            have equal height with the button pinned to the bottom — fixing the
   455→            uneven spacing issue in the Pharmacist Recommended section. */}
   456→        <div className="flex min-w-0 flex-1 flex-col gap-2 p-4">
   457→          {product.brand && (
   458→            <span className="truncate text-[10px] font-semibold uppercase tracking-wider text-emerald-600 dark:text-emerald-400 xl:text-[9px]">
   459→              {product.brand.name}
   460→            </span>
   461→          )}
   462→          <h3 className="line-clamp-2 break-words text-sm font-semibold leading-snug text-foreground group-hover:text-primary xl:text-xs">
   463→            {product.name}
   464→          </h3>
   465→          {/* Reserve the composition line height even when empty so cards
   466→              with and without composition align. Use a min-height spacer. */}
   467→          <p className="line-clamp-1 break-words text-xs text-muted-foreground xl:text-[11px] min-h-[1rem]">
   468→            {product.composition || "\u00A0"}
   469→          </p>
   470→
   471→          {/* Rating — 5-star display with numeric rating + review count.
   472→              Null-safe against partial cache objects. Reserve height so
   473→              cards without ratings don't collapse and cause uneven layout. */}
   474→          <div className="mt-0.5 flex min-h-[1rem] items-center gap-1.5">
   475→            {(Number(product.reviewCount) || 0) > 0 ? (
   476→              <>
   477→                <div className="flex items-center gap-0.5">
   478→                  {[1, 2, 3, 4, 5].map((n) => {
   479→                    const rating = Number(product.avgRating) || 0;
   480→                    return (
   481→                      <Star
   482→                        key={n}
   483→                        className={cn(
   484→                          "size-3",
   485→                          n <= Math.floor(rating)
   486→                            ? "fill-amber-400 text-amber-400"
   487→                            : n <= Math.round(rating)
   488→                            ? "fill-amber-200 text-amber-400"
   489→                            : "fill-muted text-muted-foreground/40"
   490→                        )}
   491→                      />
   492→                    );
   493→                  })}
   494→                </div>
   495→                <span className="text-xs font-medium text-foreground">
   496→                  {(Number(product.avgRating) || 0).toFixed(1)}
   497→                </span>
   498→                <span className="text-xs text-muted-foreground">
   499→                  ({product.reviewCount})
   500→                </span>
   501→              </>
   502→            ) : (
   503→              <span className="text-[10px] text-muted-foreground/50">New arrival</span>
   504→            )}
   505→          </div>
   506→
   507→          {/* Price — selling price larger and bolder, MRP strikethrough smaller. */}
   508→          <div className="mt-1 flex items-baseline gap-2">
   509→            <span className="text-lg font-bold tracking-tight text-foreground xl:text-base">
   510→              {formatCurrency(sellingPrice)}
   511→            </span>
   512→            {mrp > sellingPrice && (
   513→              <span className="text-xs font-medium text-muted-foreground line-through xl:text-[11px]">
   514→                {formatCurrency(mrp)}
   515→              </span>
   516→            )}
   517→            {mrp > sellingPrice && (
   518→              <span className="ml-auto text-[10px] font-semibold text-emerald-600 dark:text-emerald-400 xl:hidden">
   519→                Save {formatCurrency(mrp - sellingPrice)}
   520→              </span>
   521→            )}
   522→          </div>
   523→
   524→          {/* Availability + delivery indicator — unified row that adapts to
   525→              stock level. Always rendered with min-height so cards align. */}
   526→          <div className="mt-0.5 flex min-h-[1rem] items-center gap-1.5 text-[10px]">
   527→            {inStock ? (
   528→              <>
   529→                <span
   530→                  className={cn(
   531→                    "inline-flex items-center gap-1 font-medium",
   532→                    (Number(product.stock) || 0) <= 10
   533→                      ? "text-amber-700 dark:text-amber-300"
   534→                      : "text-emerald-700 dark:text-emerald-300"
   535→                  )}
   536→                >
   537→                  <span
   538→                    className={cn(
   539→                      "size-1.5 rounded-full",
   540→                      (Number(product.stock) || 0) <= 10
   541→                        ? "bg-amber-500"
   542→                        : "bg-emerald-500"
   543→                    )}
   544→                  />
   545→                  {(Number(product.stock) || 0) <= 10
   546→                    ? `Only ${Math.max(1, Math.floor(Number(product.stock)))} left`
   547→                    : "In stock"}
   548→                </span>
   549→                <span className="text-muted-foreground/60">·</span>
   550→                <span className="inline-flex items-center gap-0.5 text-amber-700 dark:text-amber-300">
   551→                  <Zap className="size-2.5 text-amber-500" />
   552→                  30–40 min
   553→                </span>
   554→              </>
   555→            ) : (
   556→              <span className="inline-flex items-center gap-1 font-medium text-rose-600 dark:text-rose-400">
   557→                <span className="size-1.5 rounded-full bg-rose-500" />
   558→                Out of stock
   559→              </span>
   560→            )}
   561→          </div>
   562→
   563→          {/* Add to cart — OR — Notify Me when out of stock.
   564→              `mt-auto` pins this button to the bottom of the flex column so
   565→              all cards in a row have the button at the same vertical position,
   566→              eliminating the uneven spacing issue. Button is compact (h-8) so
   567→              it fits in 5- and 6-column rows on wide screens. */}
   568→          {inStock ? (
   569→            <Button
   570→              size="sm"
   571→              variant="outline"
   572→              onClick={onAdd}
   573→              disabled={adding}
   574→              className="btn-premium mt-auto h-8 w-full gap-1 border-primary/20 px-2 text-xs text-primary hover:bg-primary hover:text-primary-foreground hover:border-primary"
   575→            >
   576→              {adding ? (
   577→                <Loader2 className="size-3.5 animate-spin" />
   578→              ) : (
   579→                <>
   580→                  <ShoppingCart className="size-3.5" />
   581→                  <span className="xl:hidden">Add to cart</span>
   582→                  <span className="hidden xl:inline">Add</span>
   583→                </>
   584→              )}
   585→            </Button>
   586→          ) : (
   587→            <Button
   588→              size="sm"
   589→              variant="outline"
   590→              onClick={(e) => {
   591→                e.stopPropagation();
   592→                // Optimistically check auth; BackInStockButton handles the real subscription
   593→                const me = qc.getQueryData<{ id: string } | null>(qk.me);
   594→                if (!me) {
   595→                  toast.info("Please login to subscribe to stock alerts");
   596→                  navigate({ name: "auth", mode: "login" });
   597→                  return;
   598→                }
   599→                // Subscribe via the stock-subscriptions API
   600→                api.post("/api/stock-subscriptions", { productId: product.id })
   601→                  .then(() => {
   602→                    toast.success("Subscribed!", {
   603→                      description: `We'll email you when "${product.name}" is back in stock.`,
   604→                    });
   605→                    qc.invalidateQueries({ queryKey: qk.stockAlerts });
   606→                  })
   607→                  .catch((err: Error) => toast.error(err.message));
   608→              }}
   609→              className="mt-auto h-8 w-full gap-1.5 border-amber-300 bg-amber-50 px-2 text-xs text-amber-700 hover:bg-amber-100 hover:text-amber-800 dark:border-amber-700 dark:bg-amber-950/40 dark:text-amber-200 dark:hover:bg-amber-900/40"
   610→            >
   611→              <Bell className="size-3.5" />
   612→              Notify Me
   613→            </Button>
   614→          )}
   615→        </div>
   616→      </Card>
   617→    </div>
   618→  );
   619→});
   620→

   624→      <div className="aspect-square skeleton-premium" />
   625→      <div className="space-y-2 p-4">
   626→        <div className="h-3 w-1/3 skeleton-premium rounded" />
   627→        <div className="h-4 w-3/4 skeleton-premium rounded" />
   628→        <div className="h-3 w-1/2 skeleton-premium rounded" />
   629→        <div className="h-5 w-1/4 skeleton-premium rounded" />
   630→        <div className="h-8 w-full skeleton-premium rounded" />
   631→      </div>
   632→    </Card>
