   453→   453→  // --------------------- Print invoices (hidden iframe per invoice) ---------------------
   454→   454→  async function printInvoices() {
   455→   455→    if (selectedIds.size === 0) return;
   456→   456→    const ids = Array.from(selectedIds);
   457→   457→    toast.info(`Preparing ${ids.length} invoice(s) for printing...`);
   458→   458→    for (const id of ids) {
   459→   459→      try {
   460→   460→        const buf = await api.raw(`/api/admin/orders/${id}/invoice`);
   461→   461→        const blob = new Blob([buf], { type: "application/pdf" });
   462→   462→        const url = URL.createObjectURL(blob);
   463→   463→        const iframe = document.createElement("iframe");
   464→   464→        iframe.style.display = "none";
   465→   465→        iframe.src = url;
   466→   466→        document.body.appendChild(iframe);
   467→   467→        iframe.onload = () => {
   468→   468→          try {
   469→   469→            iframe.contentWindow?.focus();
   470→   470→            iframe.contentWindow?.print();
   471→   471→          } catch (e) {
   472→   472→            console.error("[invoices] print failed", e);
   473→   473→          }
   474→   474→          setTimeout(() => {
   475→   475→            URL.revokeObjectURL(url);
   476→   476→            iframe.remove();
   477→   477→          }, 5000);
   478→   478→        };
   479→   479→      } catch (e: any) {
   480→   480→        toast.error(`Failed to fetch invoice for order ${id.slice(-6)}`);
   481→   481→      }
   482→   482→      await new Promise((r) => setTimeout(r, 800));
   483→   483→    }
   484→   484→  }

  1851→  1851→/** Build a compact page-number range like 1 … 4 5 [6] 7 8 … 20. */
  1852→  1852→function getPageRange(current: number, total: number): (number | "...")[] {
  1853→  1853→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1);
  1854→  1854→  if (current <= 4) return [1, 2, 3, 4, 5, "...", total];
  1855→  1855→  if (current >= total - 3)
  1856→  1856→    return [1, "...", total - 4, total - 3, total - 2, total - 1, total];
  1857→  1857→  return [1, "...", current - 1, current, current + 1, "...", total];
  1858→  1858→}
