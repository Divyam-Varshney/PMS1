     1→// ============================================================================
     2→// File: src/components/customer/device-registration-wizard.tsx
     3→// Purpose: New Device Registration Wizard (Phase 40). Replaces the old
     4→//          `notification-onboarding.tsx` with a 4-step guided flow inspired
     5→//          by Amazon / Flipkart / Blinkit / Swiggy onboarding.
     6→//
     7→//  Steps:
     8→//    1. WELCOME       — explain benefits, two buttons: "Turn On" / "Skip"
     9→//    2. PERMISSION    — request browser permission, handle deny gracefully
    10→//    3. REGISTERING   — show progress as we register SW + create subscription
    11→//                       + POST to backend + send welcome push
    12→//    4. TEST_RESULT   — show whether the welcome push was delivered
    13→//    5. DONE          — "You're All Set!" success screen
    14→//
    15→//  When the wizard should NOT appear (checked on mount + on auth change):
    16→//    • Not authenticated
    17→//    • Push API not supported by the browser
    18→//    • POST /api/device-registrations/validate returns shouldShowWizard=false
    19→//
    20→//  When the wizard SHOULD appear:
    21→//    • New device (no DeviceRegistration row)
    22→//    • status="pending" (started but not completed)
    23→//    • status="completed" but permission revoked / subscription expired
    24→//
    25→//  Skip behaviour:
    26→//    • POST /api/device-registrations/skip with the device info.
    27→//    • Wizard never re-appears on this device (status="skipped").
    28→//    • Customer can still enable from Profile → Settings → App Notifications.
    29→//
    30→//  The existing App Notification Center + automatic notification system are
    31→//  untouched. This wizard ONLY orchestrates the onboarding UX.
    32→// ============================================================================

    66→type WizardStep = "welcome" | "permission" | "registering" | "registering_failed" | "test_result" | "done" | "denied";
    67→
    68→interface StepProgress {
    69→  step: string;
    70→  label: string;
    71→  status: "pending" | "active" | "done" | "failed";
    72→}
    73→
    74→const BENEFITS = [
    75→  { icon: Package, title: "Order Updates", desc: "Real-time alerts when your order is confirmed, packed, or delivered." },
    76→  { icon: Truck, title: "Delivery Status", desc: "Know the moment your delivery is on its way to your doorstep." },
    77→  { icon: FileText, title: "Prescription Updates", desc: "Get notified when your prescription is reviewed and approved." },
    78→  { icon: Tag, title: "Exclusive Offers", desc: "Be the first to know about deals, vouchers, and seasonal discounts." },
    79→  { icon: Megaphone, title: "Pharmacy Announcements", desc: "Important updates about your account, medicine stock, and store hours." },
    80→];
    81→
    82→export function DeviceRegistrationWizard({
    83→  isAuthenticated,
    84→}: {
    85→  isAuthenticated: boolean;
    86→}) {
    87→  const qc = useQueryClient();
    88→  const [open, setOpen] = useState(false);
    89→  const [step, setStep] = useState<WizardStep>("welcome");
    90→  const [busy, setBusy] = useState(false);
    91→  const [pushSupported, setPushSupported] = useState(true);
    92→  const [testPushSent, setTestPushSent] = useState(false);
    93→  const [deviceLabel, setDeviceLabel] = useState<string>("");
    94→  const [steps, setSteps] = useState<StepProgress[]>([]);
    95→  const validatedRef = useRef(false);
    96→
    97→  // ─── Mount: detect Push API support ────────────────────────────────────
    98→  useEffect(() => {
    99→    if (typeof window === "undefined") return;
   100→    const supported =
   101→      "serviceWorker" in navigator &&
   102→      "PushManager" in window &&
   103→      "Notification" in window;
   104→    setPushSupported(supported);
   105→  }, []);
   106→
   107→  // ─── Auth change: check whether the wizard should appear ────────────────
   108→  useEffect(() => {
   109→    if (!isAuthenticated || !pushSupported) return;
   110→    if (validatedRef.current) return;
   111→    validatedRef.current = true;
   112→
   113→    let cancelled = false;
   114→    const check = async () => {
   115→      try {
   116→        const deviceInfo = getDeviceInfo();
   117→        setDeviceLabel(deviceInfo.deviceLabel);
   118→
   119→        // Check whether the browser still has permission + a local subscription.
   120→        const hasBrowserPermission = Notification.permission === "granted";
   121→        let hasLocalSubscription = false;
   122→        if (hasBrowserPermission && typeof window.__ensurePushReady === "function") {
   123→          try {
   124→            const reg = await window.__ensurePushReady();
   125→            if (reg) {
   126→              const sub = await reg.pushManager.getSubscription();
   127→              hasLocalSubscription = !!sub;
   128→            }
   129→          } catch {}
   130→        }
   131→
   132→        // POST /api/device-registrations/validate
   133→        const res = await fetch("/api/device-registrations/validate", {
   134→          method: "POST",
   135→          headers: { "Content-Type": "application/json" },
   136→          body: JSON.stringify({
   137→            deviceId: deviceInfo.deviceId,
   138→            hasBrowserPermission,
   139→            hasLocalSubscription,
   140→          }),
   141→        });
   142→        if (!res.ok) return;
   143→        const json = await res.json();
   144→        const data = json.data;
   145→        if (!data) return;
   146→
   147→        if (data.shouldShowWizard && !cancelled) {
   148→          // Delay so it doesn't appear instantly on page load — gives the
   149→          // page time to settle.
   150→          setTimeout(() => {
   151→            if (!cancelled) {
   152→              setStep(data.reason === "incomplete" ? "welcome" : "welcome");
   153→              setOpen(true);
   154→            }
   155→          }, 2000);
   156→        }
   157→      } catch {
   158→        // Best-effort — if anything fails, just don't show the wizard.
   159→      }
   160→    };
   161→
   162→    check();
   163→    return () => { cancelled = true; };
   164→  }, [isAuthenticated, pushSupported]);
   165→
   166→  // ─── Skip handler ───────────────────────────────────────────────────────
   167→  const handleSkip = useCallback(async () => {
   168→    setBusy(true);
   169→    try {
   170→      const info = getDeviceInfo();
   171→      await fetch("/api/device-registrations/skip", {
   172→        method: "POST",
   173→        headers: { "Content-Type": "application/json" },
   174→        body: JSON.stringify({
   175→          deviceId: info.deviceId,
   176→          deviceLabel: info.deviceLabel,
   177→          browserName: info.browserName,
   178→          osName: info.osName,
   179→          deviceType: info.deviceType,
   180→        }),
   181→      });
   182→    } catch {}
   183→    setOpen(false);
   184→    setBusy(false);
   185→  }, []);
   186→
   187→  // ─── Start the wizard: request permission ───────────────────────────────
   188→  const handleStart = useCallback(async () => {
   189→    setBusy(true);
   190→    setStep("permission");
   191→    try {
   192→      const perm = await Notification.requestPermission();
   193→      if (perm !== "granted") {
   194→        setStep("denied");
   195→        setBusy(false);
   196→        return;
   197→      }
   198→      // Permission granted — proceed to registration.
   199→      await runRegistration();
   200→    } catch (e) {
   201→      console.error("[wizard] permission request failed:", e);
   202→      setStep("denied");
   203→      setBusy(false);
   204→    }
   205→  }, []);
   206→
   207→  // ─── Registration: SW + subscription + backend + welcome push ───────────
   208→  const runRegistration = useCallback(async () => {
   209→    setBusy(true);
   210→    setStep("registering");
   211→
   212→    const info = getDeviceInfo();
   213→    const stepList: StepProgress[] = [
   214→      { step: "sw", label: "Activating Service Worker", status: "pending" },
   215→      { step: "subscription", label: "Creating Push Subscription", status: "pending" },
   216→      { step: "backend", label: "Registering Device on Server", status: "pending" },
   217→      { step: "test", label: "Sending Welcome Notification", status: "pending" },
   218→    ];
   219→    setSteps(stepList);
   220→
   221→    try {
   222→      // Step 1: ensure SW is registered + active.
   223→      stepList[0].status = "active";
   224→      setSteps([...stepList]);
   225→
   226→      let reg: ServiceWorkerRegistration | null = null;
   227→      if (typeof window.__ensurePushReady === "function") {
   228→        reg = await window.__ensurePushReady();
   229→      } else {
   230→        reg = await navigator.serviceWorker.ready;
   231→      }
   232→      if (!reg) throw new Error("Service Worker activation failed");
   233→      stepList[0].status = "done";
   234→      setSteps([...stepList]);
   235→
   236→      // Step 2: create push subscription (or reuse existing).
   237→      stepList[1].status = "active";
   238→      setSteps([...stepList]);
   239→
   240→      let subscription = await reg.pushManager.getSubscription();
   241→      if (!subscription) {
   242→        const vapidRes = await fetch("/api/push/vapid-public");
   243→        const vapidJson = await vapidRes.json();
   244→        if (!vapidJson.data?.publicKey) {
   245→          throw new Error("Push notifications are not configured on the server.");
   246→        }
   247→        const applicationServerKey = urlBase64ToUint8Array(vapidJson.data.publicKey);
   248→        subscription = await reg.pushManager.subscribe({
   249→          userVisibleOnly: true,
   250→          applicationServerKey: applicationServerKey as BufferSource,
   251→        });
   252→      }
   253→      stepList[1].status = "done";
   254→      setSteps([...stepList]);
   255→
   256→      // Step 3: POST subscription to backend (saves PushSubscription + ensures
   257→      // AppNotifPreference exists).
   258→      stepList[2].status = "active";
   259→      setSteps([...stepList]);
   260→
   261→      const sub = subscription.toJSON();
   262→      const subscribeRes = await fetch("/api/push/subscribe", {
   263→        method: "POST",
   264→        headers: { "Content-Type": "application/json" },
   265→        body: JSON.stringify({
   266→          endpoint: sub.endpoint,
   267→          keys: sub.keys,
   268→          userAgent: info.userAgent.slice(0, 500),
   269→        }),
   270→      });
   271→      if (!subscribeRes.ok) {
   272→        const errData = await subscribeRes.json().catch(() => ({}));
   273→        throw new Error(errData.error || "Failed to save subscription");
   274→      }
   275→      stepList[2].status = "done";
   276→      setSteps([...stepList]);
   277→
   278→      // Step 4: register the device + send welcome push via the dedicated endpoint.
   279→      stepList[3].status = "active";
   280→      setSteps([...stepList]);
   281→
   282→      const registerRes = await fetch("/api/device-registrations/register", {
   283→        method: "POST",
   284→        headers: { "Content-Type": "application/json" },
   285→        body: JSON.stringify({
   286→          deviceId: info.deviceId,
   287→          deviceLabel: info.deviceLabel,
   288→          browserName: info.browserName,
   289→          osName: info.osName,
   290→          deviceType: info.deviceType,
   291→          pushEndpoint: sub.endpoint,
   292→        }),
   293→      });
   294→      if (!registerRes.ok) {
   295→        const errData = await registerRes.json().catch(() => ({}));
   296→        throw new Error(errData.error || "Failed to register device");
   297→      }
   298→      const regJson = await registerRes.json();
   299→      const regData = regJson.data;
   300→      const welcomed = regData?.welcomed === true;
   301→      setTestPushSent(welcomed);
   302→      stepList[3].status = welcomed ? "done" : "failed";
   303→      setSteps([...stepList]);
   304→
   305→      // Invalidate preferences query so the toggle in settings reflects the
   306→      // new "enabled" state.
   307→      qc.invalidateQueries({ queryKey: ["customer", "app-notif-prefs"] });
   308→
   309→      // Move to the test result step.
   310→      setStep("test_result");
   311→      setBusy(false);
   312→    } catch (e: any) {
   313→      console.error("[wizard] registration failed:", e);
   314→      // Mark any active step as failed.
   315→      setSteps((prev) =>
   316→        prev.map((s) =>
   317→          s.status === "active" ? { ...s, status: "failed" } : s
   318→        )
   319→      );
   320→      // Move to the failed step — shows retry + skip options.
   321→      setStep("registering_failed");
   322→      setBusy(false);
   323→    }
   324→  }, [qc]);
   325→
   326→  // ─── Done handler ───────────────────────────────────────────────────────
   327→  const handleDone = useCallback(() => {
   328→    setOpen(false);
   329→    // Reset to welcome in case the wizard is ever re-opened.
   330→    setTimeout(() => setStep("welcome"), 300);
   331→  }, []);
   332→
   333→  // ─── Retry handler ──────────────────────────────────────────────────────
   334→  const handleRetry = useCallback(async () => {
   335→    await runRegistration();
   336→  }, [runRegistration]);
   337→
   338→  if (!isAuthenticated || !pushSupported) return null;
   339→
   340→  return (
   341→    <Dialog
   342→      open={open}
   343→      onOpenChange={(o) => {
   344→        // Prevent closing while busy (registration in progress) — the user
   345→        // must wait for the async flow to finish or fail.
   346→        if (!o && busy) return;
   347→        if (!o) handleSkip();
   348→      }}
   349→    >
   350→      <DialogContent
   351→        className="max-w-md overflow-hidden p-0 sm:rounded-2xl"
   352→        onInteractOutside={(e) => e.preventDefault()}
   353→        onEscapeKeyDown={(e) => {
   354→          // Only allow escape when not busy + on the welcome/denied/done steps.
   355→          if (busy || step === "registering" || step === "permission") {
   356→            e.preventDefault();
   357→          }
   358→        }}
   359→      >
   360→        {/* ── Step: WELCOME ─────────────────────────────────────────────── */}
   361→        {step === "welcome" && (
   362→          <WelcomeStep
   363→            deviceLabel={deviceLabel}
   364→            onTurnOn={handleStart}
   365→            onSkip={handleSkip}
   366→            busy={busy}
   367→          />
   368→        )}
   369→
   370→        {/* ── Step: PERMISSION (waiting on browser prompt) ──────────────── */}
   371→        {step === "permission" && (
   372→          <CenteredMessage
   373→            icon={<Bell className="size-8" />}
   374→            title="Waiting for permission"
   375→            message="Your browser is asking for permission to send notifications. Click 'Allow' to continue."
   376→            spinner
   377→          />
   378→        )}
   379→
   380→        {/* ── Step: REGISTERING (progress) ──────────────────────────────── */}
   381→        {step === "registering" && (
   382→          <RegisteringStep steps={steps} />
   383→        )}
   384→
   385→        {/* ── Step: REGISTERING FAILED ──────────────────────────────────── */}
   386→        {step === "registering_failed" && (
   387→          <RegisteringFailedStep
   388→            steps={steps}
   389→            onRetry={handleRetry}
   390→            onSkip={handleSkip}
   391→            busy={busy}
   392→          />
   393→        )}
   394→
   395→        {/* ── Step: TEST_RESULT ─────────────────────────────────────────── */}
   396→        {step === "test_result" && (
   397→          <TestResultStep
   398→            sent={testPushSent}
   399→            onContinue={() => setStep("done")}
   400→            onRetry={handleRetry}
   401→            busy={busy}
   402→          />
   403→        )}
   404→
   405→        {/* ── Step: DENIED (permission blocked) ─────────────────────────── */}
   406→        {step === "denied" && (
   407→          <DeniedStep onSkip={handleSkip} onRetry={handleStart} />
   408→        )}
   409→
   410→        {/* ── Step: DONE ────────────────────────────────────────────────── */}
   411→        {step === "done" && (
   412→          <DoneStep onDone={handleDone} deviceLabel={deviceLabel} />
   413→        )}
   414→      </DialogContent>
   415→    </Dialog>
   416→  );
   417→}

   517→function RegisteringStep({ steps }: { steps: StepProgress[] }) {
   518→  return (
   519→    <>
   520→      <div className="bg-gradient-to-br from-emerald-600 via-emerald-700 to-teal-700 px-6 pb-6 pt-7 text-white">
   521→        <div className="mb-3 flex size-14 items-center justify-center rounded-2xl bg-white/15 backdrop-blur-sm ring-1 ring-white/30">
   522→          <Loader2 className="size-7 animate-spin" />
   523→        </div>
   524→        <h2 className="text-xl font-bold leading-tight">Setting up notifications</h2>
   525→        <p className="mt-1.5 text-sm text-emerald-50/90">
   526→          We're registering this device. This only takes a few seconds.
   527→        </p>
   528→      </div>
   529→      <div className="space-y-3 px-6 py-5">
   530→        {steps.map((s) => (
   531→          <div key={s.step} className="flex items-center gap-3">
   532→            <div className="flex size-8 shrink-0 items-center justify-center rounded-full">
   533→              {s.status === "done" ? (
   534→                <CheckCircle2 className="size-5 text-emerald-600" />
   535→              ) : s.status === "active" ? (
   536→                <Loader2 className="size-5 animate-spin text-emerald-600" />
   537→              ) : s.status === "failed" ? (
   538→                <XCircle className="size-5 text-rose-500" />
   539→              ) : (
   540→                <div className="size-5 rounded-full border-2 border-muted" />
   541→              )}
   542→            </div>
   543→            <p className={`text-sm ${s.status === "done" ? "font-medium text-foreground" : s.status === "active" ? "font-medium text-foreground" : "text-muted-foreground"}`}>
   544→              {s.label}
   545→            </p>
   546→          </div>
   547→        ))}
   548→      </div>
   549→    </>
   550→  );
   551→}

   553→function RegisteringFailedStep({
   554→  steps,
   555→  onRetry,
   556→  onSkip,
   557→  busy,
   558→}: {
   559→  steps: StepProgress[];
   560→  onRetry: () => void;
   561→  onSkip: () => void;
   562→  busy: boolean;
   563→}) {
   564→  return (
   565→    <>
   566→      <div className="bg-gradient-to-br from-amber-500 to-orange-600 px-6 pb-6 pt-7 text-white">
   567→        <div className="mb-3 flex size-14 items-center justify-center rounded-2xl bg-white/15 backdrop-blur-sm ring-1 ring-white/30">
   568→          <AlertCircle className="size-7" />
   569→        </div>

   799→function urlBase64ToUint8Array(base64Url: string): Uint8Array {
   800→  const padding = "=".repeat((4 - (base64Url.length % 4)) % 4);
   801→  const base64 = (base64Url + padding).replace(/-/g, "+").replace(/_/g, "/");
   802→  const raw = atob(base64);
   803→  const arr = new Uint8Array(raw.length);
   804→  for (let i = 0; i < raw.length; i++) {
   805→    arr[i] = raw.charCodeAt(i);
   806→  }
   807→  return arr;
   808→}
