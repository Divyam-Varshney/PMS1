   569→function resolveHeroIcon(name?: string) {
   570→  if (!name) return null;
   571→  return HERO_ICONS[name] || null;
   572→}
   573→
   574→/** Static component that renders a hero icon by name. Uses createElement
   575→ *  rather than JSX <Icon /> to satisfy the react-hooks/static-components
   576→ *  lint rule (the icon reference is a runtime lookup, not a static def). */
   577→function HeroIcon({ name, className }: { name?: string; className?: string }) {
   578→  const Icon = resolveHeroIcon(name);
   579→  if (!Icon) return null;
   580→  return createElement(Icon, { className });
   581→}

   583→/** Resolve a CTA "url" which may be a customer view name (e.g. "shop") or a
   584→ *  full URL (e.g. "https://..."). Internal view names navigate via the SPA
   585→ *  router; full URLs open in a new tab. */
   586→function go(navigate: (v: any) => void, url?: string) {
   587→  if (!url) return;
   588→  if (/^https?:\/\//i.test(url) || url.startsWith("mailto:") || url.startsWith("tel:")) {
   589→    window.open(url, "_blank", "noopener,noreferrer");
   590→    return;
   591→  }
   592→  navigate({ name: url as any } as any);
   593→}

   595→/** Check if a scheduled item (promo banner / announcement) is currently
   596→ *  visible based on its optional start/end ISO date window. */
   597→function isWithinSchedule(start?: string, end?: string): boolean {
   598→  const now = Date.now();
   599→  if (start && now < new Date(start).getTime()) return false;
   600→  if (end && now > new Date(end).getTime()) return false;
   601→  return true;
   602→}
   603→
   604→/** Hook that returns true after the component has mounted on the client.
   605→ *  Used to defer time-sensitive rendering (schedule checks, countdowns) until
   606→ *  after hydration so the SSR HTML and the first client render always match.
   607→ *  Uses useSyncExternalStore (the React-recommended mounted pattern) to avoid
   608→ *  the set-state-in-effect lint rule. */
   609→function useMounted(): boolean {
   610→  return useSyncExternalStore(
   611→    () => () => {},
   612→    () => true,   // client snapshot — always true once hydrated
   613→    () => false,  // server snapshot — always false during SSR
   614→  );
   615→}

   715→// — Hero Banner — the main hero section. ----------------------------------
   716→function HeroBanner({
   717→  hero,
   718→  heroOffer,
   719→  isStoreOpen,
   720→  store,
   721→  heroQuery,
   722→  setHeroQuery,
   723→  onHeroSearch,
   724→  navigate,
   725→  anim,
   726→}: {
   727→  hero?: HeroConfig;
   728→  heroOffer?: any;
   729→  isStoreOpen: boolean;
   730→  store?: PublicSettings["store"];
   731→  heroQuery: string;
   732→  setHeroQuery: (v: string) => void;
   733→  onHeroSearch: (e: React.FormEvent) => void;
   734→  navigate: (v: any) => void;
   735→  anim: boolean;
   736→}) {
   737→  // Admin "hero offer" override — simple banner with the offer's content.
   738→  if (heroOffer) {
   739→    return (
   740→      <section
   741→        className="relative overflow-hidden rounded-3xl px-6 py-12 shadow-2xl sm:px-12 sm:py-16"

   746→          {heroOffer.subtitle && <p className="mt-3 text-sm opacity-90 sm:text-lg">{heroOffer.subtitle}</p>}

   748→            <Button

   750→              className="mt-6 gap-1.5 bg-white text-foreground hover:bg-white/90"

   757→        </div>
   758→      </section>
   759→    );
   760→  }
   761→
   762→  if (!hero) return null;
   763→
   764→  // Resolve background gradient.
   765→  const preset = STYLE_PRESETS[hero.stylePreset] || STYLE_PRESETS.emerald;
   766→  const gFrom = hero.stylePreset === "custom" ? hero.gradientFrom : preset.from;
   767→  const gVia = hero.stylePreset === "custom" ? hero.gradientVia : preset.via;
   768→  const gTo = hero.stylePreset === "custom" ? hero.gradientTo : preset.to;
   769→
   770→  const layout = hero.layout || "split-left";
   771→  // An image is "available" when the image feature is enabled AND a URL is set.
   772→  const hasImage = hero.bgImageEnabled && !!(hero.bgImageDesktop || hero.bgImageMobile);
   773→  // Desktop/tablet side image (split layouts) and full-bg image.
   774→  const sideImage = hero.bgImageDesktop || hero.bgImageMobile || "";
   775→  // Mobile always falls back to the mobile image (or desktop if no mobile set).
   776→  const mobileImage = hero.bgImageMobile || hero.bgImageDesktop || "";
   777→
   778→  // Layout semantics:
   779→  //  - split-left / split-right: 2-col grid (text + side image) when an image
   780→  //    is available; otherwise a single text column. Image shows on md+.
   781→  //  - centered: single text column, centered, no side image.
   782→  //  - full-bg: image fills the whole hero as a background with a dark
   783→  //    overlay; text sits on top (alignment per contentAlign).
   784→  const isSplit = layout === "split-left" || layout === "split-right";
   785→  const isFullBg = layout === "full-bg";
   786→  const isCenteredLayout = layout === "centered";
   787→  // Show the side-by-side image only for split layouts that have an image.
   788→  const hasSplitImage = isSplit && hasImage;
   789→  // Full-bg background image shows for full-bg layout (or any layout with an
   790→  // image when not split — treated as a background).
   791→  const hasFullBgImage = isFullBg && hasImage;
   792→
   793→  const popularSearches = hero.popularSearches
   794→    ? hero.popularSearches.split(",").map((s) => s.trim()).filter(Boolean)
   795→    : [];
   796→
   797→  // Background base style for the <section>.
   798→  const bgStyle: React.CSSProperties = {};
   799→  if (hasFullBgImage) {
   800→    // Full-bg: solid base color (image layer drawn separately on top).
   801→    bgStyle.backgroundColor = gFrom;
   802→  } else if (hero.gradientEnabled) {
   803→    bgStyle.background = `linear-gradient(135deg, ${gFrom}, ${gVia}, ${gTo})`;
   804→  } else {
   805→    bgStyle.background = hero.bgColor || gFrom;
   806→  }
   807→
   808→  // Pattern overlay style.
   809→  const patternBg =
   810→    hero.bgPattern === "dots"
   811→      ? "radial-gradient(circle, rgba(255,255,255,0.6) 1px, transparent 1px)"
   812→      : hero.bgPattern === "grid"
   813→      ? "linear-gradient(rgba(255,255,255,0.4) 1px, transparent 1px), linear-gradient(90deg, rgba(255,255,255,0.4) 1px, transparent 1px)"
   814→      : hero.bgPattern === "waves"
   815→      ? "repeating-linear-gradient(45deg, rgba(255,255,255,0.15) 0, rgba(255,255,255,0.15) 1px, transparent 1px, transparent 12px)"
   816→      : "none";
   817→  const patternSize =
   818→    hero.bgPattern === "grid" ? "24px 24px" : hero.bgPattern === "dots" ? "22px 22px" : undefined;
   819→
   820→  const radius = hero.borderRadius ?? 24;
   821→  const heightClass = HERO_HEIGHTS[hero.height || "lg"];
   822→
   823→  // Grid template per layout (only activates at md+ for split layouts).
   824→  const gridCols = hasSplitImage
   825→    ? layout === "split-right"
   826→      ? "md:grid-cols-[1fr_1.1fr]"
   827→      : "md:grid-cols-[1.1fr_1fr]"
   828→    : "grid-cols-1";
   829→
   830→  // contentAlign controls text alignment WITHIN the text column. It does NOT
   831→  // override the layout choice. Centered layout always centers; full-bg and
   832→  // split layouts honor the contentAlign setting.
   833→  const isTextCentered = isCenteredLayout || hero.contentAlign === "center";
   834→
   835→  return (
   836→    <section
   837→      className={`relative overflow-hidden text-white shadow-2xl ${heightClass}`}
   838→      style={{ ...bgStyle, borderRadius: `${radius}px` }}
   839→    >
   840→      {/* Full-bg background image layer (desktop/tablet/mobile) */}
   841→      {hasFullBgImage && (
   842→        <div className="absolute inset-0 z-0">
   843→          {/* Desktop / tablet */}
   844→          <img
   845→            src={sideImage}
   846→            alt={hero.imageAltText || ""}
   847→            className="hidden h-full w-full object-cover sm:block"
   848→          />
   849→          {/* Mobile */}
   850→          <img
   851→            src={mobileImage}
   852→            alt={hero.imageAltText || ""}
   853→            className="block h-full w-full object-cover sm:hidden"
   854→          />
   855→          {/* Dark overlay for text legibility */}
   856→          <div
   857→            className="absolute inset-0"
   858→            style={{ background: `linear-gradient(135deg, rgba(4,78,55,${hero.bgOverlay / 100}), rgba(15,118,110,${hero.bgOverlay / 100}))` }}
   859→          />
   860→        </div>
   861→      )}
   862→
   863→      {/* Gradient depth overlay (when no full-bg image) */}
   864→      {!hasFullBgImage && hero.gradientEnabled && (
   865→        <div
   866→          className="absolute inset-0 z-0"
   867→          style={{ opacity: hero.bgOpacity / 100, background: `linear-gradient(135deg, ${gFrom}, ${gVia}, ${gTo})` }}
   868→        />
   869→      )}
   870→
   871→      {/* Pattern overlay */}
   872→      {hero.bgPattern !== "none" && (
   873→        <div
   874→          className="absolute inset-0 z-0 opacity-[0.12]"
   875→          style={{ backgroundImage: patternBg, backgroundSize: patternSize }}
   876→        />
   877→      )}
   878→
   879→      {/* Animated orbs */}
   880→      {anim && (
   881→        <>
   882→          <motion.div
   883→            className="absolute right-0 top-0 -mr-20 -mt-20 size-80 rounded-full bg-white/10 blur-3xl"
   884→            animate={{ y: [0, -16, 0], x: [0, 8, 0] }}
   885→            transition={{ duration: 8, repeat: Infinity, ease: "easeInOut" }}
   886→          />
   887→          <motion.div
   888→            className="absolute bottom-0 left-0 -mb-24 -ml-16 size-96 rounded-full bg-teal-300/20 blur-3xl"
   889→            animate={{ y: [0, 14, 0], x: [0, -8, 0] }}
   890→            transition={{ duration: 10, repeat: Infinity, ease: "easeInOut" }}
   891→          />
   892→        </>
   893→      )}
   894→      {/* Top sheen */}
   895→      <div className="absolute inset-x-0 top-0 z-0 h-px bg-gradient-to-r from-transparent via-white/40 to-transparent" />
   896→
   897→      {/* Content — relative z-10 so it sits above all background layers.
   898→          The grid is single-column on mobile and 2-column on md+ for split
   899→          layouts. For centered/full-bg, it stays single-column with the text
