   239→function NavList({ admin, searchQuery }: { admin: AdminInfo; searchQuery: string }) {
   240→  const view = useAdminStore((s) => s.view);
   241→  const navigate = useAdminStore((s) => s.navigate);
   242→  const [collapsedGroups, setCollapsedGroups] = useState<Set<string>>(new Set());
   243→
   244→  // Fetch lightweight badge counts
   245→  const { data: counts } = useQuery({
   246→    queryKey: ["admin-counts"],
   247→    queryFn: () => api.get<{ pendingOrders: number; lowStock: number; pendingPrescriptions: number; pendingManualRequests: number; unreadNotifications: number }>("/api/admin/counts"),
   248→    refetchInterval: 30 * 1000,
   249→  });
   250→
   251→  const badgeFor = (viewName: string): number | null => {
   252→    if (!counts) return null;
   253→    switch (viewName) {
   254→      case "orders": return counts.pendingOrders || null;
   255→      case "products": return counts.lowStock || null;
   256→      case "prescriptions": return counts.pendingPrescriptions || null;
   257→      case "manual-requests": return counts.pendingManualRequests || null;
   258→      default: return null;
   259→    }
   260→  };
   261→
   262→  const toggleGroup = (label: string) => {
   263→    setCollapsedGroups((prev) => {
   264→      const next = new Set(prev);
   265→      if (next.has(label)) next.delete(label);
   266→      else next.add(label);
   267→      return next;
   268→    });
   269→  };
   270→
   271→  const isVisible = (item: { permission: AdminPermissionKey }) => hasPermission(admin, item.permission);
   272→
   273→  // When searching, show all items flat (ignore group collapse)
   274→  const isSearching = searchQuery.trim().length > 0;
   275→  const query = searchQuery.toLowerCase().trim();
   276→
   277→  const visibleGroups = NAV_GROUPS.map((g) => ({
   278→    ...g,
   279→    items: g.items.filter((item) => {
   280→      if (!isVisible(item)) return false;
   281→      if (!isSearching) return true;
   282→      return item.label.toLowerCase().includes(query);
   283→    }),
   284→  })).filter((g) => g.items.length > 0);
   285→
   286→  // Dashboard is always visible (if permitted)
   287→  const showDashboard = hasPermission(admin, "dashboard") && (!isSearching || "dashboard".includes(query));
   288→
   289→  return (
   290→    <SidebarContent>
   291→      {/* Dashboard — always at top, outside groups */}
   292→      {showDashboard && (
   293→        <SidebarGroup>
   294→          <SidebarMenu>
   295→            <SidebarMenuItem>
   296→              <SidebarMenuButton
   297→                isActive={view.name === "dashboard"}
   298→                onClick={() => navigate({ name: "dashboard" })}
   299→                tooltip="Dashboard"
   300→                className="transition-premium"
   301→              >
   302→                <LayoutDashboard />
   303→                <span>Dashboard</span>
   304→              </SidebarMenuButton>
   305→            </SidebarMenuItem>
   306→          </SidebarMenu>
   307→        </SidebarGroup>
   308→      )}
   309→
   310→      {/* Collapsible nav groups */}
   311→      {visibleGroups.map((g) => {
   312→        const isCollapsed = collapsedGroups.has(g.label) && !isSearching;
   313→        return (
   314→          <Collapsible
   315→            key={g.label}
   316→            open={!isCollapsed}
   317→            // When searching, all groups are forced open so matches are visible.
   318→            // Don't call toggleGroup during search — it would silently mutate
   319→            // the collapsed state and cause unexpected expand/collapse when the
   320→            // search is cleared.
   321→            onOpenChange={() => !isSearching && toggleGroup(g.label)}
   322→          >
   323→            <SidebarGroup>
   324→              <CollapsibleTrigger asChild>
   325→                <SidebarGroupLabel className="cursor-pointer select-none hover:bg-sidebar-accent/60 transition-colors admin-section-label">
   326→                  <span>{g.label}</span>
   327→                  <ChevronRight className={`ml-auto size-3.5 transition-transform ${isCollapsed ? "" : "rotate-90"}`} />
   328→                </SidebarGroupLabel>
   329→              </CollapsibleTrigger>
   330→              <CollapsibleContent>
   331→                <SidebarMenu>
   332→                  {g.items.map((item) => {
   333→                    const isActive =
   334→                      view.name === item.view.name ||
   335→                      (item.view.name === "products" && view.name === "product-edit") ||
   336→                      (item.view.name === "orders" && view.name === "order-detail") ||
   337→                      (item.view.name === "prescriptions" && view.name === "prescription-detail") ||
   338→                      (item.view.name === "manual-requests" && view.name === "manual-request-detail") ||
   339→                      (item.view.name === "customers" && view.name === "customer-detail");
   340→                    return (
   341→                      <SidebarMenuItem key={item.label}>
   342→                        <SidebarMenuButton
   343→                          isActive={isActive}
   344→                          onClick={() => navigate(item.view)}
   345→                          tooltip={item.label}
   346→                          className="transition-premium"
   347→                        >
   348→                          <item.icon />
   349→                          <span>{item.label}</span>
   350→                          {badgeFor(item.view.name) !== null && (
   351→                            <Badge className="ml-auto h-5 min-w-5 justify-center px-1.5 text-[10px] font-bold bg-amber-500 text-white hover:bg-amber-500 animate-badge-pulse">
   352→                              {badgeFor(item.view.name)}
   353→                            </Badge>
   354→                          )}
   355→                        </SidebarMenuButton>
   356→                      </SidebarMenuItem>
   357→                    );
   358→                  })}
   359→                </SidebarMenu>
   360→              </CollapsibleContent>
   361→            </SidebarGroup>
   362→          </Collapsible>
   363→        );
   364→      })}
   365→
   366→      {/* No results when searching */}
   367→      {isSearching && visibleGroups.length === 0 && !showDashboard && (
   368→        <div className="px-4 py-8 text-center text-xs text-muted-foreground">
   369→          No navigation items match “{searchQuery}”
   370→        </div>
   371→      )}
   372→    </SidebarContent>
   373→  );
   374→}

   377→// A3: BREADCRUMB NAVIGATION
   378→// ---------------------------------------------------------------------------
   379→
   380→function Breadcrumbs({ viewName, navigate }: { viewName: string; navigate: (v: AdminView) => void }) {
   381→  const crumbs: Array<{ label: string; view?: AdminView }> = [];
   382→
   383→  // Always start with Dashboard
   384→  crumbs.push({ label: "Dashboard", view: { name: "dashboard" } });
   385→
   386→  // Add parent if this is a detail page
   387→  const parentName = BREADCRUMB_PARENT[viewName];
   388→  if (parentName) {
   389→    crumbs.push({ label: TITLE_MAP[parentName] || parentName, view: { name: parentName } as AdminView });
   390→  }
   391→
   392→  // Add current page (not clickable)
   393→  const currentLabel = TITLE_MAP[viewName] || viewName;
   394→  if (viewName !== "dashboard") {
   395→    crumbs.push({ label: currentLabel });
   396→  }
   397→
   398→  return (
   399→    <nav className="flex items-center gap-1 text-sm text-muted-foreground min-w-0">
   400→      {crumbs.map((crumb, i) => {
   401→        const isLast = i === crumbs.length - 1;
   402→        return (
   403→          <span key={i} className="flex items-center gap-1 min-w-0">
   404→            {i > 0 && <ChevronRight className="size-3.5 shrink-0 text-muted-foreground/50" />}
   405→            {crumb.view && !isLast ? (
   406→              <button
   407→                onClick={() => navigate(crumb.view!)}
   408→                className="hover:text-foreground transition-colors truncate"
   409→              >
   410→                {crumb.label}
   411→              </button>
   412→            ) : (
   413→              <span className={`truncate ${isLast ? "font-semibold text-foreground" : ""}`}>
   414→                {crumb.label}
   415→              </span>
   416→            )}
   417→          </span>
   418→        );
   419→      })}
   420→    </nav>
   421→  );
   422→}

   428→function CommandPalette({ open, onOpenChange, navigate }: { open: boolean; onOpenChange: (v: boolean) => void; navigate: (v: AdminView) => void }) {
   429→  const [query, setQuery] = useState("");
   430→
   431→  const filtered = useMemo(() => {
   432→    if (!query.trim()) return ALL_NAV_ITEMS;
   433→    const q = query.toLowerCase();
   434→    return ALL_NAV_ITEMS.filter((item) => item.label.toLowerCase().includes(q));
   435→  }, [query]);
   436→
   437→  const handleSelect = (view: AdminView) => {
   438→    navigate(view);
   439→    onOpenChange(false);
   440→    setQuery("");
   441→  };
   442→
   443→  return (
   444→    <Dialog open={open} onOpenChange={onOpenChange}>
   445→      <DialogContent className="p-0 gap-0 overflow-hidden max-w-lg shadow-premium-xl">
   446→        {/* Accessibility: DialogTitle is required by Radix Dialog for screen
   447→            reader support. Rendered visually-hidden (sr-only). */}
   448→        <DialogTitle className="sr-only">Command palette</DialogTitle>
   449→        <div className="flex items-center border-b border-border/70 px-4 bg-premium-gradient">
   450→          <Search className="size-4 mr-3 text-muted-foreground shrink-0" />
   451→          <Input
   452→            autoFocus
   453→            placeholder="Type a command or search..."
   454→            value={query}
   455→            onChange={(e) => setQuery(e.target.value)}
   456→            aria-label="Search navigation items"
   457→            className="h-12 border-0 px-0 focus-visible:ring-0 text-sm bg-transparent"
   458→            onKeyDown={(e) => {
   459→              if (e.key === "Enter" && filtered.length > 0) {
   460→                handleSelect(filtered[0].view);
   461→              }
   462→            }}
   463→          />
   464→          <kbd className="ml-2 shrink-0 rounded border bg-muted px-1.5 py-0.5 text-[10px] text-muted-foreground font-mono">ESC</kbd>
   465→        </div>
   466→        <div className="max-h-80 overflow-y-auto p-2 scrollbar-premium">
   467→          {filtered.length === 0 ? (
   468→            <div className="px-3 py-8 text-center text-sm text-muted-foreground">
   469→              No results found
   470→            </div>
   471→          ) : (
   472→            filtered.map((item) => (
   473→              <button
   474→                key={item.label}
   475→                onClick={() => handleSelect(item.view)}
   476→                className="flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm hover:bg-accent transition-premium text-left"
   477→              >
   478→                <span className="flex size-7 items-center justify-center rounded-md bg-primary/10 text-primary shrink-0">
   479→                  <item.icon className="size-3.5" />
   480→                </span>
   481→                <span className="flex-1 font-medium">{item.label}</span>
   482→                <span className="text-[10px] text-muted-foreground uppercase tracking-wide">{item.group}</span>
   483→              </button>
   484→            ))
   485→          )}
   486→        </div>
   487→      </DialogContent>
   488→    </Dialog>
   489→  );
   490→}

   495→
   496→function StoreOpenToggle() {
   497→  const qc = useQueryClient();
   498→  const { data, isLoading } = useQuery({
   499→    queryKey: ["admin-store-status"],
   500→    queryFn: () => api.get<{ open: boolean; openStatus: boolean }>("/api/admin/settings/store-status"),
   501→  });
   502→
   503→  async function toggle(next: boolean) {
   504→    const r = await run(
   505→      () => api.patch("/api/admin/settings/store-status", { openStatus: next }),
   506→      { success: next ? "Store opened" : "Store closed", error: "Failed to toggle" }
   507→    );
   508→    if (r) {
   509→      qc.invalidateQueries({ queryKey: ["admin-store-status"] });
   510→    }
   511→  }
   512→
   513→  const isOpen = !!data?.openStatus;
   514→  return (
   515→    <div className={`flex items-center gap-2 px-3 py-2 rounded-lg border transition-colors ${isOpen ? "border-emerald-200/60 bg-emerald-50/50 dark:bg-emerald-950/20 dark:border-emerald-800/40" : "border-border bg-muted/30"}`}>
   516→      <Store className={`size-4 ${isOpen ? "text-emerald-600 dark:text-emerald-400" : "text-muted-foreground"}`} />
   517→      <div className="flex-1 min-w-0">
   518→        <div className="text-xs font-medium">Store Status</div>
   519→        <div className={`text-[11px] ${isOpen ? "text-emerald-700 dark:text-emerald-300" : "text-muted-foreground"}`}>
   520→          {isLoading ? "Loading…" : data?.open ? "Open for orders" : "Closed"}
   521→        </div>
   522→      </div>
   523→      <Switch
   524→        checked={isOpen}
   525→        disabled={isLoading}
   526→        onCheckedChange={toggle}
   527→        aria-label="Toggle store open/close"
   528→      />
   529→    </div>
   530→  );
   531→}

   596→  const canGoBack = useAdminStore((s) => s.canGoBack());

   598→  // A2: Sidebar search state
   599→  const [searchQuery, setSearchQuery] = useState("");
   600→
   601→  // A4: Command palette state
   602→  const [cmdOpen, setCmdOpen] = useState(false);
   603→
   604→  // A4: Keyboard shortcut listener (Cmd+K / Ctrl+K)
   605→  useEffect(() => {
   606→    const handler = (e: KeyboardEvent) => {
   607→      if ((e.metaKey || e.ctrlKey) && e.key === "k") {
   608→        e.preventDefault();
   609→        setCmdOpen((v) => !v);
   610→      }
   611→    };
   612→    window.addEventListener("keydown", handler);
   613→    return () => window.removeEventListener("keydown", handler);
   614→  }, []);
   615→
   616→  // E7: Restore saved dark mode preference on mount
   617→  useEffect(() => {
   618→    const savedTheme = localStorage.getItem("admin-theme");
   619→    if (savedTheme === "dark") {
   620→      document.documentElement.classList.add("dark");
   621→    }
   622→  }, []);
   623→
   624→  // Current date for the topbar widget (B6: date widget)
   625→  const today = new Date();
   626→  const dateStr = today.toLocaleDateString("en-IN", { weekday: "short", day: "2-digit", month: "short" });

   629→    <SidebarProvider>

   639→            </div>
   640→          </div>
   641→        </SidebarHeader>
   642→
   643→        {/* A2: Sidebar Search Filter */}
   644→        <SidebarSearch value={searchQuery} onChange={setSearchQuery} />
   645→
   646→        <NavList admin={admin} searchQuery={searchQuery} />
   647→        <SidebarFooterArea admin={admin} onLogout={onLogout} />
   648→      </Sidebar>
   649→
   650→      <SidebarInset>
   651→        {/* Topbar with breadcrumbs + command palette trigger + date widget */}
   652→        <header className="sticky top-0 z-30 flex h-14 items-center gap-2 border-b border-border/70 bg-background/95 backdrop-blur-md px-4 shadow-premium-sm">

   658→          </div>
   659→
   660→          {/* A4: Command palette trigger button */}

   662→            variant="outline"
   663→            size="sm"

   666→          >
   667→            <Command className="size-3.5" />
   668→            <span className="text-xs">Search</span>

   673→          <div className="hidden md:flex items-center gap-1.5 text-xs text-muted-foreground">
   674→            <Clock className="size-3.5" />
   675→            <span>{dateStr}</span>

   677→
   678→          {/* E7: Dark mode toggle — `relative` on the button is required so the

   680→              sticky header ancestor. */}
   681→          <Button
   682→            variant="ghost"
   683→            size="icon"
   684→            className="relative size-8 hover:bg-accent transition-premium"
   685→            onClick={() => {
   686→              const isDark = document.documentElement.classList.contains("dark");
   687→              if (isDark) {
   688→                document.documentElement.classList.remove("dark");
   689→                localStorage.setItem("admin-theme", "light");
   690→              } else {
   691→                document.documentElement.classList.add("dark");
   692→                localStorage.setItem("admin-theme", "dark");
   693→              }
   694→            }}
   695→            title="Toggle dark mode"
   696→          >
   697→            <Sun className="size-4 rotate-0 scale-100 transition-all dark:-rotate-90 dark:scale-0" />
   698→            <Moon className="absolute size-4 rotate-90 scale-0 transition-all dark:rotate-0 dark:scale-100" />

   702→          <AdminNotificationBell />
   703→
   704→          {/* Mobile back button */}
   705→          {canGoBack && (
   706→            <Button
   707→              variant="ghost"
   708→              size="sm"
   709→              onClick={back}
   710→              className="sm:hidden text-muted-foreground"

   712→              Back

   714→          )}
   715→
   716→          {/* Role badge (desktop) */}
   717→          <Badge variant="outline" className="hidden md:flex bg-primary/10 text-primary border-primary/20 font-medium">
   718→            {roleLabel(admin.role)}
   719→          </Badge>
   720→
   721→          <Button variant="outline" size="sm" onClick={onLogout} className="hidden md:flex transition-premium">
   722→            <LogOut className="size-4" />
   723→          </Button>
   724→          <Button variant="ghost" size="icon" className="md:hidden" onClick={onLogout}>
   725→            <LogOut className="size-4" />
   726→          </Button>
   727→        </header>
   728→
   729→        {/* A6: Page transition animations — the key includes the view name AND
   730→            any id field so navigating between two records of the same type
   731→            (e.g. order-detail → different order-detail) re-triggers the
   732→            fade/slide animation. Covers all detail views: product-edit,
   733→            order-detail, prescription-detail, manual-request-detail,

   744→            >
   745→              {children}

   753→    </SidebarProvider>

   757→/** Wrapper to fetch admin from /me on mount; renders children only when logged in. */
   758→export function AdminGuard({
   759→  onLoggedOut,
   760→  children,
   761→}: {
   762→  onLoggedOut: () => void;
   763→  children: (admin: AdminInfo) => ReactNode;
   764→}) {
   765→  const [admin, setAdmin] = useState<AdminInfo | null>(null);
   766→  const [loading, setLoading] = useState(true);
   767→
   768→  useEffect(() => {
   769→    let mounted = true;
   770→    api
   771→      .get<AdminInfo>("/api/admin-auth/me")
   772→      .then((a) => mounted && setAdmin(a))
   773→      .catch(() => mounted && onLoggedOut())
   774→      .finally(() => mounted && setLoading(false));
   775→    return () => {
   776→      mounted = false;
   777→    };
   778→  }, [onLoggedOut]);
   779→
   780→  if (loading) {
   781→    return (
   782→      <div className="min-h-screen flex items-center justify-center bg-background">
   783→        <Loader2 className="size-8 animate-spin text-primary" />
   784→      </div>
   785→    );
   786→  }
   787→  if (!admin) return null;
   788→  return <>{children(admin)}</>;
   789→}

   792→  await run(() => api.post("/api/admin-auth/logout"), { success: "Signed out" });
   793→}
