   164→      description: "View and track your past orders",
   165→      icon: Package,
   166→      onClick: () => navigate({ name: "orders" }),
   167→      stat: me._count?.orders ?? 0,
   168→      tint: "bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300",
   169→    },
   170→    {

   205→      icon: UserIcon,

   207→      tint: "bg-teal-100 text-teal-700 dark:bg-teal-950/40 dark:text-teal-300",

   216→    {

   226→    <div className="mx-auto max-w-4xl px-4 py-6 sm:px-6 sm:py-8">

   295→              </span>
   296→            </div>
   297→          </div>

   305→          label="Orders"
   306→          value={me._count?.orders ?? 0}
   307→          tint="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300"
   308→          onClick={() => navigate({ name: "orders" })}
   309→        />

   311→          icon={Heart}
   312→          label="Wishlist"
   313→          value={wishlistCount}
   314→          tint="bg-rose-100 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300"
   315→          onClick={() => navigate({ name: "wishlist" })}

   320→          value={stockAlertCount}
   321→          tint="bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300"
   322→          onClick={() => navigate({ name: "stock-alerts" })}

   539→              const diffDays = Math.ceil(diffMs / (24 * 60 * 60 * 1000));
   540→              const isOverdue = diffDays < 0;
   541→              const isDueSoon = diffDays >= 0 && diffDays <= 7;

   598→            })}
   599→          </div>
