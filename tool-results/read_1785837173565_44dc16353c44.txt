     1→// ============================================================================
     2→// File: src/lib/ai-knowledge-layer.ts
     3→// Purpose: Pharmacy-specific knowledge layer for the PMS Assistant. This module
     4→//          is the "brain" of the assistant — it provides:
     5→//
     6→//            1. buildAssistantSystemPrompt()
     7→//               A rich system prompt that teaches the LLM about pharmacy
     8→//               practice: therapeutic categories, dosage forms, safety rules,
     9→//               composition vs brand vs generic, when to escalate to a doctor,
    10→//               and how to recommend PMS features (upload prescription, request
    11→//               a medicine, health bundles).
    12→//
    13→//            2. SYMPTOM_TO_PRODUCT_MAP
    14→//               Maps common symptoms (fever, cough, acidity, …) to the product
    15→//               keywords / generic names that should be searched in the catalog
    16→//               when the customer describes their symptoms instead of naming a
    17→//               specific medicine. This is what makes "I have a fever" return
    18→//               Paracetamol — instead of nothing.
    19→//
    20→//            3. expandQueryWithSymptoms()
    21→//               If the user's message contains a symptom phrase, the search
    22→//               query is enriched with the matching medicine keywords so the
    23→//               catalog search is more likely to surface the right products.
    24→//
    25→//            4. CATEGORY_KEYWORDS
    26→//               Maps category names (e.g. "Pain Relief") to a list of
    27→//               keywords that customers might use to search for them. Used by
    28→//               the assistant route to search the catalog by category name in
    29→//               addition to product name / composition.
    30→//
    31→//            5. buildAlternativeContext()
    32→//               A helper that turns a list of "alternative" products into a
    33→//               plain-text explanation of WHY each one is a suitable
    34→//               substitute, so the LLM can present them sensibly to the customer.
    35→//
    36→//            6. PHARMACY_FEATURE_CUES
    37→//               Rules for guiding the customer toward PMS features based on
    38→//               the assistant action (product found → suggest Related Products
    39→//               and Frequently Bought Together; prescription required → suggest
    40→//               Upload Prescription; product not found → suggest Medicine
    41→//               Request; health query → suggest Medical Bundles).
    42→//
    43→// Role: Pure functions + constants — no DB access. Imported by the
    44→//       /api/health-assistant route. Keeping the knowledge here (rather than
    45→//       inline in the route) means the pharmacy owner can review and extend
    46→//       the assistant's medical knowledge in one file.
    47→// ============================================================================
    48→

   392→// Helpers
   393→// ---------------------------------------------------------------------------
   394→
   395→/**
   396→ * Look at the user's message and return the symptom mapping(s) that match.
   397→ * Returns [] if no symptom phrases are detected.
   398→ *
   399→ * Used by the assistant route to enrich the catalog search query.
   400→ */
   401→export function matchSymptoms(message: string): SymptomMapping[] {
   402→  if (!message) return [];
   403→  const lower = message.toLowerCase();
   404→  const matched: SymptomMapping[] = [];
   405→  for (const mapping of SYMPTOM_TO_PRODUCT_MAP) {
   406→    if (mapping.symptoms.some((s) => lower.includes(s))) {
   407→      matched.push(mapping);
   408→    }
   409→  }
   410→  return matched;
   411→}
   412→
   413→/**
   414→ * Expand a user query with product keywords derived from any symptom phrases
   415→ * it contains. Returns the original query if no symptoms match.
   416→ *
   417→ * Example:
   418→ *   expandQueryWithSymptoms("I have fever and body pain")
   419→ *   → "I have fever and body pain paracetamol crocin dolo calpol thermometer cold compress ibuprofen diclofenac volini moov iodex"
   420→ */
   421→export function expandQueryWithSymptoms(query: string): string {
   422→  if (!query) return query;
   423→  const matches = matchSymptoms(query);
   424→  if (matches.length === 0) return query;
   425→  const extraKeywords = new Set<string>();
   426→  for (const m of matches) {
   427→    for (const kw of m.productKeywords) extraKeywords.add(kw);
   428→  }
   429→  if (extraKeywords.size === 0) return query;
   430→  return `${query} ${Array.from(extraKeywords).join(" ")}`;
   431→}
   432→
   433→/**
   434→ * Given a user query, return the matching category names (if any). Used to
   435→ * ALSO search the catalog by category name (e.g. "pain relief" matches the
   436→ * Pain Relief category even if no product name contains that phrase).
   437→ */
   438→export function matchCategoryKeywords(query: string): string[] {
   439→  if (!query) return [];
   440→  const lower = query.toLowerCase();
   441→  const matched: string[] = [];
   442→  for (const [categoryName, keywords] of Object.entries(CATEGORY_KEYWORDS)) {
   443→    if (lower.includes(categoryName) || keywords.some((kw) => lower.includes(kw))) {
   444→      matched.push(categoryName);
   445→    }
   446→  }
   447→  return matched;
   448→}

   450→// ---------------------------------------------------------------------------
   451→// 5. buildAssistantSystemPrompt
   452→// ---------------------------------------------------------------------------
   453→
   454→/**
   455→ * Build the system prompt for the PMS Assistant LLM call.
   456→ *
   457→ * This prompt encodes:
   458→ *   - The assistant's role and identity
   459→ *   - Store details (location, hours, delivery, payment)
   460→ *   - Response priority (catalog first, alternatives second, request third)
   461→ *   - Medical disclaimer + escalation rules
   462→ *   - Therapeutic category knowledge
   463→ *   - Dosage form explanations
   464→ *   - Safety rules (don't share prescriptions, complete antibiotic courses,
   465→ *     never self-medicate antibiotics, never recommend doses for Rx medicines)
   466→ *   - Composition / generic / brand awareness
   467→ *   - PMS feature guidance (when to nudge which feature)
   468→ *
   469→ * The product context (catalog matches) and any feature cues are appended at
   470→ * call-time by the route handler, NOT here.
   471→ */

   637→/**
   638→ * Look up the user's query against the BRAND_TO_GENERIC map and return any
   639→ * matching generic names. Used as a fallback when no exact catalog match is
   640→ * found, so we can still suggest the generic equivalent.
   641→ */
   642→export function lookupBrandToGeneric(query: string): string[] {
   643→  if (!query) return [];
   644→  const lower = query.toLowerCase();
   645→  const out: string[] = [];
   646→  const tokens = lower.split(/[^a-z0-9]+/g).filter((t) => t.length > 1);
   647→  for (const [brand, generics] of Object.entries(BRAND_TO_GENERIC)) {
   648→    // Match the brand as a whole token (not a substring — so "Dolo" doesn't
   649→    // match "Dolomite" etc.).
   650→    if (tokens.includes(brand)) {
   651→      for (const g of generics) if (!out.includes(g)) out.push(g);
   652→    }
   653→  }
   654→  return out;
   655→}

   657→/**
   658→ * Build a plain-text explanation of WHY each alternative was suggested.
   659→ * The LLM uses this to make the recommendation sound sensible to the customer
   660→ * ("Same active ingredient as X, but at a lower price" / "Same category — both
   661→ * treat fever and body ache").
   662→ */
   663→export function buildAlternativeContext(
   664→  alternatives: AlternativeProduct[],
   665→  context: { genericName?: string | null; categoryId?: string | null; categoryName?: string | null }
   666→): string {
   667→  if (alternatives.length === 0) return "";
   668→  const lines = alternatives.map((p, i) => {
   669→    const reasons: string[] = [];
   670→    if (context.genericName && p.genericName && p.genericName.toLowerCase() === context.genericName.toLowerCase()) {
   671→      reasons.push(`Same active ingredient (${p.genericName})`);
   672→    } else if (p.genericName) {
   673→      reasons.push(`Active ingredient: ${p.genericName}`);
   674→    }
   675→    if (context.categoryName) {
   676→      reasons.push(`Same category (${context.categoryName})`);
   677→    }
   678→    if (p.brandName) reasons.push(`Brand: ${p.brandName}`);
   679→    reasons.push(`₹${p.sellingPrice}`);
   680→    if (p.prescriptionRequired) reasons.push("Rx required");
   681→    if (p.stock <= 0) reasons.push("out of stock");
   682→    return `${i + 1}. ${p.name} — ${reasons.join(", ")}`;
   683→  });
   684→  return `\n\nSuggested alternatives (reference these by name and price):\n${lines.join("\n")}\n\nExplain briefly WHY each alternative is suitable (same active ingredient / same therapeutic category). Always remind the customer to consult a doctor or pharmacist before switching medicines.`;
   685→}

   687→/**
   688→ * Decide which pharmacy feature cues to include in the response, based on the
   689→ * assistant's action and whether the matched products are Rx / out of stock.
   690→ */
   691→export function pickFeatureCues(
   692→  action: "product_results" | "medicine_request" | "bundle_results" | "faq_answer" | "general_info",
   693→  context: {
   694→    hasProducts: boolean;
   695→    hasRxProduct: boolean;
   696→    hasOutOfStock: boolean;
   697→    hasSymptoms: boolean;
   698→    isHealthQuery: boolean;
   699→  }
   700→): PharmacyFeature[] {
   701→  const cues: PharmacyFeature[] = [];
   702→  if (action === "product_results" && context.hasProducts) {
   703→    cues.push("related_products", "frequently_bought");
   704→    if (context.hasRxProduct) cues.push("upload_prescription");
   705→    if (context.hasOutOfStock) cues.push("medicine_request");
   706→  } else if (action === "medicine_request") {
   707→    cues.push("medicine_request");
   708→    if (context.hasRxProduct) cues.push("upload_prescription");
   709→  } else if (action === "bundle_results") {
   710→    cues.push("health_bundle");
   711→  } else if (action === "general_info") {
   712→    if (context.hasSymptoms || context.isHealthQuery) {
   713→      cues.push("consult_doctor");
   714→      if (context.hasSymptoms) cues.push("health_bundle");
   715→    }
   716→  }
   717→  return cues;
   718→}
