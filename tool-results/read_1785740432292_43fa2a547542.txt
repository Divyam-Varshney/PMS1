    95→// ---------------------------------------------------------------------------
    96→// Key auto-generation: "Cash on Delivery" → "cod", "UPI" → "upi",
    97→// "Razorpay" → "razorpay", "QR Code" → "qc". Multi-word → initials joined;
    98→// single-word → lowercased whole word. Special-cased common terms so
    99→// "QR Code" → "qr" (more conventional than "qc").
   100→// ---------------------------------------------------------------------------
   101→function generateKey(label: string): string {
   102→  const trimmed = label.trim();
   103→  if (!trimmed) return "";
   104→  const words = trimmed.split(/\s+/).filter(Boolean);
   105→  if (words.length === 1) return words[0].toLowerCase();
   106→  // Common multi-word payment terms — use conventional short forms.
   107→  const lower = trimmed.toLowerCase();
   108→  if (lower.includes("qr")) return "qr";
   109→  if (lower.includes("upi")) return "upi";
   110→  if (lower.includes("cash") && lower.includes("delivery")) return "cod";
   111→  if (lower.includes("net banking")) return "netbanking";
   112→  return words.map((w) => w[0]).join("").toLowerCase();
   113→}

   115→/** Safe JSON parse of PaymentMethod.config — never throws. */
   116→function parseConfig<T = Record<string, unknown>>(raw?: string | null): T {
   117→  if (!raw) return {} as T;
   118→  try {
   119→    return JSON.parse(raw) as T;
   120→  } catch {
   121→    return {} as T;
   122→  }
   123→}

   141→  const [qrImage, setQrImage] = useState<string>("");
   142→  const [showSecret, setShowSecret] = useState(false);
   143→  const [testingRazorpay, setTestingRazorpay] = useState(false);
   144→  // When true, the key field auto-follows the label. Turned off when the
   145→  // admin manually edits the key field so the override survives.
   146→  const [autoKey, setAutoKey] = useState(true);
   147→  const [saving, setSaving] = useState(false);
   148→  const [togglingId, setTogglingId] = useState<string | null>(null);
   149→  const qrFileInputRef = useRef<HTMLInputElement>(null);

   151→  function openNew() {
   152→    setEditing(null);
   153→    setForm(EMPTY);
   154→    setRazorpayCreds({ keyId: "", keySecret: "" });
   155→    setQrImage("");
   156→    setShowSecret(false);
   157→    setAutoKey(true);
   158→    setOpen(true);
   159→  }

   161→  function openEdit(pm: PaymentMethod) {
   162→    setEditing(pm);
   163→    setForm({
   164→      key: pm.key,
   165→      label: pm.label,
   166→      description: pm.description || "",
   167→      icon: pm.icon || "",
   168→      gateway: pm.gateway || "",
   169→      displayOrder: pm.displayOrder,
   170→      isActive: pm.isActive,
   171→    });
   172→    // Load method-specific config from the JSON string.
   173→    if (pm.key === "razorpay") {
   174→      const cfg = parseConfig<RazorpayConfig>(pm.config);
   175→      setRazorpayCreds({ keyId: cfg.keyId ?? "", keySecret: cfg.keySecret ?? "" });
   176→    } else {
   177→      setRazorpayCreds({ keyId: "", keySecret: "" });
   178→    }
   179→    if (pm.key === "qr") {
   180→      const cfg = parseConfig<QrConfig>(pm.config);
   181→      setQrImage(cfg.qrImage ?? "");
   182→    } else {
   183→      setQrImage("");
   184→    }
   185→    setShowSecret(false);
   186→    // Existing records have an explicit key — don't auto-overwrite it.
   187→    setAutoKey(false);
   188→    setOpen(true);
   189→  }

   191→  function onLabelChange(value: string) {
   192→    setForm((f: any) => ({
   193→      ...f,
   194→      label: value,
   195→      ...(autoKey ? { key: generateKey(value) } : {}),
   196→    }));
   197→  }
   198→
   199→  function onKeyChange(value: string) {
   200→    setAutoKey(false);
   201→    // Force lowercase + strip whitespace (keys must be lowercase identifiers).
   202→    setForm((f: any) => ({ ...f, key: value.toLowerCase().replace(/\s+/g, "") }));
   203→  }

   205→  // Build the config object to send, based on the method key.
   206→  function buildConfig(): Record<string, string> | undefined {
   207→    if (form.key === "razorpay") {
   208→      const cfg: Record<string, string> = {};
   209→      if (razorpayCreds.keyId?.trim()) cfg.keyId = razorpayCreds.keyId.trim();
   210→      if (razorpayCreds.keySecret?.trim()) cfg.keySecret = razorpayCreds.keySecret.trim();
   211→      return Object.keys(cfg).length ? cfg : undefined;
   212→    }
   213→    if (form.key === "qr") {
   214→      const cfg: Record<string, string> = {};
   215→      if (qrImage) cfg.qrImage = qrImage;
   216→      return Object.keys(cfg).length ? cfg : undefined;
   217→    }
   218→    return undefined;
   219→  }

   221→  async function save() {
   222→    if (!form.label.trim()) {
   223→      toast.error("Label is required");
   224→      return;
   225→    }
   226→    if (!form.key.trim()) {
   227→      toast.error("Key is required");
   228→      return;
   229→    }
   230→    // Razorpay-specific validation — require both fields IF the admin is
   231→    // enabling the method OR entering any value. Allows saving an empty
   232→    // disabled Razorpay method (the seeded default).
   233→    if (form.key === "razorpay" && form.isActive) {
   234→      if (!razorpayCreds.keyId?.trim() || !razorpayCreds.keySecret?.trim()) {
   235→        toast.error("Razorpay Key ID and Key Secret are required to enable this method");
   236→        return;
   237→      }
   238→    }
   239→    setSaving(true);
   240→    const payload = {
   241→      label: form.label.trim(),
   242→      description: form.description.trim(),
   243→      icon: form.icon.trim(),
   244→      gateway: form.gateway.trim(),
   245→      config: buildConfig(),
   246→      displayOrder: parseInt(form.displayOrder, 10) || 0,
   247→      isActive: form.isActive,
   248→    };
   249→    const r = editing
   250→      ? await run(() => api.patch(`/api/admin/payment-methods/${editing.id}`, payload), {
   251→          success: "Payment method updated",
   252→          error: "Update failed",
   253→        })
   254→      : await run(
   255→          () => api.post("/api/admin/payment-methods", { key: form.key.trim(), ...payload }),
   256→          { success: "Payment method created", error: "Create failed" }
   257→        );
   258→    setSaving(false);
   259→    if (r) {
   260→      setOpen(false);
   261→      qc.invalidateQueries({ queryKey: ["admin-payment-methods"] });
   262→    }
   263→  }
   264→
   265→  async function del(pm: PaymentMethod) {
   266→    if (!confirm(`Delete payment method "${pm.label}"?`)) return;
   267→    const r = await run(() => api.del(`/api/admin/payment-methods/${pm.id}`), {
   268→      success: "Payment method deleted",
   269→      error: "Delete failed",
   270→    });
   271→    if (r) qc.invalidateQueries({ queryKey: ["admin-payment-methods"] });
   272→  }

   274→  async function quickToggle(pm: PaymentMethod, next: boolean) {
   275→    setTogglingId(pm.id);
   276→    const r = await run(
   277→      () => api.patch(`/api/admin/payment-methods/${pm.id}`, { isActive: next }),
   278→      { success: next ? "Method enabled" : "Method disabled", error: "Toggle failed", silent: true }
   279→    );
   280→    setTogglingId(null);
   281→    if (r) {
   282→      toast.success(next ? `${pm.label} enabled` : `${pm.label} disabled`);
   283→      qc.invalidateQueries({ queryKey: ["admin-payment-methods"] });
   284→    }
   285→  }

   287→  // --- QR image upload ------------------------------------------------------
   288→  async function onQrImagePicked(e: React.ChangeEvent<HTMLInputElement>) {
   289→    const file = e.target.files?.[0];
   290→    if (!file || !editing) return;
   291→    // Basic client-side validation — server re-validates.
   292→    if (!file.type.startsWith("image/")) {
   293→      toast.error("Please upload an image file");
   294→      return;
   295→    }
   296→    if (file.size > 4 * 1024 * 1024) {
   297→      toast.error("Image must be under 4MB");
   298→      return;
   299→    }
   300→    const fd = new FormData();
   301→    fd.append("file", file);
   302→    const r = await run(
   303→      () => api.upload<{ qrImage: string }>(`/api/admin/payment-methods/${editing.id}/qr-image`, fd),
   304→      { success: "QR image uploaded", error: "Upload failed" }
   305→    );
   306→    if (r) {
   307→      setQrImage(r.qrImage);
   308→      qc.invalidateQueries({ queryKey: ["admin-payment-methods"] });
   309→    }
   310→    // Reset file input so the same file can be re-uploaded if needed.
   311→    if (qrFileInputRef.current) qrFileInputRef.current.value = "";
   312→  }
