     1→// ============================================================================
     2→// File: prisma/schema.prisma
     3→// Purpose: Central database schema for PMS — Pradeep Medical Store.
     4→//          Online pharmacy platform with customer storefront + admin panel.
     5→// Database: PostgreSQL 15+ — hosted on Supabase (managed).
     6→// ORM:      Prisma 6.x
     7→//
     8→// Design principles:
     9→//   1. Normalized — no duplicate data storage (product images live ONLY in
    10→//      ProductImage, not duplicated across Product columns).
    11→//   2. Indexed — every foreign key and frequently-filtered column has an index.
    12→//   3. Typed — monetary values use Decimal(10,2) for exact precision; long
    13→//      text fields use @db.Text to leverage PostgreSQL's native text type.
    14→//   4. Cascading — child records (OrderItem, CartItem, etc.) are deleted when
    15→//      their parent is deleted, preventing orphan rows.
    16→//
    17→// Connection notes (Supabase):
    18→//   - DATABASE_URL uses the Supavisor transaction-mode pooler (port 6543) with
    19→//     ?pgbouncer=true for efficient connection reuse in serverless (Vercel).
    20→//   - DIRECT_URL uses the session-mode pooler (port 5432) for migrations.
    21→//   - FK constraints are enforced at the PostgreSQL level (relationMode left at
    22→//     default) — Supabase Supavisor supports prepared statements in transaction
    23→//     mode, so no ORM-level emulation is required.
    24→// ============================================================================

   197→/// Products (medicines, OTC, wellness, devices). Central catalog entity.
   198→model Product {
   199→  id                   String  @id @default(cuid())
   200→  name                 String  @db.VarChar(255)
   201→  slug                 String  @unique @db.VarChar(255)
   202→  sku                  String? @db.VarChar(100)

   209→  prescriptionRequired Boolean @default(false)
   210→  isGeneric            Boolean @default(false)
   211→
   212→  brandId    String?
   213→  categoryId String?
   214→  unit       String? @db.VarChar(50) // strip, bottle, tube, etc.
   215→  packSize   String? @db.VarChar(100)
   216→
   217→  // Pricing — margin-protected discount model. Decimal for exact precision.
   218→  mrp             Decimal  @db.Decimal(10, 2)
   219→  sellingPrice    Decimal  @db.Decimal(10, 2)
   220→  baseDiscountPct Decimal  @default(0) @db.Decimal(5, 2)

   222→  costPrice       Decimal? @db.Decimal(10, 2)

   224→
   225→  // Inventory
   226→  stock             Int @default(0)
   227→  lowStockThreshold Int @default(10)
   228→
   229→  // Denormalized primary image cache (mirrors ProductImage.isPrimary=true).
   230→  // Used by product cards/lists to avoid a JOIN on every catalog query.
   231→  primaryImage  String? @db.VarChar(500)
   232→  galleryImages String? @db.Text // JSON array of image URLs (cache)
   233→
   234→  // Display
   235→  displayOrder Int     @default(0)
   236→  isFeatured   Boolean @default(false)
   237→  isBestSeller Boolean @default(false)
   238→  isTrending   Boolean @default(false)

   240→  visibility   String  @default("public") @db.VarChar(20)
   241→
   242→  // Rating snapshot (updated by trigger/seed when reviews are approved)
   243→  avgRating   Decimal @default(0) @db.Decimal(3, 2)

   245→
   246→  createdAt DateTime @default(now())
   247→  updatedAt DateTime @updatedAt
   248→
   249→  brand       Brand?              @relation(fields: [brandId], references: [id], onDelete: SetNull)

   251→  orderItems  OrderItem[]
   252→  cartItems   CartItem[]
   253→  reviews     Review[]
   254→  wishlist    WishlistItem[]
   255→  deals       Deal[]
   256→  images      ProductImage[]
   257→  stockAlerts StockSubscription[]
   258→  refillReminders RefillReminder[]
   259→
   260→  @@index([slug])
   261→  @@index([brandId])
   262→  @@index([categoryId])
   263→  @@index([status])
   264→  @@index([isFeatured])
   265→  @@index([isBestSeller])
   266→  // ── Phase 97 (templates-perf): indexes for the home-feed & catalog queries.
   267→  //   Without these, every home page load (6 parallel queries) scans the full
   268→  //   Product table — fine at 325 rows today, but it scales O(n) as the
   269→  //   catalog grows. These composite + single-column indexes cover the most
   270→  //   common filters and sort orders used by /api/catalog/home-feed and
   271→  //   /api/catalog/featured.
   272→  @@index([isTrending])
   273→  @@index([visibility, status])
   274→  @@index([avgRating])
   275→  @@index([baseDiscountPct])
   276→  @@index([createdAt])
   277→  @@index([displayOrder])
   278→}
