   454→   454→// — Revenue trend badge — up/down/neutral % vs previous period —
   455→   455→function RevenueTrendBadge({
   456→   456→  series,
   457→   457→}: {
   458→   458→  series: Array<{ date: string; revenue: number; orders: number }>;
   459→   459→}) {
   460→   460→  if (!series || series.length < 2) {
   461→   461→    return (
   462→   462→      <Badge variant="outline" className="gap-1 text-xs">
   463→   463→        <Minus className="size-3 text-muted-foreground" />
   464→   464→        No trend
   465→   465→      </Badge>
   466→   466→    );
   467→   467→  }
   468→   468→
   469→   469→  // Compare last 3 days vs previous 3 days (more stable than 1 vs 1)
   470→   470→  const recent = series.slice(-3).reduce((s, d) => s + d.revenue, 0);
   471→   471→  const previous = series.slice(-6, -3).reduce((s, d) => s + d.revenue, 0);
   472→   472→
   473→   473→  if (previous === 0) {
   474→   474→    return (
   475→   475→      <Badge variant="outline" className="gap-1 text-xs bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900/50">
   476→   476→        <Activity className="size-3" />
   477→   477→        Active
   478→   478→      </Badge>
   479→   479→    );
   480→   480→  }
   481→   481→
   482→   482→  const pct = ((recent - previous) / previous) * 100;
   483→   483→  const isUp = pct >= 0;
   484→   484→  const isFlat = Math.abs(pct) < 1;
   485→   485→
   486→   486→  return (
   487→   487→    <Badge
   488→   488→      variant="outline"
   489→   489→      className={cn(
   490→   490→        "gap-1 text-xs font-medium",
   491→   491→        isFlat
   492→   492→          ? "bg-muted text-muted-foreground border-border"
   493→   493→          : isUp
   494→   494→            ? "bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900/50"
   495→   495→            : "bg-rose-50 text-rose-700 border-rose-200 dark:bg-rose-950/40 dark:text-rose-300 dark:border-rose-900/50"
   496→   496→      )}
   497→   497→    >
   498→   498→      {isFlat ? (
   499→   499→        <Minus className="size-3" />
   500→   500→      ) : isUp ? (
   501→   501→        <TrendingUp className="size-3" />
   502→   502→      ) : (
   503→   503→        <TrendingDown className="size-3" />
   504→   504→      )}
   505→   505→      {isFlat ? "Flat" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   506→   506→    </Badge>
   507→   507→  );
   508→   508→}

   510→   510→// — Revenue sparkline — pure SVG, no chart library —
   511→   511→function RevenueSparkline({
   512→   512→  series,
   513→   513→}: {
   514→   514→  series: Array<{ date: string; revenue: number; orders: number }>;
   515→   515→}) {
   516→   516→  if (!series || series.length === 0) {
   517→   517→    return (
   518→   518→      <div className="flex h-[80px] items-center justify-center text-xs text-muted-foreground">
   519→   519→        No revenue data yet
   520→   520→      </div>
   521→   521→    );
   522→   522→  }
   523→   523→
   524→   524→  const W = 600;
   525→   525→  const H = 80;
   526→   526→  const PAD = 4;
   527→   527→  const max = Math.max(...series.map((d) => d.revenue), 1);
   528→   528→  const min = Math.min(...series.map((d) => d.revenue), 0);
   529→   529→  const range = max - min || 1;
   530→   530→
   531→   531→  const points = series.map((d, i) => {
   532→   532→    const x = PAD + (i / (series.length - 1)) * (W - PAD * 2);
   533→   533→    const y = H - PAD - ((d.revenue - min) / range) * (H - PAD * 2);
   534→   534→    return [x, y] as const;
   535→   535→  });
   536→   536→
   537→   537→  // Smooth path using simple line segments (clean + minimal)
   538→   538→  const linePath = points
   539→   539→    .map(([x, y], i) => (i === 0 ? `M ${x} ${y}` : `L ${x} ${y}`))
   540→   540→    .join(" ");
   541→   541→  const areaPath = `${linePath} L ${points[points.length - 1][0]} ${H - PAD} L ${points[0][0]} ${H - PAD} Z`;
   542→   542→
   543→   543→  return (
   544→   544→    <div className="w-full">
   545→   545→      <svg
   546→   546→        viewBox={`0 0 ${W} ${H}`}
   547→   547→        preserveAspectRatio="none"
   548→   548→        className="w-full h-[80px]"
   549→   549→      >
   550→   550→        <defs>
   551→   551→          <linearGradient id="revSparkGrad" x1="0" y1="0" x2="0" y2="1">
   552→   552→            <stop offset="0%" stopColor="#10b981" stopOpacity={0.25} />
   553→   553→            <stop offset="100%" stopColor="#10b981" stopOpacity={0} />
   554→   554→          </linearGradient>
   555→   555→        </defs>
   556→   556→        <path d={areaPath} fill="url(#revSparkGrad)" />
   557→   557→        <path
   558→   558→          d={linePath}
   559→   559→          fill="none"
   560→   560→          stroke="#10b981"
   561→   561→          strokeWidth={2}
   562→   562→          strokeLinejoin="round"
   563→   563→          strokeLinecap="round"
   564→   564→          vectorEffect="non-scaling-stroke"
   565→   565→        />
   566→   566→        {points.map(([x, y], i) => (
   567→   567→          <circle
   568→   568→            key={i}
   569→   569→            cx={x}
   570→   570→            cy={y}
   571→   571→            r={2.5}
   572→   572→            fill="#10b981"
   573→   573→            vectorEffect="non-scaling-stroke"
   574→   574→          />
   575→   575→        ))}
   576→   576→      </svg>
   577→   577→      <div className="flex justify-between mt-1.5 text-[10px] text-muted-foreground">
   578→   578→        {series.map((d, i) => (
   579→   579→          <span key={i} className="tabular-nums">
   580→   580→            {d.date}
   581→   581→          </span>
   582→   582→        ))}
   583→   583→      </div>
   584→   584→    </div>
   585→   585→  );
   586→   586→}
