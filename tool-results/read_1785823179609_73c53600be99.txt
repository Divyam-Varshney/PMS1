     1→// ============================================================================
     2→// File: src/lib/pdf.ts
     3→// Purpose: Premium PDF invoice generation using jsPDF + autoTable.
     4→//          Produces a clean, professional A4 pharmacy invoice with:
     5→//            • Header band — store logo + name (left) and INVOICE box (right)
     6→//            • Store contact block — address / phone / email / license / GST
     7→//            • Bill To + Ship To two-column block
     8→//            • Order meta — order number, date, payment method, payment status
     9→//            • Items table — # | Item Name | Qty | MRP | Disc% | Amount
    10→//            • Totals — Item Total (MRP), Product Discount, After Discount,
    11→//              Voucher Discount, Loyalty Discount, Delivery, Grand Total
    12→//            • Footer — thank you note, return policy, store contact
    13→// Role: Called from the customer "Download Invoice" button and the admin
    14→//       order detail page. Returns a Uint8Array the browser can download.
    15→// ============================================================================
    16→
    17→import { getAllSettings } from "@/lib/settings";
    18→
    19→// Lazy-load jspdf + autoTable ONLY when a PDF is actually generated.
    20→// These libraries are ~29MB combined — loading them eagerly on every server
    21→// startup wastes RAM and slows cold starts. Dynamic import means they're only
    22→// loaded when /api/invoice or /api/admin/orders/[id]/invoice is called.
    23→let _jsPDF: typeof import("jspdf").jsPDF | null = null;
    24→let _autoTable: typeof import("jspdf-autotable").default | null = null;
    25→
    26→async function loadPdfLibs() {
    27→  if (!_jsPDF) {
    28→    const [{ jsPDF }, autoTableMod] = await Promise.all([
    29→      import("jspdf"),
    30→      import("jspdf-autotable"),
    31→    ]);
    32→    _jsPDF = jsPDF;
    33→    _autoTable = autoTableMod.default;
    34→  }
    35→  return { jsPDF: _jsPDF!, autoTable: _autoTable! };
    36→}
    37→
    38→export interface InvoiceLine {
    39→  name: string;
    40→  sku?: string | null;
    41→  qty: number;
    42→  mrp: number;
    43→  unitPrice: number;
    44→  appliedDiscountPct: number;
    45→  lineTotal: number;
    46→}
    47→
    48→export interface InvoiceData {
    49→  orderNumber: string;
    50→  invoiceNumber: string;
    51→  orderDate: string;
    52→  status: string;
    53→  paymentMethod: string;
    54→  paymentStatus: string;
    55→  paymentTxnId?: string;
    56→  paymentGateway?: string;
    57→  customerName: string;
    58→  customerPhone: string;
    59→  customerEmail: string;
    60→  shipLine1: string;
    61→  shipLine2?: string;
    62→  shipCity: string;
    63→  shipState: string;
    64→  shipPincode: string;
    65→  shipDistrict?: string;
    66→  shipLocality?: string;
    67→  lines: InvoiceLine[];
    68→  itemsTotal: number;
    69→  productDiscount: number;
    70→  voucherDiscount: number;
    71→  voucherCode?: string;
    72→  loyaltyPointsRedeemed?: number;
    73→  loyaltyDiscount?: number;
    74→  deliveryCharge: number;
    75→  taxTotal: number;
    76→  grandTotal: number;
    77→  source?: string;
    78→  prescriptionId?: string | null;
    79→  notes?: string | null;
    80→}
    81→
    82→// Professional palette — emerald primary with neutral dark/gray.
    83→const EMERALD: [number, number, number] = [5, 150, 105];
    84→const EMERALD_DARK: [number, number, number] = [4, 120, 87];
    85→const DARK: [number, number, number] = [17, 24, 39];
    86→const GRAY: [number, number, number] = [107, 114, 128];
    87→const LIGHT_GRAY: [number, number, number] = [229, 231, 235];
    88→const SOFT_BG: [number, number, number] = [240, 253, 244];
    89→const AMBER: [number, number, number] = [245, 158, 11];
    90→
    91→export async function generateInvoicePdf(data: InvoiceData): Promise<Uint8Array> {
    92→  const settings = await getAllSettings();
    93→  const storeName = settings["store.name"] ?? "Pradeep Medical Store";
    94→  const storeTagline = settings["store.tagline"] ?? "";
    95→  const storeEmail = settings["store.email"] ?? "";
    96→  const storePhone = settings["store.phone"] ?? "";
    97→  const storeAddress = settings["store.address"] ?? "";
    98→  const gstNumber = settings["store.gstNumber"] ?? "";
    99→  const license = settings["store.licenseNumber"] ?? "";
   100→  const footerNote = settings["invoice.footerNote"] ?? "";
   101→  const showGst = settings["invoice.showGst"] ?? true;
   102→  // Prefer the dedicated invoice logo (uploaded via Admin → Branding) and fall
   103→  // back to the main store logo, then to the email-logo setting, then to none.
   104→  const storeLogo =
   105→    (settings["store.invoiceLogo"] as string | undefined) ||
   106→    (settings["store.logo"] as string | undefined) ||
   107→    (settings["store.emailLogo"] as string | undefined) ||
   108→    "";
   109→
   110→  const { jsPDF, autoTable } = await loadPdfLibs();
   111→  const doc = new jsPDF({ unit: "pt", format: "a4" });
   112→  const pageWidth = doc.internal.pageSize.getWidth();
   113→  const pageHeight = doc.internal.pageSize.getHeight();
   114→  const margin = 36;
   115→  const contentWidth = pageWidth - margin * 2;
   116→  let y = 0;
   117→
   118→  // =====================================================================
   119→  // 1. TOP ACCENT BAND — three thin stripes (dark → emerald → soft mint)
   120→  // =====================================================================
   121→  doc.setFillColor(...EMERALD_DARK);
   122→  doc.rect(0, 0, pageWidth, 5, "F");
   123→  doc.setFillColor(...EMERALD);
   124→  doc.rect(0, 5, pageWidth, 4, "F");
   125→  doc.setFillColor(...SOFT_BG);
   126→  doc.rect(0, 9, pageWidth, 2, "F");
   127→
   128→  y = 36;
   129→
   130→  // =====================================================================
   131→  // 2. HEADER — Logo + store name/tagline (left) | INVOICE card (right)
   132→  // =====================================================================
   133→  const logoSize = 48;
   134→  let textLeftX = margin;
   135→  if (storeLogo) {
   136→    try {
   137→      // White rounded background behind the logo for contrast against any
   138→      // logo color (especially when the logo is dark-on-transparent).
   139→      doc.setFillColor(255, 255, 255);
   140→      doc.setDrawColor(...LIGHT_GRAY);
   141→      doc.setLineWidth(0.4);
   142→      doc.roundedRect(margin - 2, y - 8, logoSize + 4, logoSize + 4, 4, 4, "FD");
   143→      doc.addImage(storeLogo, "PNG", margin, y - 6, logoSize, logoSize);
   144→      textLeftX = margin + logoSize + 14;
   145→    } catch (e) {
   146→      console.error("[pdf] logo error:", e);
   147→    }
   148→  }
   149→
   150→  // Store name + tagline
   151→  doc.setFont("helvetica", "bold");
   152→  doc.setFontSize(20);
   153→  doc.setTextColor(...EMERALD);
   154→  doc.text(storeName, textLeftX, y + 4);
   155→
   156→  if (storeTagline) {
   157→    doc.setFont("helvetica", "normal");
   158→    doc.setFontSize(8.5);
   159→    doc.setTextColor(...GRAY);
   160→    doc.text(storeTagline, textLeftX, y + 18);
   161→  }
   162→
   163→  // Store contact (under the name, on two lines if needed)
   164→  doc.setFont("helvetica", "normal");
   165→  doc.setFontSize(7.5);
   166→  doc.setTextColor(107, 114, 128);
   167→  if (storeAddress) {
   168→    const addrLines = doc.splitTextToSize(storeAddress, 220);
   169→    doc.text(addrLines.slice(0, 2), textLeftX, y + 30);
   170→  }
   171→  if (storePhone || storeEmail) {
   172→    doc.text(
   173→      `${storePhone ? `Ph: ${storePhone}` : ""}${storePhone && storeEmail ? "   ·   " : ""}${storeEmail ? storeEmail : ""}`,
   174→      textLeftX,
   175→      y + 44
   176→    );
   177→  }
   178→
   179→  // INVOICE card (right side) — emerald gradient panel
   180→  const cardW = 180;
   181→  const cardH = 84;
   182→  const cardX = pageWidth - margin - cardW;
   183→  const cardY = y - 12;
   184→  doc.setFillColor(...EMERALD_DARK);
   185→  doc.roundedRect(cardX, cardY, cardW, cardH, 8, 8, "F");
   186→  // Subtle emerald highlight band at the top of the card
   187→  doc.setFillColor(...EMERALD);
   188→  doc.roundedRect(cardX, cardY, cardW, 22, 8, 8, "F");
   189→  doc.setFillColor(...EMERALD);
   190→  doc.rect(cardX, cardY + 12, cardW, 10, "F");
   191→
   192→  doc.setTextColor(255, 255, 255);
   193→  doc.setFont("helvetica", "bold");
   194→  doc.setFontSize(13);
   195→  doc.text("TAX INVOICE", cardX + cardW / 2, cardY + 15, { align: "center" });
   196→
   197→  doc.setFont("helvetica", "normal");
   198→  doc.setFontSize(8.5);
   199→  doc.setTextColor(209, 250, 229);
   200→  doc.text("Invoice No.", cardX + 14, cardY + 38);
   201→  doc.text("Order No.", cardX + 14, cardY + 52);
   202→  doc.text("Date (IST)", cardX + 14, cardY + 66);
   203→
   204→  doc.setFont("helvetica", "bold");
   205→  doc.setTextColor(255, 255, 255);
   206→  doc.setFontSize(9);
   207→  doc.text(data.invoiceNumber, cardX + cardW - 14, cardY + 38, { align: "right" });
   208→  doc.text(data.orderNumber, cardX + cardW - 14, cardY + 52, { align: "right" });
   209→  // Split orderDate onto two lines if it includes time (it now contains " · " separator)
   210→  const dateParts = data.orderDate.split(" · ");
   211→  if (dateParts.length === 2) {
   212→    doc.setFontSize(8);
   213→    doc.text(dateParts[0], cardX + cardW - 14, cardY + 64, { align: "right" });
   214→    doc.setFontSize(7);
   215→    doc.setTextColor(209, 250, 229);
   216→    doc.text(dateParts[1], cardX + cardW - 14, cardY + 74, { align: "right" });
   217→  } else {
   218→    doc.text(data.orderDate, cardX + cardW - 14, cardY + 66, { align: "right" });
   219→  }
   220→
   221→  y = cardY + cardH + 18;
   222→
   223→  // Light divider
   224→  doc.setDrawColor(...LIGHT_GRAY);
   225→  doc.setLineWidth(0.6);
   226→  doc.line(margin, y - 6, pageWidth - margin, y - 6);
   227→
   228→  // =====================================================================
   229→  // 3. BILL TO + SHIP TO + PAYMENT INFO (three columns)
   230→  // =====================================================================
   231→  const colW = contentWidth / 3;
   232→  const billX = margin;
   233→  const shipX = margin + colW + 6;
   234→  const payX = margin + colW * 2 + 12;
   235→
   236→  // -- BILL TO --
   237→  doc.setFont("helvetica", "bold");
   238→  doc.setFontSize(7.5);
   239→  doc.setTextColor(...EMERALD_DARK);
   240→  doc.text("BILL TO", billX, y);
   241→
   242→  doc.setFont("helvetica", "bold");
   243→  doc.setFontSize(10);
   244→  doc.setTextColor(...DARK);
   245→  doc.text(data.customerName || "Customer", billX, y + 14);
   246→
   247→  doc.setFont("helvetica", "normal");
   248→  doc.setFontSize(8.5);
   249→  doc.setTextColor(...GRAY);
   250→  let billY = y + 26;
   251→  if (data.customerPhone) {
   252→    doc.text(`Phone: ${data.customerPhone}`, billX, billY);
   253→    billY += 11;
   254→  }
   255→  if (data.customerEmail) {
   256→    const emailLines = doc.splitTextToSize(data.customerEmail, colW - 8);
   257→    doc.text(emailLines.slice(0, 2), billX, billY);
   258→    billY += emailLines.slice(0, 2).length * 11;
   259→  }
   260→
   261→  // -- SHIP TO --
   262→  doc.setFont("helvetica", "bold");
   263→  doc.setFontSize(7.5);
   264→  doc.setTextColor(...EMERALD_DARK);
   265→  doc.text("SHIP TO", shipX, y);
   266→
   267→  doc.setFont("helvetica", "normal");
   268→  doc.setFontSize(8.5);
   269→  doc.setTextColor(...DARK);
   270→  let shipY = y + 13;
   271→  if (data.shipLine1) {
   272→    const shipLines1 = doc.splitTextToSize(data.shipLine1, colW - 8);
   273→    doc.text(shipLines1, shipX, shipY);
   274→    shipY += shipLines1.length * 11;
   275→  }
   276→  if (data.shipLine2) {
   277→    const shipLines2 = doc.splitTextToSize(data.shipLine2, colW - 8);
   278→    doc.text(shipLines2, shipX, shipY);
   279→    shipY += shipLines2.length * 11;
   280→  }
   281→  const localityPart = data.shipLocality ? `${data.shipLocality}, ` : "";
   282→  const cityLine = `${localityPart}${data.shipCity}${data.shipDistrict ? `, ${data.shipDistrict}` : ""}`;
   283→  if (cityLine.trim()) {
   284→    const cityLines = doc.splitTextToSize(cityLine, colW - 8);
   285→    doc.text(cityLines, shipX, shipY);
   286→    shipY += cityLines.length * 11;
   287→  }
   288→  doc.text(`${data.shipState} - ${data.shipPincode}`, shipX, shipY);
   289→
   290→  // -- PAYMENT INFO --
   291→  doc.setFont("helvetica", "bold");
   292→  doc.setFontSize(7.5);
   293→  doc.setTextColor(...EMERALD_DARK);
   294→  doc.text("PAYMENT INFO", payX, y);
   295→
   296→  doc.setFont("helvetica", "normal");
   297→  doc.setFontSize(8.5);
   298→  doc.setTextColor(...GRAY);
   299→  doc.text("Method:", payX, y + 14);
   300→  doc.setTextColor(...DARK);
   301→  doc.setFont("helvetica", "bold");
   302→  const payMethodText = data.paymentMethod.toUpperCase();
   303→  doc.text(doc.splitTextToSize(payMethodText, colW - 50), payX + 42, y + 14);
   304→
   305→  doc.setFont("helvetica", "normal");
   306→  doc.setFontSize(8.5);
   307→  doc.setTextColor(...GRAY);
   308→  doc.text("Status:", payX, y + 28);
   309→  // Color-coded payment status badge
   310→  const payStatusUpper = data.paymentStatus.toUpperCase();
   311→  const statusColor: [number, number, number] =
   312→    data.paymentStatus === "paid"
   313→      ? EMERALD
   314→      : data.paymentStatus === "failed"
   315→        ? [239, 68, 68]
   316→        : data.paymentStatus === "refunded"
   317→          ? [156, 163, 175]
   318→          : AMBER;
   319→  doc.setFillColor(...statusColor);
   320→  doc.roundedRect(payX + 42, y + 20, doc.getTextWidth(payStatusUpper) + 14, 12, 3, 3, "F");
   321→  doc.setTextColor(255, 255, 255);
   322→  doc.setFont("helvetica", "bold");
   323→  doc.setFontSize(7.5);
   324→  doc.text(payStatusUpper, payX + 42 + 7, y + 28);
   325→
   326→  let payInfoY = y + 42;
   327→  if (data.paymentTxnId) {
   328→    doc.setFont("helvetica", "normal");
   329→    doc.setFontSize(8);
   330→    doc.setTextColor(...GRAY);
   331→    doc.text("Txn ID:", payX, payInfoY);
   332→    doc.setTextColor(...DARK);
   333→    doc.setFont("courier", "bold");
   334→    doc.setFontSize(7.5);
   335→    const txnLines = doc.splitTextToSize(data.paymentTxnId, colW - 50);
   336→    doc.text(txnLines.slice(0, 2), payX + 42, payInfoY);
   337→    payInfoY += txnLines.slice(0, 2).length * 10 + 2;
   338→    doc.setFont("helvetica", "normal");
   339→  }
   340→  if (data.paymentGateway) {
   341→    doc.setFont("helvetica", "normal");
   342→    doc.setFontSize(8);
   343→    doc.setTextColor(...GRAY);
   344→    doc.text("Gateway:", payX, payInfoY);
   345→    doc.setTextColor(...DARK);
   346→    doc.text(data.paymentGateway.toUpperCase(), payX + 42, payInfoY);
   347→  }
   348→
   349→  y = Math.max(billY, shipY, payInfoY) + 18;
   350→
   351→  // =====================================================================
   352→  // 4. ORDER META PANEL — single-row soft panel with key info
   353→  // =====================================================================
   354→  const panelH = 26;
   355→  doc.setFillColor(...SOFT_BG);
   356→  doc.roundedRect(margin, y, contentWidth, panelH, 4, 4, "F");
   357→  // Left accent stripe
   358→  doc.setFillColor(...EMERALD);
   359→  doc.roundedRect(margin, y, 3, panelH, 1.5, 1.5, "F");
   360→
   361→  const metaItems: Array<{ label: string; value: string }> = [
   362→    { label: "ORDER DATE", value: data.orderDate },
   363→    { label: "ORDER STATUS", value: data.status.toUpperCase() },
   364→    { label: "SOURCE", value: (data.source || "cart").toUpperCase() },
   365→  ];
   366→  if (data.prescriptionId) {
   367→    metaItems.push({ label: "PRESCRIPTION", value: "VERIFIED" });
   368→  }
   369→  const metaColW = contentWidth / metaItems.length;
   370→  metaItems.forEach((m, i) => {
   371→    const cx = margin + metaColW * i + metaColW / 2;
   372→    doc.setFont("helvetica", "normal");
   373→    doc.setFontSize(6.5);
   374→    doc.setTextColor(...GRAY);
   375→    doc.text(m.label, cx, y + 9, { align: "center" });
   376→    doc.setFont("helvetica", "bold");
   377→    doc.setFontSize(8.5);
   378→    doc.setTextColor(...DARK);
   379→    // Truncate value if too long for the column
   380→    let val = m.value;
   381→    const maxW = metaColW - 12;
   382→    while (doc.getTextWidth(val) > maxW && val.length > 4) {
   383→      val = val.slice(0, -2);
   384→    }
   385→    if (val !== m.value) val = val.slice(0, -1) + "…";
   386→    doc.text(val, cx, y + 19, { align: "center" });
   387→  });
   388→
   389→  y += panelH + 14;
   390→
   391→  // =====================================================================
   392→  // 5. ITEMS TABLE — # | Item Name | HSN | Qty | MRP | Disc% | Rate | Amount
   393→  // =====================================================================
   394→  autoTable(doc, {
   395→    startY: y,
   396→    head: [["#", "Item Name", "Qty", "MRP", "Disc%", "Rate", "Amount"]],
   397→    body: data.lines.map((l, i) => {
   398→      const rate = l.qty > 0 ? Number(l.lineTotal) / l.qty : Number(l.lineTotal);
   399→      return [
   400→        String(i + 1),
   401→        l.name + (l.sku ? `\n${l.sku}` : ""),
   402→        String(l.qty),
   403→        `Rs. ${Number(l.mrp).toFixed(2)}`,
   404→        l.appliedDiscountPct > 0 ? `${Number(l.appliedDiscountPct).toFixed(1)}%` : "-",
   405→        `Rs. ${rate.toFixed(2)}`,
   406→        `Rs. ${Number(l.lineTotal).toFixed(2)}`,
   407→      ];
   408→    }),
   409→    theme: "striped",
   410→    headStyles: {
   411→      fillColor: EMERALD,
   412→      textColor: 255,
   413→      fontSize: 9,
   414→      fontStyle: "bold",
   415→      lineColor: EMERALD_DARK,
   416→      lineWidth: 0.3,
   417→      halign: "left",
   418→    },
   419→    bodyStyles: {
   420→      fontSize: 9,
   421→      textColor: DARK,
   422→      lineColor: LIGHT_GRAY,
   423→      lineWidth: 0.2,
   424→      cellPadding: { top: 6, bottom: 6, left: 6, right: 6 },
   425→    },
   426→    alternateRowStyles: { fillColor: SOFT_BG },
   427→    columnStyles: {
   428→      0: { halign: "center", cellWidth: 26, fontStyle: "bold" },
   429→      1: { halign: "left", cellWidth: "auto" },
   430→      2: { halign: "center", cellWidth: 32 },
   431→      3: { halign: "right", cellWidth: 60 },
   432→      4: { halign: "center", cellWidth: 42 },
   433→      5: { halign: "right", cellWidth: 60 },
   434→      6: { halign: "right", cellWidth: 70, fontStyle: "bold" },
   435→    },
   436→    margin: { left: margin, right: margin },
   437→  });
   438→
   439→  // @ts-ignore — lastAutoTable is added by the plugin
   440→  y = (doc as any).lastAutoTable.finalY + 16;
   441→
   442→  // =====================================================================
   443→  // 6. TOTALS — right-aligned totals block + amount-in-words (left)
   444→  // =====================================================================
   445→  const totalsW = 250;
   446→  const tx = pageWidth - margin - totalsW;
   447→  const totalsLeftPad = 10;
   448→
   449→  const afterDiscount = Math.max(0, Number(data.itemsTotal) - Number(data.productDiscount));
   450→  const totalDiscount = Number(data.productDiscount) + Number(data.loyaltyDiscount ?? 0) + Number(data.voucherDiscount);
   451→
   452→  // Soft panel behind totals
   453→  const totalsStartY = y - 4;
   454→  const line = (
   455→    label: string,
   456→    value: string,
   457→    opts: { bold?: boolean; color?: [number, number, number]; size?: number } = {}
   458→  ) => {
   459→    doc.setFont("helvetica", opts.bold ? "bold" : "normal");
   460→    doc.setFontSize(opts.size ?? (opts.bold ? 10 : 9));
   461→    doc.setTextColor(...(opts.color ?? (opts.bold ? DARK : GRAY)));
   462→    doc.text(label, tx + totalsLeftPad, y);
   463→    doc.text(value, pageWidth - margin - totalsLeftPad, y, { align: "right" });
   464→    y += opts.bold ? 18 : 14;
   465→  };
   466→
   467→  line("Item Total (MRP)", `Rs. ${Number(data.itemsTotal).toFixed(2)}`);
   468→  if (data.productDiscount > 0) {
   469→    line("Product Discount", `- Rs. ${Number(data.productDiscount).toFixed(2)}`, { color: EMERALD });
   470→  }
   471→  line("After Discount", `Rs. ${Number(afterDiscount).toFixed(2)}`, { bold: true });
   472→  if (data.voucherDiscount > 0) {
   473→    line(
   474→      `Voucher Discount${data.voucherCode ? ` (${data.voucherCode})` : ""}`,
   475→      `- Rs. ${Number(data.voucherDiscount).toFixed(2)}`,
   476→      { color: EMERALD }
   477→    );
   478→  }
   479→  if (data.loyaltyDiscount && data.loyaltyDiscount > 0) {
   480→    line(
   481→      `Loyalty Discount${data.loyaltyPointsRedeemed ? ` (${data.loyaltyPointsRedeemed} pts)` : ""}`,
   482→      `- Rs. ${Number(data.loyaltyDiscount).toFixed(2)}`,
   483→      { color: AMBER }
   484→    );
   485→  }
   486→  if (totalDiscount > 0) {
   487→    line("Total Discount", `- Rs. ${totalDiscount.toFixed(2)}`, { bold: true, color: EMERALD });
   488→  }
   489→  line(
   490→    "Delivery Charges",
   491→    Number(data.deliveryCharge) === 0 ? "FREE" : `Rs. ${Number(data.deliveryCharge).toFixed(2)}`,
   492→    { color: Number(data.deliveryCharge) === 0 ? EMERALD : undefined }
   493→  );

   649→// ---------------------------------------------------------------------------
   650→// Helper — convert a number to Indian English words for the "Amount in
   651→// words" line on the invoice. Handles up to crores.
   652→// ---------------------------------------------------------------------------
   653→function numberToIndianWords(num: number): string {
   654→  if (!isFinite(num)) return "Zero";
   655→  const integerPart = Math.floor(num);
   656→  const paisa = Math.round((num - integerPart) * 100);
   657→  if (integerPart === 0 && paisa === 0) return "Zero Rupees";
   658→
   659→  const ones = ["", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine", "Ten",
   660→    "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen", "Seventeen", "Eighteen", "Nineteen"];
   661→  const tens = ["", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"];
   662→
   663→  function twoDigits(n: number): string {
   664→    if (n < 20) return ones[n];
   665→    return tens[Math.floor(n / 10)] + (n % 10 ? " " + ones[n % 10] : "");
   666→  }
   667→  function threeDigits(n: number): string {
   668→    const h = Math.floor(n / 100);
   669→    const r = n % 100;
   670→    const parts: string[] = [];
   671→    if (h) parts.push(ones[h] + " Hundred");
   672→    if (r) parts.push(twoDigits(r));
   673→    return parts.join(" ");
   674→  }
   675→
   676→  function inWords(n: number): string {
   677→    if (n === 0) return "";
   678→    const crore = Math.floor(n / 10000000);
   679→    n %= 10000000;
   680→    const lakh = Math.floor(n / 100000);
   681→    n %= 100000;
   682→    const thousand = Math.floor(n / 1000);
   683→    n %= 1000;
   684→    const rest = n;
   685→    const parts: string[] = [];
   686→    if (crore) parts.push(inWords(crore) + " Crore");
   687→    if (lakh) parts.push(twoDigits(lakh) + " Lakh");
   688→    if (thousand) parts.push(twoDigits(thousand) + " Thousand");
   689→    if (rest) parts.push(threeDigits(rest));
   690→    return parts.join(" ").trim();
   691→  }
   692→
   693→  const rupeeWords = inWords(integerPart);
   694→  let result = rupeeWords ? `${rupeeWords} Rupees` : "Zero Rupees";
   695→  if (paisa > 0) {
   696→    result += ` and ${twoDigits(paisa)} Paise`;
   697→  }
   698→  return result;
   699→}
