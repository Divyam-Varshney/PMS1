     3→// Purpose: Unified customer activity history — merges Orders, Prescriptions,
     4→//          and Manual Requests into a single timeline so customers can see
     5→//          everything that has happened in their account in one place.
     6→//

     8→//
     9→//          Each card shows:
    10→//            - Type badge   (Order emerald, Prescription violet, Request amber)
    11→//            - Number       (order number, RX-XXXXXXXX, or MR-XXXXXXXX)

    13→//            - Status badge (colored by status)

    15→//            - Type-specific content + actions:
    16→//                Orders        → items preview + total + Track/Reorder/Invoice
    17→//                Prescriptions → image count + View Details
    18→//                Manual Reqs   → medicine preview + View Details
    19→//
    20→//          "View Details" for prescriptions / manual requests opens a dialog
    21→//          with images (lightbox), status timeline, admin remarks, and
    22→//          created/updated timestamps.
    23→//
    24→//          Auto-refresh: staleTime 10s + refetchInterval 30s so newly

   146→export function OrdersView() {
   147→  const { customer, isLoading: custLoading } = useRequireAuth();
   148→  const navigate = useUI((s) => s.navigate);
   149→  const setCartOpen = useUI((s) => s.setCartOpen);
   150→  const qc = useQueryClient();
   151→  const [filter, setFilter] = useState<FilterTab>("all");
   152→  // Secondary segmented control for orders — only visible when filter === "orders".
   153→  // Defaults to "active" so the customer lands on what they're waiting for.
   154→  const [orderSegment, setOrderSegment] = useState<OrderSegment>("active");
   155→  const [reorderingId, setReorderingId] = useState<string | null>(null);
   156→  // Detail dialog state — holds the prescription or manual request being viewed.
   157→  const [detailItem, setDetailItem] = useState<
   158→    | Extract<UnifiedHistoryItem, { type: "prescription" }>
   159→    | Extract<UnifiedHistoryItem, { type: "manual_request" }>
   160→    | null
   161→  >(null);
   162→  // Lightbox image URL — when set, shows a full-screen image viewer for a
   163→  // prescription image.
   164→  const [lightboxSrc, setLightboxSrc] = useState<string | null>(null);
   165→
   166→  // ---- Unified history query. Auto-refreshes on a 30s interval so newly
   167→  //      converted orders / status changes appear without a manual reload.
   168→  //      staleTime of 10s prevents redundant refetches within that window.
   169→  const { data: history, isLoading } = useQuery<UnifiedHistoryResponse>({
   170→    queryKey: qk.history,
   171→    queryFn: () => api<UnifiedHistoryResponse>("/api/customer/history"),
   172→    enabled: !!customer,
   173→    staleTime: 10 * 1000,
   174→    refetchInterval: 10 * 1000,
   175→  });
   176→
   177→  // ---- Reorder mutation — reuses the existing /api/orders/[id]/reorder
   178→  //      endpoint. Identical to the previous OrdersView behavior.
   179→  const reorderMutation = useMutation({
   180→    mutationFn: (orderId: string) =>
   181→      api.post<{ added: number; skipped: string[]; totalItems: number }>(
   182→        `/api/orders/${orderId}/reorder`
   183→      ),
   184→    onMutate: (orderId) => setReorderingId(orderId),
   185→    onSuccess: (data) => {
   186→      qc.invalidateQueries({ queryKey: qk.cart });
   187→      if (data.added > 0 && data.skipped.length === 0) {
   188→        toast.success(`Added ${data.added} item(s) to your cart`, {
   189→          action: { label: "View cart", onClick: () => setCartOpen(true) },
   190→        });
   191→      } else if (data.added > 0 && data.skipped.length > 0) {
   192→        toast.success(
   193→          `Added ${data.added} item(s). ${data.skipped.length} unavailable.`,
   194→          {
   195→            description: `Unavailable: ${data.skipped.slice(0, 2).join(", ")}${data.skipped.length > 2 ? "…" : ""}`,
   196→            action: { label: "View cart", onClick: () => setCartOpen(true) },
   197→          }
   198→        );
   199→      } else {
   200→        toast.error("None of the items are available to reorder");
   201→      }
   202→    },
   203→    onError: (e: Error) => toast.error(e.message),
   204→    onSettled: () => setReorderingId(null),
   205→  });
   206→
   207→  // ---- Filter by active tab + optional order-segment sub-filter.
   208→  //      When the type filter is "orders", a secondary segmented control
   209→  //      ("Active Orders" / "Past Orders") further splits the list by status.
   210→  const visibleItems = useMemo(() => {
   211→    if (!history?.items) return [];
   212→    let list = history.items;
   213→    if (filter === "orders") list = list.filter((i) => i.type === "order");
   214→    else if (filter === "prescriptions")
   215→      list = list.filter((i) => i.type === "prescription");
   216→    else if (filter === "requests")
   217→      list = list.filter((i) => i.type === "manual_request");
   218→
   219→    // Apply the Active / Past sub-segment only when the user is on the
   220→    // "orders" tab — keeps the segmented control's behavior predictable.
   221→    if (filter === "orders") {
   222→      list = list.filter((i) => {
   223→        if (i.type !== "order") return true;
   224→        const isActive = ACTIVE_ORDER_STATUSES.has(i.status);
   225→        return orderSegment === "active" ? isActive : !isActive;
   226→      });
   227→    }
   228→    return list;
   229→  }, [history, filter, orderSegment]);
   230→
   231→  // ---- Counts for the segmented control badges + header subtitle.
   232→  const orderCounts = useMemo(() => {
   233→    const orders = (history?.items ?? []).filter((i) => i.type === "order");
   234→    return {
   235→      active: orders.filter((o) => ACTIVE_ORDER_STATUSES.has(o.status)).length,
   236→      past: orders.filter((o) => !ACTIVE_ORDER_STATUSES.has(o.status)).length,
   237→    };
   238→  }, [history]);
   239→
   240→  // ---- Loading state — premium skeleton cards.
   241→  if (custLoading || isLoading) {
   242→    return <OrdersSkeleton />;
   243→  }
   244→
   245→  if (!customer) {
   246→    // useRequireAuth handles redirect in an effect
   247→    return null;
   248→  }
   249→
   250→  // ---- Empty state — if the customer has no activity at all.
   251→  if (!history || history.items.length === 0) {
   252→    return (
   253→      <div className="mx-auto max-w-3xl px-4 py-6">
   254→        <EmptyState
   255→          icon={Package}
   256→          title="No activity yet"
   257→          description="When you place an order, upload a prescription, or request a medicine, it will appear here."
   258→          action={
   259→            <Button onClick={() => navigate({ name: "shop" })} className="gap-2">
   260→              Start shopping <ArrowRight className="size-4" />
   261→            </Button>
   262→          }
   263→        />
   264→      </div>
   265→    );
   266→  }
   267→
   268→  // ---- Empty state — when on the Orders tab and the selected segment is empty
   269→  //      (e.g. no active orders but the customer has past orders, or vice versa).
   270→  if (filter === "orders" && visibleItems.length === 0) {
   271→    const isEmptyActive = orderSegment === "active";
   272→    return (
   273→      <div className="mx-auto max-w-3xl px-4 py-6 sm:px-6">
   274→        <OrdersHeader
   275→          total={history.items.length}
   276→          ordersCount={orderCounts.active + orderCounts.past}
   277→          prescriptionsCount={
   278→            history.items.filter((i) => i.type === "prescription").length
   279→          }
   280→          requestsCount={
   281→            history.items.filter((i) => i.type === "manual_request").length
   282→          }
   283→        />
   284→        <Tabs
   285→          value={filter}
   286→          onValueChange={(v) => setFilter(v as FilterTab)}
   287→          className="mb-3"
   288→        >
   289→          <TabsList className="w-full justify-start overflow-x-auto">
   290→            <TabsTrigger value="all">All</TabsTrigger>
   291→            <TabsTrigger value="orders">
   292→              Orders ({orderCounts.active + orderCounts.past})
   293→            </TabsTrigger>
   294→            <TabsTrigger value="prescriptions">Prescriptions</TabsTrigger>
   295→            <TabsTrigger value="requests">Requests</TabsTrigger>
   296→          </TabsList>
   297→        </Tabs>
   298→        <OrderSegmentedControl
   299→          value={orderSegment}
   300→          onChange={setOrderSegment}
   301→          activeCount={orderCounts.active}
   302→          pastCount={orderCounts.past}
   303→        />
   304→        <EmptyState
   305→          icon={isEmptyActive ? Truck : Package}
   306→          title={isEmptyActive ? "No active orders" : "No past orders"}
   307→          description={
   308→            isEmptyActive
   309→              ? "You have no orders in progress. Browse the catalog and place a new order to see it tracked here in real time."
   310→              : "You haven't completed any orders yet. Once an order is delivered or cancelled, it will appear in this list."
   311→          }
   312→          action={
   313→            isEmptyActive ? (
   314→              <Button onClick={() => navigate({ name: "shop" })} className="gap-2">
   315→                Start shopping <ArrowRight className="size-4" />
   316→              </Button>
   317→            ) : (
   318→              <Button
   319→                variant="outline"
   320→                onClick={() => setOrderSegment("active")}
   321→                className="gap-2"
   322→              >
   323→                View active orders <Truck className="size-4" />
   324→              </Button>
   325→            )
   326→          }
   327→          className="py-12"
   328→        />
   329→      </div>
   330→    );
   331→  }
   332→
   333→  const onDownloadInvoice = async (orderId: string, orderNumber: string) => {
   334→    try {
   335→      const buf = await api.raw(`/api/invoice/${orderId}`);
   336→      const blob = new Blob([buf], { type: "application/pdf" });
   337→      const url = URL.createObjectURL(blob);
   338→      const a = document.createElement("a");
   339→      a.href = url;
   340→      a.download = `invoice-${orderNumber}.pdf`;
   341→      document.body.appendChild(a);
   342→      a.click();
   343→      a.remove();
   344→      URL.revokeObjectURL(url);
   345→    } catch (e: unknown) {
   346→      const msg = e instanceof Error ? e.message : "Failed to download invoice";
   347→      toast.error(msg);
   348→    }
   349→  };
   350→
   351→  // ---- Count breakdown for the header subtitle.
   352→  const counts = {
   353→    orders: history.items.filter((i) => i.type === "order").length,
   354→    prescriptions: history.items.filter((i) => i.type === "prescription").length,
   355→    requests: history.items.filter((i) => i.type === "manual_request").length,
   356→  };
   357→
   358→  return (
   359→    <div className="mx-auto max-w-3xl px-4 py-6 sm:px-6">
   360→      <OrdersHeader
   361→        total={history.items.length}
   362→        ordersCount={counts.orders}
   363→        prescriptionsCount={counts.prescriptions}
   364→        requestsCount={counts.requests}
   365→      />
   366→
   367→      {/* Filter tabs */}
   368→      <Tabs
   369→        value={filter}
   370→        onValueChange={(v) => setFilter(v as FilterTab)}
   371→        className="mb-3"
   372→      >
   373→        <TabsList className="w-full justify-start overflow-x-auto">
   374→          <TabsTrigger value="all">All</TabsTrigger>
   375→          <TabsTrigger value="orders">
   376→            Orders ({counts.orders})
   377→          </TabsTrigger>
   378→          <TabsTrigger value="prescriptions">Prescriptions</TabsTrigger>
   379→          <TabsTrigger value="requests">Requests</TabsTrigger>
   380→        </TabsList>
   381→      </Tabs>
   382→
   383→      {/* Active / Past segmented control — only visible on the Orders tab. */}
   384→      {filter === "orders" && (
   385→        <OrderSegmentedControl
   386→          value={orderSegment}
   387→          onChange={setOrderSegment}
   388→          activeCount={orderCounts.active}
   389→          pastCount={orderCounts.past}
   390→          className="mb-4"
   391→        />
   392→      )}
   393→
   394→      {visibleItems.length === 0 ? (
   395→        <EmptyState
   396→          icon={Package}
   397→          title="Nothing in this view"
   398→          description="Try a different filter to see more activity."
   399→          className="py-10"
   400→        />
   401→      ) : (
   402→        <div className="space-y-3">
   403→          {visibleItems.map((item, idx) => (
   404→            <motion.div
   405→              key={`${item.type}-${item.id}`}
   406→              initial={{ opacity: 0, y: 12 }}
   407→              animate={{ opacity: 1, y: 0 }}
   408→              transition={{ duration: 0.25, delay: Math.min(idx * 0.04, 0.3) }}
   409→            >
   410→              <ActivityCard
   411→                item={item}
   412→                onViewDetails={(it) => setDetailItem(it)}
   413→                onReorder={(orderId) => reorderMutation.mutate(orderId)}
   414→                onDownloadInvoice={onDownloadInvoice}
   415→                onTrack={(orderId) =>
   416→                  navigate({ name: "track-order", orderId })
   417→                }
   418→                reorderingId={reorderingId}
   419→              />
   420→            </motion.div>
   421→          ))}
   422→        </div>
   423→      )}
   424→
   425→      <Button
   426→        variant="outline"
   427→        onClick={() => navigate({ name: "shop" })}
   428→        className="mt-6 w-full gap-2"
   429→      >
   430→        <ShoppingBag className="size-4" /> Continue shopping
   431→      </Button>
   432→
   433→      <div className="mt-4 flex items-center justify-center gap-1 text-[10px] text-muted-foreground">
   434→        <Receipt className="size-3" /> Tip: Use{" "}
   435→        <Badge variant="outline" className="px-1 py-0 text-[10px]">
   436→          Track
   437→        </Badge>{" "}
   438→        to see live order progress.
   439→      </div>
   440→
   441→      {/* ---- Detail dialog for prescriptions / manual requests ---- */}
   442→      <DetailDialog
   443→        item={detailItem}
   444→        onClose={() => setDetailItem(null)}
   445→        onOpenLightbox={setLightboxSrc}
   446→      />
   447→
   448→      {/* ---- Lightbox for prescription images ---- */}
   449→      <Lightbox src={lightboxSrc} onClose={() => setLightboxSrc(null)} />
   450→    </div>
   451→  );
   452→}

   454→// ---------------------------------------------------------------------------
   455→// Activity card — renders the appropriate layout for each activity type.
   456→// ---------------------------------------------------------------------------
   457→function ActivityCard({
   458→  item,
   459→  onViewDetails,
   460→  onReorder,
   461→  onDownloadInvoice,
   462→  onTrack,
   463→  reorderingId,
   464→}: {
   465→  item: UnifiedHistoryItem;
   466→  onViewDetails: (
   467→    item:
   468→      | Extract<UnifiedHistoryItem, { type: "prescription" }>
   469→      | Extract<UnifiedHistoryItem, { type: "manual_request" }>
   470→  ) => void;
   471→  onReorder: (orderId: string) => void;
   472→  onDownloadInvoice: (orderId: string, orderNumber: string) => void;
   473→  onTrack: (orderId: string) => void;
   474→  reorderingId: string | null;
   475→}) {
   476→  const typeMeta = TYPE_META[item.type];
   477→  const TypeIcon = typeMeta.icon;
   478→
   479→  return (
   480→    <Card className="gap-0 overflow-hidden rounded-xl border-border/50 p-0 shadow-premium-sm">
   481→      {/* ---- Header: type badge + number + date + status ---- */}
   482→      <div className="flex flex-wrap items-start justify-between gap-2 border-b border-border/50 p-4">
   483→        <div className="min-w-0 flex-1">
   484→          <div className="flex flex-wrap items-center gap-2">
   485→            <Badge variant="outline" className={`gap-1 ${typeMeta.className}`}>
   486→              <TypeIcon className="size-3" /> {typeMeta.label}
   487→            </Badge>
   488→            <p className="truncate font-mono text-sm font-semibold">{item.number}</p>
   489→            <StatusBadge status={item.status} label={item.statusLabel} />
   490→          </div>
   491→          <p className="mt-1 flex items-center gap-1 text-xs text-muted-foreground">
   492→            <Clock className="size-3" /> {formatDate(item.date)}
   493→          </p>
   494→        </div>
   495→      </div>
   496→
   497→      {/* ---- Body: type-specific content ---- */}
   498→      <div className="p-4">
   499→        {item.type === "order" ? (
   500→          <OrderBody item={item} />
   501→        ) : item.type === "prescription" ? (
   502→          <PrescriptionBody item={item} />
   503→        ) : (
   504→          <ManualRequestBody item={item} />
   505→        )}
   506→      </div>
   507→
   508→      {/* ---- Admin remarks (prescriptions / manual requests only) ---- */}
   509→      {item.type !== "order" && item.adminRemarks && (
   510→        <div className="mx-4 mb-3 flex items-start gap-2 rounded-md border border-muted bg-muted/40 p-2 text-xs">
   511→          <MessageCircle className="mt-0.5 size-3.5 shrink-0 text-muted-foreground" />
   512→          <div>
   513→            <p className="font-semibold text-foreground">Pharmacist&apos;s note</p>
   514→            <p className="mt-0.5 whitespace-pre-wrap text-muted-foreground">
   515→              {item.adminRemarks}
   516→            </p>
   517→          </div>
   518→        </div>
   519→      )}
   520→
   521→      {/* ---- Action buttons ---- */}
   522→      <ActivityActions
   523→        item={item}
   524→        onViewDetails={onViewDetails}
   525→        onReorder={onReorder}
   526→        onDownloadInvoice={onDownloadInvoice}
   527→        onTrack={onTrack}
   528→        reorderingId={reorderingId}
   529→      />
   530→    </Card>
   531→  );
   532→}

   534→// ---------------------------------------------------------------------------
   535→// Order body — items preview + grand total + payment + delivery meta.
   536→// ---------------------------------------------------------------------------
   537→function OrderBody({ item }: { item: Extract<UnifiedHistoryItem, { type: "order" }> }) {
   538→  const d = item.details;
   539→  const previewItems = d.items.slice(0, 3);
   540→  const extraItems = d.itemsCount - previewItems.length;
   541→  const isActive = ACTIVE_ORDER_STATUSES.has(item.status);
   542→  const hasDeliveryCharge = typeof d.deliveryCharge === "number" && d.deliveryCharge > 0;
   543→  const showEta =
   544→    isActive &&
   545→    d.estimatedDelivery &&
   546→    item.status !== "delivered" &&
   547→    item.status !== "cancelled";
   548→
   549→  return (
   550→    <>
   551→      {previewItems.length > 0 && (
   552→        <div className="mb-3">
   553→          <p className="mb-1 text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   554→            Items
   555→          </p>
   556→          <p className="text-sm text-foreground">
   557→            {previewItems.map((it) => it.name).join(", ")}

   940→// ---------------------------------------------------------------------------
   941→// Status timeline — visual progression of the prescription / manual request
   942→// lifecycle. Each step shows a status icon, label, and whether it's done /
   943→// current / pending. Stages: Submitted → Under Review → Approved/Rejected →
   944→// Order Created → Completed. We compute the reached stage from `status`.
   945→// ---------------------------------------------------------------------------
   946→type Stage =
   947→  | { key: "submitted"; label: string; done: boolean; current: boolean; icon: typeof CheckCircle2 }
   948→  | { key: "under_review"; label: string; done: boolean; current: boolean; icon: typeof Clock }
   949→  | { key: "verified"; label: string; done: boolean; current: boolean; icon: typeof CheckCircle2 }
   950→  | { key: "converted"; label: string; done: boolean; current: boolean; icon: typeof CheckCircle2 }
   951→  | { key: "rejected"; label: string; done: boolean; current: boolean; icon: typeof XCircle };
   952→
   953→function StatusTimeline({
   954→  status,
   955→  createdAt,
   956→  updatedAt,
   957→  convertedOrderId,
   958→}: {
   959→  status: string;
   960→  createdAt: string;
   961→  updatedAt: string;
   962→  convertedOrderId: string | null;
   963→}) {
   964→  // Determine which stages are "done" / "current" based on `status`.
   965→  // Order: pending → under_review → verified → converted
   966→  // Alt:   pending → under_review → rejected (rejected is terminal)
   967→  const stages: Stage[] = [
   968→    {
   969→      key: "submitted",
   970→      label: "Submitted",
   971→      done: true,
   972→      current: status === "pending",
   973→      icon: CheckCircle2,
   974→    },
   975→    {
   976→      key: "under_review",
   977→      label: "Under Review",
   978→      done:
   979→        status === "under_review" ||
   980→        status === "verified" ||
   981→        status === "converted" ||
   982→        status === "rejected",
   983→      current: status === "under_review",
   984→      icon: Clock,
   985→    },
   986→    {
   987→      key: "verified",
   988→      label: "Approved",
   989→      done: status === "verified" || status === "converted",
   990→      current: status === "verified",
   991→      icon: CheckCircle2,
   992→    },
   993→    {
   994→      key: "converted",
   995→      label: "Order Created",
   996→      done: status === "converted",
   997→      current: status === "converted",
   998→      icon: CheckCircle2,
   999→    },
  1000→  ];
  1001→
  1002→  // If rejected, show a terminal "Rejected" stage instead of verified/converted.
  1003→  const isRejected = status === "rejected";
  1004→
  1005→  return (
  1006→    <div>
  1007→      <p className="mb-3 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
  1008→        Status Timeline
  1009→      </p>
  1010→      <ol className="relative border-l border-muted pl-4">
  1011→        {/* Submitted stage — always present */}
  1012→        <TimelineStage
  1013→          stage={stages[0]}
  1014→          timestamp={createdAt}
  1015→          timestampLabel="Submitted"
  1016→        />
  1017→        {/* Under review stage — skip if rejected without going through review */}
  1018→        <TimelineStage
  1019→          stage={stages[1]}
  1020→          timestamp={stages[1].done ? updatedAt : null}
  1021→          timestampLabel="Marked Under Review"
  1022→        />
  1023→        {isRejected ? (
  1024→          <TimelineStage
  1025→            stage={{
  1026→              key: "rejected",
  1027→              label: "Rejected",
  1028→              done: true,
  1029→              current: true,
  1030→              icon: XCircle,
  1031→            }}
  1032→            timestamp={updatedAt}
  1033→            timestampLabel="Rejected"
  1034→            tone="danger"
  1035→          />
  1036→        ) : (
  1037→          <>
  1038→            <TimelineStage
  1039→              stage={stages[2]}
  1040→              timestamp={stages[2].done ? updatedAt : null}
  1041→              timestampLabel="Approved"
  1042→            />
  1043→            <TimelineStage
  1044→              stage={stages[3]}
  1045→              timestamp={stages[3].done ? updatedAt : null}
  1046→              timestampLabel="Order Created"
  1047→              isLast
  1048→              convertedOrderId={convertedOrderId}
  1049→            />
  1050→          </>
  1051→        )}
  1052→      </ol>
  1053→    </div>
  1054→  );
  1055→}

  1057→// ---------------------------------------------------------------------------
  1058→// Single timeline stage row — icon, label, timestamp, optional "View order"
  1059→// link for the converted stage.
  1060→// ---------------------------------------------------------------------------
  1061→function TimelineStage({
  1062→  stage,
  1063→  timestamp,
  1064→  timestampLabel,
  1065→  tone = "default",
  1066→  isLast = false,
  1067→  convertedOrderId,
  1068→}: {
  1069→  stage: Stage;
  1070→  timestamp: string | null;
  1071→  timestampLabel: string;
  1072→  tone?: "default" | "danger";
  1073→  isLast?: boolean;
  1074→  convertedOrderId?: string | null;
  1075→}) {
  1076→  const navigate = useUI((s) => s.navigate);
  1077→  const Icon = stage.icon;
  1078→
  1079→  // Color logic:
  1080→  //   done && !current → emerald (completed)
  1081→  //   current          → amber (in progress) or rose (rejected)
  1082→  //   !done            → muted (pending)
  1083→  let iconColor = "text-muted-foreground";
  1084→  let iconBg = "bg-muted";
  1085→  if (tone === "danger" && stage.current) {
  1086→    iconColor = "text-rose-700";
  1087→    iconBg = "bg-rose-100";
  1088→  } else if (stage.done && !stage.current) {
  1089→    iconColor = "text-emerald-700";
  1090→    iconBg = "bg-emerald-100";
  1091→  } else if (stage.current) {
  1092→    iconColor = "text-amber-700";
  1093→    iconBg = "bg-amber-100";
  1094→  }
  1095→
  1096→  return (
  1097→    <li className={`relative ${isLast ? "" : "mb-4"}`}>
  1098→      {/* Icon — positioned over the left border */}
  1099→      <span
  1100→        className={`absolute -left-[1.65rem] flex size-6 items-center justify-center rounded-full ${iconBg}`}
  1101→      >
  1102→        <Icon className={`size-3.5 ${iconColor}`} />
  1103→      </span>
  1104→      <p className={`text-sm font-medium ${stage.current ? "text-foreground" : stage.done ? "text-foreground" : "text-muted-foreground"}`}>
  1105→        {stage.label}
  1106→      </p>
  1107→      {timestamp && (
  1108→        <p className="text-xs text-muted-foreground">
  1109→          {timestampLabel} · {formatDateTime(timestamp)}
  1110→        </p>
  1111→      )}
  1112→      {!timestamp && !stage.done && (
  1113→        <p className="text-xs italic text-muted-foreground">Pending</p>
  1114→      )}
  1115→      {/* "View order" link for the converted stage */}
  1116→      {stage.key === "converted" && stage.done && convertedOrderId && (
  1117→        <Button
  1118→          variant="link"
  1119→          size="sm"
  1120→          className="mt-1 h-auto gap-1 p-0 text-primary"
  1121→          onClick={() =>
  1122→            navigate({ name: "track-order", orderId: convertedOrderId })
  1123→          }
  1124→        >
  1125→          Track order <ChevronRight className="size-3" />
  1126→        </Button>
  1127→      )}
  1128→    </li>
  1129→  );
  1130→}
