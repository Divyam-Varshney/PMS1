   239→   239→function NavList({ admin, searchQuery }: { admin: AdminInfo; searchQuery: string }) {
   240→   240→  const view = useAdminStore((s) => s.view);
   241→   241→  const navigate = useAdminStore((s) => s.navigate);
   242→   242→  const [collapsedGroups, setCollapsedGroups] = useState<Set<string>>(new Set());
   243→   243→
   244→   244→  // Fetch lightweight badge counts
   245→   245→  const { data: counts } = useQuery({
   246→   246→    queryKey: ["admin-counts"],
   247→   247→    queryFn: () => api.get<{ pendingOrders: number; lowStock: number; pendingPrescriptions: number; pendingManualRequests: number; unreadNotifications: number }>("/api/admin/counts"),
   248→   248→    refetchInterval: 30 * 1000,
   249→   249→  });
   250→   250→
   251→   251→  const badgeFor = (viewName: string): number | null => {
   252→   252→    if (!counts) return null;
   253→   253→    switch (viewName) {
   254→   254→      case "orders": return counts.pendingOrders || null;
   255→   255→      case "products": return counts.lowStock || null;
   256→   256→      case "prescriptions": return counts.pendingPrescriptions || null;
   257→   257→      case "manual-requests": return counts.pendingManualRequests || null;
   258→   258→      default: return null;
   259→   259→    }
   260→   260→  };
   261→   261→
   262→   262→  const toggleGroup = (label: string) => {
   263→   263→    setCollapsedGroups((prev) => {
   264→   264→      const next = new Set(prev);
   265→   265→      if (next.has(label)) next.delete(label);
   266→   266→      else next.add(label);
   267→   267→      return next;
   268→   268→    });
   269→   269→  };
   270→   270→
   271→   271→  const isVisible = (item: { permission: AdminPermissionKey }) => hasPermission(admin, item.permission);
   272→   272→
   273→   273→  // When searching, show all items flat (ignore group collapse)
   274→   274→  const isSearching = searchQuery.trim().length > 0;
   275→   275→  const query = searchQuery.toLowerCase().trim();
   276→   276→
   277→   277→  const visibleGroups = NAV_GROUPS.map((g) => ({
   278→   278→    ...g,
   279→   279→    items: g.items.filter((item) => {
   280→   280→      if (!isVisible(item)) return false;
   281→   281→      if (!isSearching) return true;
   282→   282→      return item.label.toLowerCase().includes(query);
   283→   283→    }),
   284→   284→  })).filter((g) => g.items.length > 0);
   285→   285→
   286→   286→  // Dashboard is always visible (if permitted)
   287→   287→  const showDashboard = hasPermission(admin, "dashboard") && (!isSearching || "dashboard".includes(query));
   288→   288→
   289→   289→  return (
   290→   290→    <SidebarContent>
   291→   291→      {/* Dashboard — always at top, outside groups */}
   292→   292→      {showDashboard && (
   293→   293→        <SidebarGroup>
   294→   294→          <SidebarMenu>
   295→   295→            <SidebarMenuItem>
   296→   296→              <SidebarMenuButton
   297→   297→                isActive={view.name === "dashboard"}
   298→   298→                onClick={() => navigate({ name: "dashboard" })}
   299→   299→                tooltip="Dashboard"
   300→   300→                className="transition-premium"
   301→   301→              >
   302→   302→                <LayoutDashboard />
   303→   303→                <span>Dashboard</span>
   304→   304→              </SidebarMenuButton>
   305→   305→            </SidebarMenuItem>
   306→   306→          </SidebarMenu>
   307→   307→        </SidebarGroup>
   308→   308→      )}
   309→   309→
   310→   310→      {/* Collapsible nav groups */}
   311→   311→      {visibleGroups.map((g) => {
   312→   312→        const isCollapsed = collapsedGroups.has(g.label) && !isSearching;
   313→   313→        return (
   314→   314→          <Collapsible
   315→   315→            key={g.label}
   316→   316→            open={!isCollapsed}
   317→   317→            // When searching, all groups are forced open so matches are visible.
   318→   318→            // Don't call toggleGroup during search — it would silently mutate
   319→   319→            // the collapsed state and cause unexpected expand/collapse when the
   320→   320→            // search is cleared.
   321→   321→            onOpenChange={() => !isSearching && toggleGroup(g.label)}
   322→   322→          >
   323→   323→            <SidebarGroup>
   324→   324→              <CollapsibleTrigger asChild>
   325→   325→                <SidebarGroupLabel className="cursor-pointer select-none hover:bg-sidebar-accent/60 transition-colors admin-section-label">
   326→   326→                  <span>{g.label}</span>
   327→   327→                  <ChevronRight className={`ml-auto size-3.5 transition-transform ${isCollapsed ? "" : "rotate-90"}`} />
   328→   328→                </SidebarGroupLabel>
   329→   329→              </CollapsibleTrigger>
   330→   330→              <CollapsibleContent>
   331→   331→                <SidebarMenu>
   332→   332→                  {g.items.map((item) => {
   333→   333→                    const isActive =
   334→   334→                      view.name === item.view.name ||
   335→   335→                      (item.view.name === "products" && view.name === "product-edit") ||
   336→   336→                      (item.view.name === "orders" && view.name === "order-detail") ||
   337→   337→                      (item.view.name === "prescriptions" && view.name === "prescription-detail") ||
   338→   338→                      (item.view.name === "manual-requests" && view.name === "manual-request-detail") ||
   339→   339→                      (item.view.name === "customers" && view.name === "customer-detail");
   340→   340→                    return (
   341→   341→                      <SidebarMenuItem key={item.label}>
   342→   342→                        <SidebarMenuButton
   343→   343→                          isActive={isActive}
   344→   344→                          onClick={() => navigate(item.view)}
   345→   345→                          tooltip={item.label}
   346→   346→                          className="transition-premium"
   347→   347→                        >
   348→   348→                          <item.icon />
   349→   349→                          <span>{item.label}</span>
   350→   350→                          {badgeFor(item.view.name) !== null && (
   351→   351→                            <Badge className="ml-auto h-5 min-w-5 justify-center px-1.5 text-[10px] font-bold bg-amber-500 text-white hover:bg-amber-500 animate-badge-pulse">
   352→   352→                              {badgeFor(item.view.name)}
   353→   353→                            </Badge>
   354→   354→                          )}
   355→   355→                        </SidebarMenuButton>
   356→   356→                      </SidebarMenuItem>
   357→   357→                    );
   358→   358→                  })}
   359→   359→                </SidebarMenu>
   360→   360→              </CollapsibleContent>
   361→   361→            </SidebarGroup>
   362→   362→          </Collapsible>
   363→   363→        );
   364→   364→      })}
   365→   365→
   366→   366→      {/* No results when searching */}
   367→   367→      {isSearching && visibleGroups.length === 0 && !showDashboard && (
   368→   368→        <div className="px-4 py-8 text-center text-xs text-muted-foreground">
   369→   369→          No navigation items match “{searchQuery}”
   370→   370→        </div>
   371→   371→      )}
   372→   372→    </SidebarContent>
   373→   373→  );
   374→   374→}

   380→   380→function Breadcrumbs({ viewName, navigate }: { viewName: string; navigate: (v: AdminView) => void }) {
   381→   381→  const crumbs: Array<{ label: string; view?: AdminView }> = [];
   382→   382→
   383→   383→  // Always start with Dashboard
   384→   384→  crumbs.push({ label: "Dashboard", view: { name: "dashboard" } });
   385→   385→
   386→   386→  // Add parent if this is a detail page
   387→   387→  const parentName = BREADCRUMB_PARENT[viewName];
   388→   388→  if (parentName) {
   389→   389→    crumbs.push({ label: TITLE_MAP[parentName] || parentName, view: { name: parentName } as AdminView });
   390→   390→  }
   391→   391→
   392→   392→  // Add current page (not clickable)
   393→   393→  const currentLabel = TITLE_MAP[viewName] || viewName;
   394→   394→  if (viewName !== "dashboard") {
   395→   395→    crumbs.push({ label: currentLabel });
   396→   396→  }
   397→   397→
   398→   398→  return (
   399→   399→    <nav className="flex items-center gap-1 text-sm text-muted-foreground min-w-0">
   400→   400→      {crumbs.map((crumb, i) => {
   401→   401→        const isLast = i === crumbs.length - 1;
   402→   402→        return (
   403→   403→          <span key={i} className="flex items-center gap-1 min-w-0">
   404→   404→            {i > 0 && <ChevronRight className="size-3.5 shrink-0 text-muted-foreground/50" />}
   405→   405→            {crumb.view && !isLast ? (
   406→   406→              <button
   407→   407→                onClick={() => navigate(crumb.view!)}
   408→   408→                className="hover:text-foreground transition-colors truncate"
   409→   409→              >
   410→   410→                {crumb.label}
   411→   411→              </button>
   412→   412→            ) : (
   413→   413→              <span className={`truncate ${isLast ? "font-semibold text-foreground" : ""}`}>
   414→   414→                {crumb.label}
   415→   415→              </span>
   416→   416→            )}
   417→   417→          </span>
   418→   418→        );
   419→   419→      })}
   420→   420→    </nav>
   421→   421→  );
   422→   422→}

   428→   428→function CommandPalette({ open, onOpenChange, navigate }: { open: boolean; onOpenChange: (v: boolean) => void; navigate: (v: AdminView) => void }) {
   429→   429→  const [query, setQuery] = useState("");
   430→   430→
   431→   431→  const filtered = useMemo(() => {
   432→   432→    if (!query.trim()) return ALL_NAV_ITEMS;
   433→   433→    const q = query.toLowerCase();
   434→   434→    return ALL_NAV_ITEMS.filter((item) => item.label.toLowerCase().includes(q));
   435→   435→  }, [query]);
   436→   436→
   437→   437→  const handleSelect = (view: AdminView) => {
   438→   438→    navigate(view);
   439→   439→    onOpenChange(false);
   440→   440→    setQuery("");
   441→   441→  };
   442→   442→
   443→   443→  return (
   444→   444→    <Dialog open={open} onOpenChange={onOpenChange}>
   445→   445→      <DialogContent className="p-0 gap-0 overflow-hidden max-w-lg shadow-premium-xl">
   446→   446→        {/* Accessibility: DialogTitle is required by Radix Dialog for screen
   447→   447→            reader support. Rendered visually-hidden (sr-only). */}
   448→   448→        <DialogTitle className="sr-only">Command palette</DialogTitle>
   449→   449→        <div className="flex items-center border-b border-border/70 px-4 bg-premium-gradient">
   450→   450→          <Search className="size-4 mr-3 text-muted-foreground shrink-0" />
   451→   451→          <Input
   452→   452→            autoFocus

   496→   496→function StoreOpenToggle() {

   498→   498→  const { data, isLoading } = useQuery({

   757→   757→/** Wrapper to fetch admin from /me on mount; renders children only when logged in. */
   758→   758→export function AdminGuard({
   759→   759→  onLoggedOut,
   760→   760→  children,
   761→   761→}: {
   762→   762→  onLoggedOut: () => void;
   763→   763→  children: (admin: AdminInfo) => ReactNode;
   764→   764→}) {
   765→   765→  const [admin, setAdmin] = useState<AdminInfo | null>(null);
   766→   766→  const [loading, setLoading] = useState(true);
   767→   767→
   768→   768→  useEffect(() => {
   769→   769→    let mounted = true;
   770→   770→    api
   771→   771→      .get<AdminInfo>("/api/admin-auth/me")
   772→   772→      .then((a) => mounted && setAdmin(a))
   773→   773→      .catch(() => mounted && onLoggedOut())
   774→   774→      .finally(() => mounted && setLoading(false));
   775→   775→    return () => {
   776→   776→      mounted = false;
   777→   777→    };
   778→   778→  }, [onLoggedOut]);
   779→   779→
   780→   780→  if (loading) {
   781→   781→    return (
   782→   782→      <div className="min-h-screen flex items-center justify-center bg-background">
   783→   783→        <Loader2 className="size-8 animate-spin text-primary" />
   784→   784→      </div>
   785→   785→    );
   786→   786→  }
   787→   787→  if (!admin) return null;
   788→   788→  return <>{children(admin)}</>;
   789→   789→}
