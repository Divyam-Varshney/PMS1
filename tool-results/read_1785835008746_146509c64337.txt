    53→    53→  name: string;
    54→    54→  slug: string;

    88→    88→  products?: ProductSearchResult[];
    89→    89→  action?: AssistantAction;
    90→    90→  suggestions?: string[];
    91→    91→  bundleIds?: string[];
    92→    92→  faqQuestion?: string;

   109→   109→];
   110→   110→

   113→   113→  "Paracetamol for fever",
   114→   114→  "Cough syrup for cold",
   115→   115→  "Vitamin C tablets",
   116→   116→  "Diabetes care products",

   118→   118→
   119→   119→export function HealthAssistantWidget() {
   120→   120→  const navigate = useUI((s) => s.navigate);
   121→   121→  const [open, setOpen] = useState(false);
   122→   122→  const [messages, setMessages] = useState<Message[]>([WELCOME_MESSAGE]);
   123→   123→  const [input, setInput] = useState("");
   124→   124→  const [loading, setLoading] = useState(false);
   125→   125→  const [unread, setUnread] = useState(false);
   126→   126→  const scrollRef = useRef<HTMLDivElement>(null);
   127→   127→  const inputRef = useRef<HTMLInputElement>(null);
   128→   128→
   129→   129→  // Auto-scroll to bottom on new message / loading change.
   130→   130→  useEffect(() => {
   131→   131→    if (scrollRef.current) {
   132→   132→      scrollRef.current.scrollTo({
   133→   133→        top: scrollRef.current.scrollHeight,
   134→   134→        behavior: "smooth",
   135→   135→      });
   136→   136→    }
   137→   137→  }, [messages, loading]);
   138→   138→
   139→   139→  // Focus the input when the panel opens, clear the unread dot.
   140→   140→  useEffect(() => {
   141→   141→    if (open) {
   142→   142→      setUnread(false);
   143→   143→      const t = setTimeout(() => inputRef.current?.focus(), 220);
   144→   144→      return () => clearTimeout(t);
   145→   145→    }
   146→   146→  }, [open]);
   147→   147→
   148→   148→  // Broadcast a custom event whenever the panel opens — the WelcomePopup
   149→   149→  // listens for this so it can auto-dismiss if the user opens the assistant
   150→   150→  // themselves (instead of clicking the popup's "Chat Now" button).
   151→   151→  useEffect(() => {
   152→   152→    if (open) {
   153→   153→      window.dispatchEvent(new CustomEvent("pms:assistant-open"));
   154→   154→    }
   155→   155→  }, [open]);
   156→   156→
   157→   157→  // Listen for open-requests from the WelcomePopup's "Chat Now" button.
   158→   158→  // This decouples the popup from the widget — no shared parent state needed.
   159→   159→  useEffect(() => {
   160→   160→    const handler = () => setOpen(true);
   161→   161→    window.addEventListener("pms:assistant-open-request", handler);
   162→   162→    return () => window.removeEventListener("pms:assistant-open-request", handler);
   163→   163→  }, []);
   164→   164→
   165→   165→  // -------------------------------------------------------------------------
   166→   166→  // Send a message to the assistant API.
   167→   167→  // -------------------------------------------------------------------------
   168→   168→  const send = useCallback(
   169→   169→    async (text?: string) => {
   170→   170→      const content = (text ?? input).trim();
   171→   171→      if (!content || loading) return;
   172→   172→
   173→   173→      const userMsg: Message = { role: "user", content, ts: Date.now() };
   174→   174→      const historyForApi = [...messages, userMsg]
   175→   175→        .filter((m) => m.ts !== 0) // exclude the static welcome
   176→   176→        .map((m) => ({ role: m.role, content: m.content }));
   177→   177→
   178→   178→      setMessages((prev) => [...prev, userMsg]);
   179→   179→      setInput("");
   180→   180→      setLoading(true);
   181→   181→
   182→   182→      try {
   183→   183→        const res = await fetch("/api/health-assistant", {
   184→   184→          method: "POST",
   185→   185→          headers: { "Content-Type": "application/json" },
   186→   186→          body: JSON.stringify({ messages: historyForApi }),
   187→   187→        });
   188→   188→        const data = await res.json();
   189→   189→        if (data?.ok && data.data) {
   190→   190→          const r = data.data as AssistantResponse;
   191→   191→          setMessages((prev) => [
   192→   192→            ...prev,
   193→   193→            {
   194→   194→              role: "assistant",
   195→   195→              content: r.reply,
   196→   196→              ts: Date.now(),
   197→   197→              products: r.products,
   198→   198→              action: r.action,
   199→   199→              suggestions: r.suggestions,
   200→   200→              bundleIds: r.bundleIds,
   201→   201→              faqQuestion: r.faqQuestion,
   202→   202→            },
   203→   203→          ]);
   204→   204→        } else {
   205→   205→          setMessages((prev) => [
   206→   206→            ...prev,
   207→   207→            {
   208→   208→              role: "assistant",
   209→   209→              content: data?.error || "Sorry, I couldn't respond right now. Please try again later.",
   210→   210→              ts: Date.now(),
   211→   211→            },
   212→   212→          ]);
   213→   213→        }
   214→   214→        if (!open) setUnread(true);
   215→   215→      } catch {
   216→   216→        setMessages((prev) => [
   217→   217→          ...prev,
   218→   218→          {
   219→   219→            role: "assistant",
   220→   220→            content: "I'm having trouble connecting. Please try again in a moment.",
   221→   221→            ts: Date.now(),
   222→   222→          },
   223→   223→        ]);
   224→   224→      } finally {
   225→   225→        setLoading(false);
   226→   226→      }
   227→   227→    },
   228→   228→    [input, loading, messages, open]
   229→   229→  );
   230→   230→
   231→   231→  const handleSubmit = (e?: FormEvent) => {
   232→   232→    e?.preventDefault();
   233→   233→    void send();
   234→   234→  };
   235→   235→
   236→   236→  // Click a starter prompt / suggestion chip — send it immediately.
   237→   237→  const sendPrompt = (prompt: string) => {
   238→   238→    void send(prompt);
   239→   239→  };
   240→   240→
   241→   241→  // Quick action: either navigate directly or focus the input.
   242→   242→  const handleQuickAction = (action: (typeof QUICK_ACTIONS)[number]) => {
   243→   243→    if (action.view) {
   244→   244→      navigate(action.view);
   245→   245→      setOpen(false);
   246→   246→    } else {
   247→   247→      // "Search a medicine" — focus the input so the user can type.
   248→   248→      setTimeout(() => inputRef.current?.focus(), 200);
   249→   249→    }
   250→   250→  };
   251→   251→
   252→   252→  // Click a product card → navigate to the product detail view.
   253→   253→  const viewProduct = (p: ProductSearchResult) => {
   254→   254→    navigate({ name: "product", productId: p.id, slug: p.slug });
   255→   255→    setOpen(false);
   256→   256→  };
   257→   257→
   258→   258→  // "Request This Medicine" button — store the user's last query as a prefill
   259→   259→  // hint, then navigate to the manual-request view.
   260→   260→  const goToMedicineRequest = () => {
   261→   261→    const lastUser = [...messages].reverse().find((m) => m.role === "user");
   262→   262→    if (lastUser?.content) {
   263→   263→      try {
   264→   264→        window.localStorage.setItem("pms:medicine-request-prefill", lastUser.content);
   265→   265→      } catch {
   266→   266→        // localStorage may be unavailable (private mode) — silently ignore.
   267→   267→      }
   268→   268→    }
   269→   269→    navigate({ name: "manual-request" });
   270→   270→    setOpen(false);
   271→   271→  };
   272→   272→
   273→   273→  const showQuickActions = messages.length === 1 && !loading;
   274→   274→
   275→   275→  return (
   276→   276→    <>
   277→   277→      {/* Floating button */}
   278→   278→      <motion.button
   279→   279→        initial={{ scale: 0, opacity: 0 }}
   280→   280→        animate={{ scale: 1, opacity: 1 }}
   281→   281→        transition={{ delay: 1, type: "spring", stiffness: 260, damping: 20 }}
   282→   282→        whileHover={{ scale: 1.08 }}
   283→   283→        whileTap={{ scale: 0.92 }}
   284→   284→        onClick={() => setOpen((v) => !v)}
   285→   285→        aria-label={open ? "Close assistant" : "Open AI Health Assistant"}
   286→   286→        className={cn(
   287→   287→          "fixed bottom-20 right-4 z-50 flex size-14 items-center justify-center rounded-full shadow-lg shadow-emerald-600/30 transition-colors sm:bottom-6",
   288→   288→          "bg-gradient-to-br from-emerald-600 to-teal-600 text-white ring-4 ring-background",
   289→   289→          "hover:shadow-xl hover:shadow-emerald-600/40"
   290→   290→        )}
   291→   291→      >
   292→   292→        <AnimatePresence mode="wait">
   293→   293→          {open ? (
   294→   294→            <motion.span
   295→   295→              key="close"
   296→   296→              initial={{ rotate: -90, opacity: 0 }}
   297→   297→              animate={{ rotate: 0, opacity: 1 }}
   298→   298→              exit={{ rotate: 90, opacity: 0 }}
   299→   299→              transition={{ duration: 0.15 }}
   300→   300→            >
   301→   301→              <X className="size-6" />
   302→   302→            </motion.span>
   303→   303→          ) : (
   304→   304→            <motion.span
   305→   305→              key="open"
   306→   306→              initial={{ rotate: 90, opacity: 0 }}
   307→   307→              animate={{ rotate: 0, opacity: 1 }}
   308→   308→              exit={{ rotate: -90, opacity: 0 }}
   309→   309→              transition={{ duration: 0.15 }}
   310→   310→              className="relative"
   311→   311→            >
   312→   312→              <MessageCircle className="size-6" />
   313→   313→              {unread && (
   314→   314→                <span className="absolute -right-1 -top-1 flex size-3">
   315→   315→                  <span className="absolute inline-flex size-full animate-ping rounded-full bg-rose-400 opacity-75" />
   316→   316→                  <span className="relative inline-flex size-3 rounded-full bg-rose-500" />
   317→   317→                </span>
   318→   318→              )}
   319→   319→            </motion.span>
   320→   320→          )}
   321→   321→        </AnimatePresence>
   322→   322→      </motion.button>
   323→   323→
   324→   324→      {/* Chat panel */}
   325→   325→      <AnimatePresence>
   326→   326→        {open && (
   327→   327→          <motion.div
   328→   328→            initial={{ opacity: 0, y: 24, scale: 0.96 }}
   329→   329→            animate={{ opacity: 1, y: 0, scale: 1 }}
   330→   330→            exit={{ opacity: 0, y: 24, scale: 0.96 }}
   331→   331→            transition={{ duration: 0.22, ease: "easeOut" }}
   332→   332→            className={cn(
   333→   333→              "fixed bottom-36 right-4 z-50 flex w-[calc(100vw-2rem)] max-w-sm flex-col overflow-hidden rounded-2xl border border-border/60 bg-background shadow-2xl shadow-emerald-900/10 sm:bottom-24",
   334→   334→              "h-[min(75vh,560px)]"
   335→   335→            )}
   336→   336→          >
   337→   337→            {/* Header */}
   338→   338→            <div className="flex items-center gap-3 border-b border-border/60 bg-gradient-to-r from-emerald-600 to-teal-600 p-3.5 text-white">
   339→   339→              <div className="flex size-9 items-center justify-center rounded-full bg-white/20 backdrop-blur">
   340→   340→                <Bot className="size-5" />
   341→   341→              </div>
   342→   342→              <div className="flex-1">
   343→   343→                <div className="flex items-center gap-1.5">
   344→   344→                  <span className="text-sm font-bold">PMS Assistant</span>
   345→   345→                  <Sparkles className="size-3.5 text-amber-200" />
   346→   346→                </div>
   347→   347→                <div className="flex items-center gap-1.5 text-[11px] text-emerald-50/90">
   348→   348→                  <span className="size-1.5 animate-pulse rounded-full bg-emerald-300" />
   349→   349→                  AI-powered · Online
   350→   350→                </div>
   351→   351→              </div>
   352→   352→              <button
   353→   353→                onClick={() => setOpen(false)}
   354→   354→                aria-label="Close chat"
   355→   355→                className="rounded-lg p-1.5 transition-colors hover:bg-white/20"
   356→   356→              >
   357→   357→                <X className="size-4" />
   358→   358→              </button>
   359→   359→            </div>
   360→   360→
   361→   361→            {/* Messages */}
   362→   362→            <div
   363→   363→              ref={scrollRef}
   364→   364→              className="flex-1 space-y-3 overflow-y-auto bg-muted/20 p-3.5 [scrollbar-width:thin]"
   365→   365→            >
   366→   366→              {messages.map((m, i) => (
   367→   367→                <MessageBubble
   368→   368→                  key={i}
   369→   369→                  message={m}
   370→   370→                  onViewProduct={viewProduct}
   371→   371→                  onGoToMedicineRequest={goToMedicineRequest}
   372→   372→                  onSuggestionClick={sendPrompt}
   373→   373→                  onBrowseBundles={() => {
   374→   374→                    navigate({ name: "bundles" });
   375→   375→                    setOpen(false);
   376→   376→                  }}
   377→   377→                />
   378→   378→              ))}
   379→   379→
   380→   380→              {/* Typing indicator */}
   381→   381→              <AnimatePresence>
   382→   382→                {loading && (
   383→   383→                  <motion.div
   384→   384→                    initial={{ opacity: 0, y: 8 }}
   385→   385→                    animate={{ opacity: 1, y: 0 }}
   386→   386→                    exit={{ opacity: 0, y: 8 }}
   387→   387→                    className="flex gap-2"
   388→   388→                  >
   389→   389→                    <div className="flex size-7 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-emerald-600 to-teal-600 text-white ring-2 ring-background">
   390→   390→                      <Bot className="size-3.5" />
   391→   391→                    </div>
   392→   392→                    <div className="flex items-center gap-1 rounded-2xl rounded-tl-sm bg-background px-4 py-3 ring-1 ring-border/50">
   393→   393→                      <span className="size-2 animate-bounce rounded-full bg-muted-foreground/60 [animation-delay:-0.3s]" />
   394→   394→                      <span className="size-2 animate-bounce rounded-full bg-muted-foreground/60 [animation-delay:-0.15s]" />
   395→   395→                      <span className="size-2 animate-bounce rounded-full bg-muted-foreground/60" />
   396→   396→                    </div>
   397→   397→                  </motion.div>
   398→   398→                )}
   399→   399→              </AnimatePresence>
   400→   400→
   401→   401→              {/* Quick action buttons — shown on first open only */}
   402→   402→              <AnimatePresence>
   403→   403→                {showQuickActions && (
   404→   404→                  <motion.div
   405→   405→                    initial={{ opacity: 0, y: 8 }}
   406→   406→                    animate={{ opacity: 1, y: 0 }}
   407→   407→                    exit={{ opacity: 0, y: 8 }}
   408→   408→                    transition={{ delay: 0.2 }}
   409→   409→                    className="space-y-2 pt-1"
   410→   410→                  >
   411→   411→                    <div className="grid grid-cols-2 gap-1.5">
   412→   412→                      {QUICK_ACTIONS.map((qa) => (
   413→   413→                        <button
   414→   414→                          key={qa.label}
   415→   415→                          onClick={() => handleQuickAction(qa)}
   416→   416→                          className="flex items-center gap-1.5 rounded-lg border border-emerald-200 bg-emerald-50 px-2.5 py-2 text-[11px] font-medium text-emerald-700 transition-colors hover:bg-emerald-100 dark:border-emerald-800 dark:bg-emerald-950/40 dark:text-emerald-300 dark:hover:bg-emerald-900/40"
   417→   417→                        >
   418→   418→                          <qa.icon className="size-3.5 shrink-0" />
   419→   419→                          <span className="truncate">{qa.label}</span>
   420→   420→                        </button>
   421→   421→                      ))}
   422→   422→                    </div>
   423→   423→                    {/* Starter prompts */}
   424→   424→                    <div className="flex flex-wrap gap-1.5 pt-1">
   425→   425→                      {STARTER_PROMPTS.map((p) => (
   426→   426→                        <button
   427→   427→                          key={p}
   428→   428→                          onClick={() => sendPrompt(p)}
   429→   429→                          className="rounded-full border border-border/60 bg-background px-2.5 py-1 text-[11px] font-medium text-muted-foreground transition-colors hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-700 dark:hover:bg-emerald-950/40 dark:hover:text-emerald-300"
   430→   430→                        >
   431→   431→                          {p}
   432→   432→                        </button>
   433→   433→                      ))}
   434→   434→                    </div>
   435→   435→                  </motion.div>
   436→   436→                )}
   437→   437→              </AnimatePresence>
   438→   438→            </div>
   439→   439→
   440→   440→            {/* Medical disclaimer */}
   441→   441→            <div className="border-t border-border/60 bg-amber-50/50 px-3.5 py-1.5 text-center text-[10px] leading-tight text-amber-700 dark:bg-amber-950/20 dark:text-amber-400">
   442→   442→              ⚕️ AI assistant — not a doctor. Always consult a licensed pharmacist for medical advice.
   443→   443→            </div>
   444→   444→
   445→   445→            {/* Input */}
   446→   446→            <form
   447→   447→              onSubmit={handleSubmit}
   448→   448→              className="flex items-center gap-2 border-t border-border/60 bg-background p-2.5"
   449→   449→            >
   450→   450→              <input
   451→   451→                ref={inputRef}
   452→   452→                type="text"
   453→   453→                value={input}
   454→   454→                onChange={(e) => setInput(e.target.value)}
   455→   455→                placeholder="Ask about medicines, health..."
   456→   456→                maxLength={1000}
   457→   457→                disabled={loading}
   458→   458→                className="flex-1 rounded-full border border-border/60 bg-muted/30 px-3.5 py-2 text-sm outline-none transition-colors placeholder:text-muted-foreground/60 focus:border-emerald-400 focus:bg-background focus:ring-2 focus:ring-emerald-400/20 disabled:opacity-50"
   459→   459→              />
   460→   460→              <button
   461→   461→                type="submit"
   462→   462→                disabled={!input.trim() || loading}
   463→   463→                aria-label="Send message"
