     2→     2→// File: src/components/customer/device-registration-wizard.tsx
     3→     3→// Purpose: New Device Registration Wizard (Phase 40). Replaces the old
     4→     4→//          `notification-onboarding.tsx` with a 4-step guided flow inspired
     5→     5→//          by Amazon / Flipkart / Blinkit / Swiggy onboarding.
     6→     6→//
     7→     7→//  Steps:
     8→     8→//    1. WELCOME       — explain benefits, two buttons: "Turn On" / "Skip"
     9→     9→//    2. PERMISSION    — request browser permission, handle deny gracefully
    10→    10→//    3. REGISTERING   — show progress as we register SW + create subscription
    11→    11→//                       + POST to backend + send welcome push
    12→    12→//    4. TEST_RESULT   — show whether the welcome push was delivered
    13→    13→//    5. DONE          — "You're All Set!" success screen
    14→    14→//
    15→    15→//  When the wizard should NOT appear (checked on mount + on auth change):
    16→    16→//    • Not authenticated
    17→    17→//    • Push API not supported by the browser
    18→    18→//    • POST /api/device-registrations/validate returns shouldShowWizard=false
    19→    19→//
    20→    20→//  When the wizard SHOULD appear:
    21→    21→//    • New device (no DeviceRegistration row)
    22→    22→//    • status="pending" (started but not completed)
    23→    23→//    • status="completed" but permission revoked / subscription expired
    24→    24→//
    25→    25→//  Skip behaviour:
    26→    26→//    • POST /api/device-registrations/skip with the device info.
    27→    27→//    • Wizard never re-appears on this device (status="skipped").
    28→    28→//    • Customer can still enable from Profile → Settings → App Notifications.
    29→    29→//
    30→    30→//  The existing App Notification Center + automatic notification system are
    31→    31→//  untouched. This wizard ONLY orchestrates the onboarding UX.
    32→    32→// ============================================================================

    66→    66→type WizardStep = "welcome" | "permission" | "registering" | "registering_failed" | "test_result" | "done" | "denied";
    67→    67→
    68→    68→interface StepProgress {
    69→    69→  step: string;
    70→    70→  label: string;
    71→    71→  status: "pending" | "active" | "done" | "failed";

    82→    82→export function DeviceRegistrationWizard({
    83→    83→  isAuthenticated,
    84→    84→}: {
    85→    85→  isAuthenticated: boolean;
    86→    86→}) {
    87→    87→  const qc = useQueryClient();
    88→    88→  const [open, setOpen] = useState(false);
    89→    89→  const [step, setStep] = useState<WizardStep>("welcome");
    90→    90→  const [busy, setBusy] = useState(false);
    91→    91→  const [pushSupported, setPushSupported] = useState(true);
    92→    92→  const [testPushSent, setTestPushSent] = useState(false);
    93→    93→  const [deviceLabel, setDeviceLabel] = useState<string>("");
    94→    94→  const [steps, setSteps] = useState<StepProgress[]>([]);
    95→    95→  const validatedRef = useRef(false);
    96→    96→
    97→    97→  // ─── Mount: detect Push API support ────────────────────────────────────
    98→    98→  useEffect(() => {
    99→    99→    if (typeof window === "undefined") return;
   100→   100→    const supported =
   101→   101→      "serviceWorker" in navigator &&
   102→   102→      "PushManager" in window &&
   103→   103→      "Notification" in window;
   104→   104→    setPushSupported(supported);
   105→   105→  }, []);
   106→   106→
   107→   107→  // ─── Auth change: check whether the wizard should appear ────────────────
   108→   108→  useEffect(() => {
   109→   109→    if (!isAuthenticated || !pushSupported) return;
   110→   110→    if (validatedRef.current) return;
   111→   111→    validatedRef.current = true;
   112→   112→
   113→   113→    let cancelled = false;
   114→   114→    const check = async () => {
   115→   115→      try {
   116→   116→        const deviceInfo = getDeviceInfo();
   117→   117→        setDeviceLabel(deviceInfo.deviceLabel);
   118→   118→
   119→   119→        // Check whether the browser still has permission + a local subscription.
   120→   120→        const hasBrowserPermission = Notification.permission === "granted";
   121→   121→        let hasLocalSubscription = false;
   122→   122→        if (hasBrowserPermission && typeof window.__ensurePushReady === "function") {
   123→   123→          try {
   124→   124→            const reg = await window.__ensurePushReady();
   125→   125→            if (reg) {
   126→   126→              const sub = await reg.pushManager.getSubscription();
   127→   127→              hasLocalSubscription = !!sub;
   128→   128→            }
   129→   129→          } catch {}
   130→   130→        }
   131→   131→
   132→   132→        // POST /api/device-registrations/validate
   133→   133→        const res = await fetch("/api/device-registrations/validate", {
   134→   134→          method: "POST",
   135→   135→          headers: { "Content-Type": "application/json" },
   136→   136→          body: JSON.stringify({
   137→   137→            deviceId: deviceInfo.deviceId,
   138→   138→            hasBrowserPermission,
   139→   139→            hasLocalSubscription,
   140→   140→          }),
   141→   141→        });
   142→   142→        if (!res.ok) return;
   143→   143→        const json = await res.json();
   144→   144→        const data = json.data;
   145→   145→        if (!data) return;
   146→   146→
   147→   147→        if (data.shouldShowWizard && !cancelled) {
   148→   148→          // Delay so it doesn't appear instantly on page load — gives the
   149→   149→          // page time to settle.
   150→   150→          setTimeout(() => {
   151→   151→            if (!cancelled) {
   152→   152→              setStep(data.reason === "incomplete" ? "welcome" : "welcome");
   153→   153→              setOpen(true);
   154→   154→            }
   155→   155→          }, 2000);
   156→   156→        }
   157→   157→      } catch {
   158→   158→        // Best-effort — if anything fails, just don't show the wizard.
   159→   159→      }
   160→   160→    };
   161→   161→
   162→   162→    check();
   163→   163→    return () => { cancelled = true; };
   164→   164→  }, [isAuthenticated, pushSupported]);
   165→   165→
   166→   166→  // ─── Skip handler ───────────────────────────────────────────────────────
   167→   167→  const handleSkip = useCallback(async () => {
   168→   168→    setBusy(true);
   169→   169→    try {
   170→   170→      const info = getDeviceInfo();
   171→   171→      await fetch("/api/device-registrations/skip", {
   172→   172→        method: "POST",
   173→   173→        headers: { "Content-Type": "application/json" },
   174→   174→        body: JSON.stringify({
   175→   175→          deviceId: info.deviceId,
   176→   176→          deviceLabel: info.deviceLabel,
   177→   177→          browserName: info.browserName,
   178→   178→          osName: info.osName,
   179→   179→          deviceType: info.deviceType,
   180→   180→        }),
   181→   181→      });
   182→   182→    } catch {}
   183→   183→    setOpen(false);
   184→   184→    setBusy(false);
   185→   185→  }, []);
   186→   186→
   187→   187→  // ─── Start the wizard: request permission ───────────────────────────────
   188→   188→  const handleStart = useCallback(async () => {
   189→   189→    setBusy(true);
   190→   190→    setStep("permission");
   191→   191→    try {
   192→   192→      const perm = await Notification.requestPermission();
   193→   193→      if (perm !== "granted") {
   194→   194→        setStep("denied");
   195→   195→        setBusy(false);
   196→   196→        return;
   197→   197→      }
   198→   198→      // Permission granted — proceed to registration.
   199→   199→      await runRegistration();
   200→   200→    } catch (e) {
   201→   201→      console.error("[wizard] permission request failed:", e);
   202→   202→      setStep("denied");
   203→   203→      setBusy(false);
   204→   204→    }
   205→   205→  }, []);
   206→   206→
   207→   207→  // ─── Registration: SW + subscription + backend + welcome push ───────────
   208→   208→  const runRegistration = useCallback(async () => {
   209→   209→    setBusy(true);
   210→   210→    setStep("registering");
   211→   211→
   212→   212→    const info = getDeviceInfo();
   213→   213→    const stepList: StepProgress[] = [
   214→   214→      { step: "sw", label: "Activating Service Worker", status: "pending" },
   215→   215→      { step: "subscription", label: "Creating Push Subscription", status: "pending" },
   216→   216→      { step: "backend", label: "Registering Device on Server", status: "pending" },
   217→   217→      { step: "test", label: "Sending Welcome Notification", status: "pending" },
   218→   218→    ];
   219→   219→    setSteps(stepList);
   220→   220→
   221→   221→    try {
   222→   222→      // Step 1: ensure SW is registered + active.
   223→   223→      stepList[0].status = "active";
   224→   224→      setSteps([...stepList]);
   225→   225→
   226→   226→      let reg: ServiceWorkerRegistration | null = null;
   227→   227→      if (typeof window.__ensurePushReady === "function") {
   228→   228→        reg = await window.__ensurePushReady();
   229→   229→      } else {
   230→   230→        reg = await navigator.serviceWorker.ready;
   231→   231→      }
   232→   232→      if (!reg) throw new Error("Service Worker activation failed");
   233→   233→      stepList[0].status = "done";
   234→   234→      setSteps([...stepList]);
   235→   235→
   236→   236→      // Step 2: create push subscription (or reuse existing).
   237→   237→      stepList[1].status = "active";
   238→   238→      setSteps([...stepList]);
   239→   239→
   240→   240→      let subscription = await reg.pushManager.getSubscription();
   241→   241→      if (!subscription) {
   242→   242→        const vapidRes = await fetch("/api/push/vapid-public");
   243→   243→        const vapidJson = await vapidRes.json();
   244→   244→        if (!vapidJson.data?.publicKey) {
   245→   245→          throw new Error("Push notifications are not configured on the server.");
   246→   246→        }
   247→   247→        const applicationServerKey = urlBase64ToUint8Array(vapidJson.data.publicKey);
   248→   248→        subscription = await reg.pushManager.subscribe({
   249→   249→          userVisibleOnly: true,
   250→   250→          applicationServerKey: applicationServerKey as BufferSource,
   251→   251→        });
   252→   252→      }
   253→   253→      stepList[1].status = "done";
   254→   254→      setSteps([...stepList]);
   255→   255→
   256→   256→      // Step 3: POST subscription to backend (saves PushSubscription + ensures
   257→   257→      // AppNotifPreference exists).
   258→   258→      stepList[2].status = "active";
   259→   259→      setSteps([...stepList]);
   260→   260→
   261→   261→      const sub = subscription.toJSON();
   262→   262→      const subscribeRes = await fetch("/api/push/subscribe", {
   263→   263→        method: "POST",
   264→   264→        headers: { "Content-Type": "application/json" },
   265→   265→        body: JSON.stringify({
   266→   266→          endpoint: sub.endpoint,
   267→   267→          keys: sub.keys,
   268→   268→          userAgent: info.userAgent.slice(0, 500),
   269→   269→        }),
   270→   270→      });
   271→   271→      if (!subscribeRes.ok) {
   272→   272→        const errData = await subscribeRes.json().catch(() => ({}));
   273→   273→        throw new Error(errData.error || "Failed to save subscription");
   274→   274→      }
   275→   275→      stepList[2].status = "done";
   276→   276→      setSteps([...stepList]);
   277→   277→
   278→   278→      // Step 4: register the device + send welcome push via the dedicated endpoint.
   279→   279→      stepList[3].status = "active";
   280→   280→      setSteps([...stepList]);
   281→   281→
   282→   282→      const registerRes = await fetch("/api/device-registrations/register", {
   283→   283→        method: "POST",
   284→   284→        headers: { "Content-Type": "application/json" },
   285→   285→        body: JSON.stringify({
   286→   286→          deviceId: info.deviceId,
   287→   287→          deviceLabel: info.deviceLabel,
   288→   288→          browserName: info.browserName,
   289→   289→          osName: info.osName,
   290→   290→          deviceType: info.deviceType,
   291→   291→          pushEndpoint: sub.endpoint,
   292→   292→        }),
   293→   293→      });
   294→   294→      if (!registerRes.ok) {
   295→   295→        const errData = await registerRes.json().catch(() => ({}));
   296→   296→        throw new Error(errData.error || "Failed to register device");
   297→   297→      }
   298→   298→      const regJson = await registerRes.json();
   299→   299→      const regData = regJson.data;
   300→   300→      const welcomed = regData?.welcomed === true;
   301→   301→      setTestPushSent(welcomed);
   302→   302→      stepList[3].status = welcomed ? "done" : "failed";
   303→   303→      setSteps([...stepList]);
   304→   304→
   305→   305→      // Invalidate preferences query so the toggle in settings reflects the
   306→   306→      // new "enabled" state.
   307→   307→      qc.invalidateQueries({ queryKey: ["customer", "app-notif-prefs"] });
   308→   308→
   309→   309→      // Move to the test result step.
   310→   310→      setStep("test_result");
   311→   311→      setBusy(false);
   312→   312→    } catch (e: any) {
   313→   313→      console.error("[wizard] registration failed:", e);
   314→   314→      // Mark any active step as failed.
   315→   315→      setSteps((prev) =>
   316→   316→        prev.map((s) =>
   317→   317→          s.status === "active" ? { ...s, status: "failed" } : s
   318→   318→        )
   319→   319→      );
   320→   320→      // Move to the failed step — shows retry + skip options.
   321→   321→      setStep("registering_failed");
   322→   322→      setBusy(false);
   323→   323→    }
   324→   324→  }, [qc]);
   325→   325→
   326→   326→  // ─── Done handler ───────────────────────────────────────────────────────
   327→   327→  const handleDone = useCallback(() => {
   328→   328→    setOpen(false);
   329→   329→    // Reset to welcome in case the wizard is ever re-opened.
   330→   330→    setTimeout(() => setStep("welcome"), 300);
   331→   331→  }, []);
   332→   332→
   333→   333→  // ─── Retry handler ──────────────────────────────────────────────────────
   334→   334→  const handleRetry = useCallback(async () => {
   335→   335→    await runRegistration();
   336→   336→  }, [runRegistration]);
   337→   337→
   338→   338→  if (!isAuthenticated || !pushSupported) return null;
   339→   339→
   340→   340→  return (
   341→   341→    <Dialog
   342→   342→      open={open}
   343→   343→      onOpenChange={(o) => {
   344→   344→        // Prevent closing while busy (registration in progress) — the user
   345→   345→        // must wait for the async flow to finish or fail.
   346→   346→        if (!o && busy) return;
   347→   347→        if (!o) handleSkip();
   348→   348→      }}
   349→   349→    >
   350→   350→      <DialogContent
   351→   351→        className="max-w-md overflow-hidden p-0 sm:rounded-2xl"
   352→   352→        onInteractOutside={(e) => e.preventDefault()}
   353→   353→        onEscapeKeyDown={(e) => {
   354→   354→          // Only allow escape when not busy + on the welcome/denied/done steps.
   355→   355→          if (busy || step === "registering" || step === "permission") {
   356→   356→            e.preventDefault();
   357→   357→          }
   358→   358→        }}
   359→   359→      >
   360→   360→        {/* ── Step: WELCOME ─────────────────────────────────────────────── */}
   361→   361→        {step === "welcome" && (
   362→   362→          <WelcomeStep
   363→   363→            deviceLabel={deviceLabel}
   364→   364→            onTurnOn={handleStart}
   365→   365→            onSkip={handleSkip}
   366→   366→            busy={busy}
   367→   367→          />
   368→   368→        )}
   369→   369→
   370→   370→        {/* ── Step: PERMISSION (waiting on browser prompt) ──────────────── */}
   371→   371→        {step === "permission" && (
   372→   372→          <CenteredMessage
   373→   373→            icon={<Bell className="size-8" />}
   374→   374→            title="Waiting for permission"
   375→   375→            message="Your browser is asking for permission to send notifications. Click 'Allow' to continue."
   376→   376→            spinner
   377→   377→          />
   378→   378→        )}
   379→   379→
   380→   380→        {/* ── Step: REGISTERING (progress) ──────────────────────────────── */}
   381→   381→        {step === "registering" && (
   382→   382→          <RegisteringStep steps={steps} />
   383→   383→        )}
   384→   384→
   385→   385→        {/* ── Step: REGISTERING FAILED ──────────────────────────────────── */}
   386→   386→        {step === "registering_failed" && (
   387→   387→          <RegisteringFailedStep
   388→   388→            steps={steps}
   389→   389→            onRetry={handleRetry}
   390→   390→            onSkip={handleSkip}
   391→   391→            busy={busy}
   392→   392→          />
   393→   393→        )}
   394→   394→
   395→   395→        {/* ── Step: TEST_RESULT ─────────────────────────────────────────── */}
   396→   396→        {step === "test_result" && (
   397→   397→          <TestResultStep
   398→   398→            sent={testPushSent}
   399→   399→            onContinue={() => setStep("done")}
   400→   400→            onRetry={handleRetry}
   401→   401→            busy={busy}
   402→   402→          />
   403→   403→        )}
   404→   404→
   405→   405→        {/* ── Step: DENIED (permission blocked) ─────────────────────────── */}
   406→   406→        {step === "denied" && (
   407→   407→          <DeniedStep onSkip={handleSkip} onRetry={handleStart} />
   408→   408→        )}
   409→   409→
   410→   410→        {/* ── Step: DONE ────────────────────────────────────────────────── */}
   411→   411→        {step === "done" && (
   412→   412→          <DoneStep onDone={handleDone} deviceLabel={deviceLabel} />
   413→   413→        )}
   414→   414→      </DialogContent>
   415→   415→    </Dialog>
   416→   416→  );
   417→   417→}

   799→   799→function urlBase64ToUint8Array(base64Url: string): Uint8Array {
   800→   800→  const padding = "=".repeat((4 - (base64Url.length % 4)) % 4);
   801→   801→  const base64 = (base64Url + padding).replace(/-/g, "+").replace(/_/g, "/");
   802→   802→  const raw = atob(base64);
   803→   803→  const arr = new Uint8Array(raw.length);
   804→   804→  for (let i = 0; i < raw.length; i++) {
   805→   805→    arr[i] = raw.charCodeAt(i);
   806→   806→  }
   807→   807→  return arr;
   808→   808→}
